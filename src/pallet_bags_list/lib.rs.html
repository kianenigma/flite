28//!  # Bags-List Pallet
29//!
30//! An onchain implementation of a semi-sorted linked list, with permissionless sorting and update
31//! operations.
32//!
33//! ## Pallet API
34//!
35//! See the [`pallet`] module for more information about the interfaces this pallet exposes,
36//! including its configuration trait, dispatchables, storage items, events and errors.
37//!
38//! This pallet provides an implementation of
39//! [`frame_election_provider_support::SortedListProvider`] and it can typically be used by another
40//! pallet via this API.
41//!
42//! ## Overview
43//!
44//! This pallet splits `AccountId`s into different bags. Within a bag, these `AccountId`s are stored
45//! as nodes in a linked-list manner. This pallet then provides iteration over all bags, which
46//! basically allows an infinitely large list of items to be kept in a sorted manner.
47//!
48//! Each bags has a upper and lower range of scores, denoted by [`Config::BagThresholds`]. All nodes
49//! within a bag must be within the range of the bag. If not, the permissionless [`Pallet::rebag`]
50//! can be used to move any node to the right bag.
51//!
52//! Once a `rebag` happens, the order within a node is still not enforced. To move a node to the
53//! optimal position in a bag, the [`Pallet::put_in_front_of`] or [`Pallet::put_in_front_of_other`]
54//! can be used.
55//!
56//! Additional reading, about how this pallet is used in the context of Polkadot's staking system:
57//! <https://polkadot.network/blog/staking-update-september-2021/#bags-list-in-depth>
58//!
59//! ## Examples
60//!
61//! See [`example`] for a diagram of `rebag` and `put_in_front_of` operations.
62//!
63//! ## Low Level / Implementation Details
64//!
65//! The data structure exposed by this pallet aims to be optimized for:
66//!
67//! - insertions and removals.
68//! - iteration over the top* N items by score, where the precise ordering of items doesn't
69//!   particularly matter.
70//!
71//! ### Further Details
72//!
73//! - items are kept in bags, which are delineated by their range of score (See
74//!   [`Config::BagThresholds`]).
75//! - for iteration, bags are chained together from highest to lowest and elements within the bag
76//!   are iterated from head to tail.
77//! - items within a bag are iterated in order of insertion. Thus removing an item and re-inserting
78//!   it will worsen its position in list iteration; this reduces incentives for some types of spam
79//!   that involve consistently removing and inserting for better position. Further, ordering
80//!   granularity is thus dictated by range between each bag threshold.
81//! - if an item's score changes to a value no longer within the range of its current bag the item's
82//!   position will need to be updated by an external actor with rebag (update), or removal and
83//!   insertion.

178	#[pallet::config]
179	pub trait Config<I: 'static = ()>: frame_system::Config {
180		/// The overarching event type.
181		type RuntimeEvent: From<Event<Self, I>>
182			+ IsType<<Self as frame_system::Config>::RuntimeEvent>;
183
184		/// Weight information for extrinsics in this pallet.
185		type WeightInfo: weights::WeightInfo;
186
187		/// Something that provides the scores of ids.
188		type ScoreProvider: ScoreProvider<Self::AccountId, Score = Self::Score>;
189
190		/// The list of thresholds separating the various bags.
191		///
192		/// Ids are separated into unsorted bags according to their score. This specifies the
193		/// thresholds separating the bags. An id's bag is the largest bag for which the id's score
194		/// is less than or equal to its upper threshold.
195		///
196		/// When ids are iterated, higher bags are iterated completely before lower bags. This means
197		/// that iteration is _semi-sorted_: ids of higher score tend to come before ids of lower
198		/// score, but peer ids within a particular bag are sorted in insertion order.
199		///
200		/// # Expressing the constant
201		///
202		/// This constant must be sorted in strictly increasing order. Duplicate items are not
203		/// permitted.
204		///
205		/// There is an implied upper limit of `Score::MAX`; that value does not need to be
206		/// specified within the bag. For any two threshold lists, if one ends with
207		/// `Score::MAX`, the other one does not, and they are otherwise equal, the two
208		/// lists will behave identically.
209		///
210		/// # Calculation
211		///
212		/// It is recommended to generate the set of thresholds in a geometric series, such that
213		/// there exists some constant ratio such that `threshold[k + 1] == (threshold[k] *
214		/// constant_ratio).max(threshold[k] + 1)` for all `k`.
215		///
216		/// The helpers in the `/utils/frame/generate-bags` module can simplify this calculation.
217		///
218		/// # Examples
219		///
220		/// - If `BagThresholds::get().is_empty()`, then all ids are put into the same bag, and
221		///   iteration is strictly in insertion order.
222		/// - If `BagThresholds::get().len() == 64`, and the thresholds are determined according to
223		///   the procedure given above, then the constant ratio is equal to 2.
224		/// - If `BagThresholds::get().len() == 200`, and the thresholds are determined according to
225		///   the procedure given above, then the constant ratio is approximately equal to 1.248.
226		/// - If the threshold list begins `[1, 2, 3, ...]`, then an id with score 0 or 1 will fall
227		///   into bag 0, an id with score 2 will fall into bag 1, etc.
228		///
229		/// # Migration
230		///
231		/// In the event that this list ever changes, a copy of the old bags list must be retained.
232		/// With that `List::migrate` can be called, which will perform the appropriate migration.
233		#[pallet::constant]
234		type BagThresholds: Get<&'static [Self::Score]>;
235
236		/// The type used to dictate a node position relative to other nodes.
237		type Score: Clone
238			+ Default
239			+ PartialEq
240			+ Eq
241			+ Ord
242			+ PartialOrd
243			+ core::fmt::Debug
244			+ Copy
245			+ AtLeast32BitUnsigned
246			+ Bounded
247			+ TypeInfo
248			+ FullCodec
249			+ MaxEncodedLen;
250	}

287	#[pallet::call]
288	impl<T: Config<I>, I: 'static> Pallet<T, I> {
289		/// Declare that some `dislocated` account has, through rewards or penalties, sufficiently
290		/// changed its score that it should properly fall into a different bag than its current
291		/// one.
292		///
293		/// Anyone can call this function about any potentially dislocated account.
294		///
295		/// Will always update the stored score of `dislocated` to the correct score, based on
296		/// `ScoreProvider`.
297		///
298		/// If `dislocated` does not exists, it returns an error.
299		#[pallet::call_index(0)]
300		#[pallet::weight(T::WeightInfo::rebag_non_terminal().max(T::WeightInfo::rebag_terminal()))]
301		pub fn rebag(origin: OriginFor<T>, dislocated: AccountIdLookupOf<T>) -> DispatchResult {
302			ensure_signed(origin)?;
303			let dislocated = T::Lookup::lookup(dislocated)?;
304			let current_score = T::ScoreProvider::score(&dislocated);
305			let _ = Pallet::<T, I>::do_rebag(&dislocated, current_score)
306				.map_err::<Error<T, I>, _>(Into::into)?;
307			Ok(())
308		}
309
310		/// Move the caller's Id directly in front of `lighter`.
311		///
312		/// The dispatch origin for this call must be _Signed_ and can only be called by the Id of
313		/// the account going in front of `lighter`. Fee is payed by the origin under all
314		/// circumstances.
315		///
316		/// Only works if:
317		///
318		/// - both nodes are within the same bag,
319		/// - and `origin` has a greater `Score` than `lighter`.
320		#[pallet::call_index(1)]
321		#[pallet::weight(T::WeightInfo::put_in_front_of())]
322		pub fn put_in_front_of(
323			origin: OriginFor<T>,
324			lighter: AccountIdLookupOf<T>,
325		) -> DispatchResult {
326			let heavier = ensure_signed(origin)?;
327			let lighter = T::Lookup::lookup(lighter)?;
328			List::<T, I>::put_in_front_of(&lighter, &heavier)
329				.map_err::<Error<T, I>, _>(Into::into)
330				.map_err::<DispatchError, _>(Into::into)
331		}
332
333		/// Same as [`Pallet::put_in_front_of`], but it can be called by anyone.
334		///
335		/// Fee is paid by the origin under all circumstances.
336		#[pallet::call_index(2)]
337		#[pallet::weight(T::WeightInfo::put_in_front_of())]
338		pub fn put_in_front_of_other(
339			origin: OriginFor<T>,
340			heavier: AccountIdLookupOf<T>,
341			lighter: AccountIdLookupOf<T>,
342		) -> DispatchResult {
343			let _ = ensure_signed(origin)?;
344			let lighter = T::Lookup::lookup(lighter)?;
345			let heavier = T::Lookup::lookup(heavier)?;
346			List::<T, I>::put_in_front_of(&lighter, &heavier)
347				.map_err::<Error<T, I>, _>(Into::into)
348				.map_err::<DispatchError, _>(Into::into)
349		}
350	}

376impl<T: Config<I>, I: 'static> Pallet<T, I> {
377	/// Move an account from one bag to another, depositing an event on success.
378	///
379	/// If the account changed bags, returns `Ok(Some((from, to)))`.
380	pub fn do_rebag(
381		account: &T::AccountId,
382		new_score: T::Score,
383	) -> Result<Option<(T::Score, T::Score)>, ListError> {
384		// If no voter at that node, don't do anything. the caller just wasted the fee to call this.
385		let node = list::Node::<T, I>::get(&account).ok_or(ListError::NodeNotFound)?;
386		let maybe_movement = List::update_position_for(node, new_score);
387		if let Some((from, to)) = maybe_movement {
388			Self::deposit_event(Event::<T, I>::Rebagged { who: account.clone(), from, to });
389		};
390		Self::deposit_event(Event::<T, I>::ScoreUpdated { who: account.clone(), new_score });
391		Ok(maybe_movement)
392	}

462	frame_election_provider_support::runtime_benchmarks_enabled! {
463		fn score_update_worst_case(who: &T::AccountId, is_increase: bool) -> Self::Score {
464			use frame_support::traits::Get as _;
465			let thresholds = T::BagThresholds::get();
466			let node = list::Node::<T, I>::get(who).unwrap();
467			let current_bag_idx = thresholds
468				.iter()
469				.chain(core::iter::once(&T::Score::max_value()))
470				.position(|w| w == &node.bag_upper)
471				.unwrap();
472
473			if is_increase {
474				let next_threshold_idx = current_bag_idx + 1;
475				assert!(thresholds.len() > next_threshold_idx);
476				thresholds[next_threshold_idx]
477			} else {
478				assert!(current_bag_idx != 0);
479				let prev_threshold_idx = current_bag_idx - 1;
480				thresholds[prev_threshold_idx]
481			}
482		}
483	}
