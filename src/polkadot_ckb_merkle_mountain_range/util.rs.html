20impl<T: Clone> MMRStoreReadOps<T> for &MemStore<T> {
21    fn get_elem(&self, pos: u64) -> Result<Option<T>> {
22        Ok(self.0.borrow().get(&pos).cloned())
23    }
24}
25
26impl<T> MMRStoreWriteOps<T> for &MemStore<T> {
27    fn append(&mut self, pos: u64, elems: Vec<T>) -> Result<()> {
28        let mut store = self.0.borrow_mut();
29        for (i, elem) in elems.into_iter().enumerate() {
30            store.insert(pos + i as u64, elem);
31        }
32        Ok(())
33    }
34}

38pub trait VeqDequeExt<T: Ord> {

42impl<T: Ord> VeqDequeExt<T> for VecDeque<T> {
43    fn insert_sorted(&mut self, value: T) {
44        match self.binary_search(&value) {
45            Ok(_pos) => {
46                // element already in vector @ `pos`
47            }
48            Err(pos) => self.insert(pos, value),
49        }
50    }
