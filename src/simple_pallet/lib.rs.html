32
33	#[pallet::call]
34	impl<T: Config> Pallet<T> {
35		#[docify::export]
36		pub fn dispatching(origin: OriginFor<T>) -> DispatchResult {
37			// Yes, the account id type is no longer generic, because `flite` uses constrained
38			// associated types.
39			let _sender: flite::types::AccountId = ensure_signed(origin)?;
40
41			Ok(())
42		}
43
44		#[docify::export]
45		/// simple-pallet can use the very very basic flite currency primitives.
46		pub fn simple_currency_usage(
47			origin: OriginFor<T>,
48			to: flite::types::AccountId,
49			amount: u128, // aka. flite::types::Balance
50		) -> DispatchResult {
51			let who: flite::types::AccountId = ensure_signed(origin)?;
52
53			// read. Yes, the balance types is just `u128`
54			let _: u128 = flite::native_currency::Simple::<T>::balance_of(&who);
55			// transfer
56			flite::native_currency::Simple::<T>::transfer(&who, &to, amount)?;
57			// 'lock', which is an abstraction over holding with more opinions
58			flite::native_currency::Simple::<T>::lock(&who, amount)?;
59
60			Ok(())
61		}
62
63		/// Or it can tap into the default FRAME currency, although as constrained by flite. For
64		/// example, `Balance` is still `u128`, `RuntimeHoldReason` is still `&static str` adn so
65		/// on.
66		#[docify::export]
67		pub fn advance_currency_usage(
68			origin: OriginFor<T>,
69			to: flite::types::AccountId,
70			amount: flite::types::Balance,
71		) -> DispatchResult {
72			use frame::traits::fungible::*;
73			let who: flite::types::AccountId = ensure_signed(origin)?;
74
75			// read
76			let _: flite::types::Balance = T::AdvanceCurrency::balance(&who);
77			// transfer
78			T::AdvanceCurrency::transfer(
79				&who,
80				&to,
81				amount,
82				frame::traits::tokens::Preservation::Expendable,
83			)?;
84			// 'hold'.
85			T::AdvanceCurrency::hold(&0u8, &who, amount)?;
86
87			Ok(())
88		}
89	}
