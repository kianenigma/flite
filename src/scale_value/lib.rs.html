16/*!
17This crate exposes the [`Value`] type and related subtypes, which are used as the runtime
18representations of SCALE encoded data (much like `serde_json::Value` is a runtime representation
19of JSON data).
20
21[`Value`]'s can be:
22
23- Encoded and decoded from SCALE bytes via [`::scale_encode::EncodeAsType`] and [`::scale_decode::DecodeAsType`]
24  traits (or by calling [`crate::scale::decode_as_type`] and [`crate::scale::encode_as_type`]).
25- Parsed to and from strings by calling [`crate::stringify::from_str`] and [`crate::stringify::to_string`]).
26  Parsing from strings requires the `from_string` feature to be enabled.
27- Serialized and deserialized via `serde` traits (for example, to and from JSON). They can also be serialized
28  from and to other types with the relevant serde derives on. These require the `serde` feature to be enabled.
29- Accessed ergonomically via the [`At`] trait.
30*/

57    /// Attempt to convert a [`crate::Value`] into another type via serde.
58    ///
59    /// # Examples
60    ///
61    /// Use serde to convert a value into a built-in type:
62    ///
63    /// ```rust
64    /// use scale_value::Value;
65    ///
66    /// let value = Value::unnamed_composite(vec![
67    ///     Value::u128(1),
68    ///     Value::u128(2),
69    ///     Value::u128(3),
70    /// ]);
71    ///
72    /// let arr: [u8; 3] = scale_value::serde::from_value(value).unwrap();
73    /// ```
74    ///
75    /// Converting values to a custom type:
76    ///
77    /// ```rust
78    /// use scale_value::Value;
79    /// use serde::{ Serialize, Deserialize };
80    ///
81    /// #[derive(Serialize, Deserialize, PartialEq, Debug)]
82    /// enum Foo {
83    ///     A { is_valid: bool, name: String },
84    ///     B(u8, bool)
85    /// }
86    ///
87    /// let value1 = Value::named_variant("A", [
88    ///     ("name", Value::string("James")),
89    ///     ("is_valid", Value::bool(true)),
90    /// ]);
91    /// let foo1: Foo = scale_value::serde::from_value(value1).unwrap();
92    /// assert_eq!(foo1, Foo::A { is_valid: true, name: "James".into() });
93    ///
94    /// let value2 = Value::unnamed_variant("B", [
95    ///     Value::u128(123),
96    ///     Value::bool(true),
97    /// ]);
98    /// let foo2: Foo = scale_value::serde::from_value(value2).unwrap();
99    /// assert_eq!(foo2, Foo::B(123, true));
100    /// ```
101    pub fn from_value<'de, Ctx, T: serde::Deserialize<'de>>(
102        value: crate::Value<Ctx>,
103    ) -> Result<T, DeserializerError> {
104        T::deserialize(value)
105    }

107    /// Attempt to convert some type into a [`crate::Value`] via serde.
108    ///
109    /// # Examples
110    ///
111    /// Convert a built-in array of values into a [`crate::Value`]:
112    ///
113    /// ```rust
114    /// use scale_value::Value;
115    ///
116    /// let arr = [1u8, 2u8, 3u8];
117    ///
118    /// let val = scale_value::serde::to_value(arr).unwrap();
119    /// assert_eq!(val, Value::unnamed_composite([
120    ///     Value::u128(1),
121    ///     Value::u128(2),
122    ///     Value::u128(3),
123    /// ]));
124    /// ```
125    ///
126    /// Converting some custom type to a [`crate::Value`]:
127    ///
128    /// ```rust
129    /// use scale_value::Value;
130    /// use serde::{ Serialize, Deserialize };
131    ///
132    /// #[derive(Serialize, Deserialize, PartialEq, Debug)]
133    /// enum Foo {
134    ///     A { is_valid: bool, name: String },
135    ///     B(u8, bool)
136    /// }
137    ///
138    /// let foo = Foo::A { is_valid: true, name: "James".into() };
139    ///
140    /// let value = scale_value::serde::to_value(foo).unwrap();
141    /// assert_eq!(value, Value::named_variant("A", [
142    ///     ("is_valid", Value::bool(true)),
143    ///     ("name", Value::string("James")),
144    /// ]));
145    /// ```
146    pub fn to_value<T: serde::Serialize>(ty: T) -> Result<crate::Value<()>, SerializerError> {
147        ty.serialize(ValueSerializer)
148    }

151/// Encoding and decoding SCALE bytes into a [`crate::Value`].
152///
153/// # Exmaple
154///
155/// Given some known metadata type ID, encode and decode some [`crate::Value`]
156/// to SCALE bytes.
157///
158/// ```rust
159/// # fn make_type<T: scale_info::TypeInfo + 'static>() -> (u32, scale_info::PortableRegistry) {
160/// #     let m = scale_info::MetaType::new::<T>();
161/// #     let mut types = scale_info::Registry::new();
162/// #     let id = types.register_type(&m);
163/// #     let portable_registry: scale_info::PortableRegistry = types.into();
164/// #     (id.id(), portable_registry)
165/// # }
166/// # let (type_id, registry) = make_type::<Foo>();
167/// use scale_value::Value;
168///
169/// // Imagine we have a `registry` (of type [`scale_info::PortableRegistry`]) containing this type,
170/// // and a `type_id` (a `u32`) pointing to it in the registry.
171/// #[derive(scale_info::TypeInfo)]
172/// enum Foo {
173///     A { is_valid: bool, name: String }
174/// }
175///
176/// // Given that, we can encode/decode something with that shape to/from SCALE bytes:
177/// let value = Value::named_variant("A", [
178///     ("is_valid", Value::bool(true)),
179///     ("name", Value::string("James")),
180/// ]);
181///
182/// // Encode the Value to bytes:
183/// let mut bytes = Vec::new();
184/// scale_value::scale::encode_as_type(&value, type_id, &registry, &mut bytes).unwrap();
185///
186/// // Decode the bytes back into a matching Value.
187/// // This value contains contextual information about which type was used
188/// // to decode each part of it, which we can throw away with `.remove_context()`.
189/// let new_value = scale_value::scale::decode_as_type(&mut &*bytes, type_id, &registry).unwrap();
190///
191/// assert_eq!(value, new_value.remove_context());
192/// ```
193pub mod scale {
194    use crate::prelude::*;
195    use scale_decode::FieldIter;
196    use scale_encode::EncodeAsType;
197
198    pub use crate::scale_impls::{DecodeError, ValueVisitor};
199    pub use scale_encode::Error as EncodeError;
200    pub use scale_info::PortableRegistry;
201    pub use scale_type_resolver::TypeResolver;
202
203    /// Attempt to decode some SCALE encoded bytes into a value, by providing a pointer
204    /// to the bytes (which will be moved forwards as bytes are used in the decoding),
205    /// a type ID, and a type registry from which we'll look up the relevant type information.
206    pub fn decode_as_type<R>(
207        data: &mut &[u8],
208        ty_id: R::TypeId,
209        types: &R,
210    ) -> Result<crate::Value<R::TypeId>, DecodeError>
211    where
212        R: TypeResolver,
213        R::TypeId: Clone,
214    {
215        crate::scale_impls::decode_value_as_type(data, ty_id, types)
216    }
217
218    /// Attempt to decode some SCALE encoded bytes into multiple values, by providing a pointer
219    /// to the bytes (which will be moved forwards as bytes are used in the decoding),
220    /// and an iterator of fields, where each field contains a type ID and optionally a field name.
221    pub fn decode_as_fields<'resolver, R>(
222        input: &mut &[u8],
223        fields: &mut dyn FieldIter<'resolver, R::TypeId>,
224        types: &'resolver R,
225    ) -> Result<crate::Composite<R::TypeId>, DecodeError>
226    where
227        R: TypeResolver,
228        R::TypeId: Clone,
229    {
230        crate::scale_impls::decode_composite_as_fields(input, fields, types)
231    }
232
233    /// Attempt to encode some [`crate::Value<T>`] into SCALE bytes, by providing a pointer to the
234    /// type ID that we'd like to encode it as, a type registry from which we'll look
235    /// up the relevant type information, and a buffer to encode the bytes to.
236    pub fn encode_as_type<R: TypeResolver, T>(
237        value: &crate::Value<T>,
238        ty_id: R::TypeId,
239        types: &R,
240        buf: &mut Vec<u8>,
241    ) -> Result<(), EncodeError> {
242        value.encode_as_type_to(ty_id, types, buf)
243    }
244
245    /// A visitor and function to decode some bytes into a [`crate::Value`] while tracing the current
246    /// decoding state so that a more detailed error can be returned in the event of a failure.
247    pub mod tracing {
248        pub use crate::scale_impls::{TraceDecodingError, TraceDecodingVisitor};
249
250        /// Decode a value using the [`TraceDecodingVisitor`], which internally keeps track of the current decoding state, and as
251        /// a result hands back a much more detailed error than [`crate::scale::decode_as_type()`] if decoding fails.
252        ///
253        /// One approach is to use the standard visitor for decoding on the "happy path", and if you need more information about
254        /// the decode error, to try decoding the same bytes again using this function to obtain more information about what failed.
255        pub fn decode_as_type<R>(
256            data: &mut &[u8],
257            ty_id: R::TypeId,
258            types: &R,
259        ) -> Result<crate::Value<R::TypeId>, TraceDecodingError<crate::Value<R::TypeId>>>
260        where
261            R: scale_type_resolver::TypeResolver,
262        {
263            scale_decode::visitor::decode_with_visitor(
264                data,
265                ty_id,
266                types,
267                TraceDecodingVisitor::new(),
268            )
269        }
270    }
271}

300    /// Attempt to parse a string into a [`crate::Value<()>`], returning a tuple
301    /// consisting of a result (either the value or a [`ParseError`] containing
302    /// location and error information) and the remainder of the string that wasn't
303    /// parsed.
304    ///
305    /// # Examples
306    ///
307    /// ```rust
308    /// use scale_value::Value;
309    ///
310    /// fn to_value(str: &str) -> Value {
311    ///     scale_value::stringify::from_str(str).0.unwrap()
312    /// }
313    ///
314    /// // Primitive values:
315    /// assert_eq!(to_value("1"), Value::u128(1));
316    /// assert_eq!(to_value("-1"), Value::i128(-1));
317    /// assert_eq!(to_value("true"), Value::bool(true));
318    /// assert_eq!(to_value("'a'"), Value::char('a'));
319    /// assert_eq!(to_value("\"hi\""), Value::string("hi"));
320    ///
321    /// // Named composite values look a bit like rust structs:
322    /// let value = to_value("{ a: true, b: \"hello\" }");
323    /// assert_eq!(
324    ///     value,
325    ///     Value::named_composite(vec![
326    ///         ("a", Value::bool(true)),
327    ///         ("b", Value::string("hello"))
328    ///     ])
329    /// );
330    ///
331    /// // Unnamed composite values look a bit like rust tuples:
332    /// let value = to_value("(true, \"hello\")");
333    /// assert_eq!(
334    ///     value,
335    ///     Value::unnamed_composite(vec![
336    ///         Value::bool(true),
337    ///         Value::string("hello")
338    ///     ])
339    /// );
340    ///
341    /// // Variant values (named or unnamed) are just the above with a variant name
342    /// // prefixed:
343    /// let value = to_value("MyVariant { a: true, b: \"hello\" }");
344    /// assert_eq!(
345    ///     value,
346    ///     Value::named_variant(
347    ///         "MyVariant",
348    ///         vec![
349    ///             ("a", Value::bool(true)),
350    ///             ("b", Value::string("hello"))
351    ///         ]
352    ///     )
353    /// );
354    ///
355    /// // Bit sequences can be encoded from unnamed composites, but we have a
356    /// // compact syntax for them too:
357    /// assert_eq!(
358    ///     to_value("<0101>"),
359    ///     Value::bit_sequence(scale_bits::Bits::from_iter([false, true, false, true]))
360    /// );
361    /// ```
362    #[cfg(feature = "from-string")]
363    pub fn from_str(s: &str) -> (Result<crate::Value<()>, ParseError>, &str) {
364        crate::string_impls::FromStrBuilder::new().parse(s)
365    }

367    /// This is similar to [`from_str`], except that it returns a [`FromStrBuilder`],
368    /// which allows for some additional configuration in how strings are parsed.
369    ///
370    /// # Example
371    ///
372    /// ```rust
373    /// # // Example only runs when parser-ss58 feature is enabled:
374    /// # #[cfg(not(feature = "parser-ss58"))]
375    /// # fn main() {}
376    /// # #[cfg(feature = "parser-ss58")]
377    /// # fn main() {
378    /// #
379    /// use scale_value::Value;
380    /// use scale_value::stringify::custom_parsers;
381    ///
382    /// fn to_value(str: &str) -> Value {
383    ///     scale_value::stringify::from_str_custom()
384    ///         // You can write your own custom parser, but for
385    ///         // this example, we just use some provided ones.
386    ///         .add_custom_parser(custom_parsers::parse_hex)
387    ///         // Requires the parser-ss58 feature:
388    ///         .add_custom_parser(custom_parsers::parse_ss58)
389    ///         .parse(str)
390    ///         .0
391    ///         .unwrap()
392    /// }
393    ///
394    /// // Hex strings will now be parsed into unnamed composite types
395    /// let value = to_value("(1,2,0x030405)");
396    /// assert_eq!(
397    ///     value,
398    ///     Value::unnamed_composite(vec![
399    ///         Value::u128(1),
400    ///         Value::u128(2),
401    ///         Value::unnamed_composite(vec![
402    ///             Value::u128(3),
403    ///             Value::u128(4),
404    ///             Value::u128(5),
405    ///         ])
406    ///     ])
407    /// );
408    ///
409    /// // ss58 addresses will also become unnamed composite types
410    /// let value = to_value(r#"{
411    ///     name: "Alice",
412    ///     address: 5FHneW46xGXgs5mUiveU4sbTyGBzmstUspZC92UhjJM694ty
413    /// }"#);
414    ///
415    /// // Manually obtain and decode the hex value for the address:
416    /// let addr: Vec<_> = hex::decode("8eaf04151687736326c9fea17e25fc5287613693c912909cb226aa4794f26a48")
417    ///     .unwrap()
418    ///     .into_iter()
419    ///     .map(|b| Value::u128(b as u128))
420    ///     .collect();
421    ///
422    /// assert_eq!(
423    ///     value,
424    ///     Value::named_composite(vec![
425    ///         ("name", Value::string("Alice")),
426    ///         ("address", Value::unnamed_composite(addr))
427    ///     ])
428    /// )
429    /// #
430    /// # }
431    /// ```
432    #[cfg(feature = "from-string")]
433    pub fn from_str_custom() -> FromStrBuilder {
434        crate::string_impls::FromStrBuilder::new()
435    }

473    /// Format a [`crate::Value`] to a string. Several options can be configured in the
474    /// process, such as the indentation, custom formatters, printing of context, and
475    /// style (compact or spaced).
476    ///
477    /// # Example
478    ///
479    /// ```rust
480    /// use scale_value::{Value, ValueDef, Primitive, value};
481    /// use scale_value::stringify::custom_formatters::format_hex;
482    /// use core::fmt::Write;
483    ///
484    /// let value = value!({
485    ///     foo: true,
486    ///     bar: (1u8,2u8,3u8,4u8)
487    /// });
488    ///
489    /// let mut s = String::new();
490    ///
491    /// fn capitalise_bools<T, W: Write>(v: &Value<T>, w: &mut W) -> Option<core::fmt::Result> {
492    ///     if let ValueDef::Primitive(Primitive::Bool(b)) = &v.value {
493    ///         match b {
494    ///             true => Some(write!(w, "TRUE")),
495    ///             false => Some(write!(w, "FALSE"))
496    ///         }
497    ///     } else {
498    ///         None
499    ///     }
500    /// }
501    ///
502    /// scale_value::stringify::to_writer_custom()
503    ///     .compact()
504    ///     .add_custom_formatter(|v, w| format_hex(v, w))
505    ///     .add_custom_formatter(|v, w| capitalise_bools(v, w))
506    ///     .write(&value, &mut s);
507    ///
508    /// assert_eq!(s, r#"{foo:TRUE,bar:0x01020304}"#)
509    /// ```
510    pub fn to_writer_custom<T, W: core::fmt::Write>() -> ToWriterBuilder<T, W> {
511        crate::string_impls::ToWriterBuilder::new()
512    }
