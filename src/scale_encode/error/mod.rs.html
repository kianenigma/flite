16//! An error that is emitted whenever some encoding fails.
17mod context;
18
19use alloc::{borrow::Cow, boxed::Box, string::String};
20use core::fmt::Display;
21
22pub use context::{Context, Location};
23
24/// An error produced while attempting to encode some type.
25#[derive(Debug)]
26pub struct Error {
27    context: Context,
28    kind: ErrorKind,
29}
30
31#[cfg(feature = "std")]
32impl std::error::Error for Error {}
33
34impl Error {
35    /// Construct a new error given an error kind.
36    pub fn new(kind: ErrorKind) -> Error {
37        Error {
38            context: Context::new(),
39            kind,
40        }
41    }
42    /// Construct a new, custom error.
43    pub fn custom(error: impl CustomError) -> Error {
44        Error::new(ErrorKind::Custom(Box::new(error)))
45    }
46    /// Construct a custom error from a static string.
47    pub fn custom_str(error: &'static str) -> Error {
48        #[derive(derive_more::Display, Debug)]
49        pub struct StrError(pub &'static str);
50        #[cfg(feature = "std")]
51        impl std::error::Error for StrError {}
52
53        Error::new(ErrorKind::Custom(Box::new(StrError(error))))
54    }
55    /// Construct a custom error from an owned string.
56    pub fn custom_string(error: String) -> Error {
57        #[derive(derive_more::Display, Debug)]
58        pub struct StringError(String);
59        #[cfg(feature = "std")]
60        impl std::error::Error for StringError {}
61
62        Error::new(ErrorKind::Custom(Box::new(StringError(error))))
63    }
64    /// Retrieve more information about what went wrong.
65    pub fn kind(&self) -> &ErrorKind {
66        &self.kind
67    }
68    /// Retrieve details about where the error occurred.
69    pub fn context(&self) -> &Context {
70        &self.context
71    }
72    /// Give some context to the error.
73    pub fn at(mut self, loc: Location) -> Self {
74        self.context.push(loc);
75        Error {
76            context: self.context,
77            kind: self.kind,
78        }
79    }
80    /// Note which sequence index the error occurred in.
81    pub fn at_idx(mut self, idx: usize) -> Self {
82        self.context.push(Location::idx(idx));
83        Error {
84            context: self.context,
85            kind: self.kind,
86        }
87    }
88    /// Note which field the error occurred in.
89    pub fn at_field(mut self, field: impl Into<Cow<'static, str>>) -> Self {
90        self.context.push(Location::field(field));
91        Error {
92            context: self.context,
93            kind: self.kind,
94        }
95    }
96    /// Note which variant the error occurred in.
97    pub fn at_variant(mut self, variant: impl Into<Cow<'static, str>>) -> Self {
98        self.context.push(Location::variant(variant));
99        Error {
100            context: self.context,
101            kind: self.kind,
102        }
103    }
104}
105
106impl Display for Error {
107    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
108        let path = self.context.path();
109        let kind = &self.kind;
110        write!(f, "Error at {path}: {kind}")
111    }
112}

114/// The underlying nature of the error.
115#[derive(Debug, derive_more::From, derive_more::Display)]
116pub enum ErrorKind {
117    /// There was an error resolving the type via the given [`crate::TypeResolver`].
118    #[display("Failed to resolve type: {_0}")]
119    TypeResolvingError(String),
120    /// Cannot find a given type.
121    #[display("Cannot find type with identifier {_0}")]
122    TypeNotFound(String),
123    /// Cannot encode the actual type given into the target type ID.
124    #[display("Cannot encode {actual:?} into type with ID {expected_id}")]
125    WrongShape {
126        /// The actual kind we have to encode
127        actual: Kind,
128        /// Identifier for the expected type
129        expected_id: String,
130    },
131    /// The types line up, but the expected length of the target type is different from the length of the input value.
132    #[display("Cannot encode to type; expected length {expected_len} but got length {actual_len}")]
133    WrongLength {
134        /// Length we have
135        actual_len: usize,
136        /// Length expected for type.
137        expected_len: usize,
138    },
139    /// We cannot encode the number given into the target type; it's out of range.
140    #[display("Number {value} is out of range for target type with identifier {expected_id}")]
141    NumberOutOfRange {
142        /// A string represenatation of the numeric value that was out of range.
143        value: String,
144        /// Identifier for the expected numeric type that we tried to encode it to.
145        expected_id: String,
146    },
147    /// Cannot find a variant with a matching name on the target type.
148    #[display("Variant {name} does not exist on type with identifier {expected_id}")]
149    CannotFindVariant {
150        /// Variant name we can't find in the expected type.
151        name: String,
152        /// Identifier for the expected type.
153        expected_id: String,
154    },
155    /// Cannot find a field on our source type that's needed for the target type.
156    #[display("Field {name} does not exist in our source struct")]
157    CannotFindField {
158        /// Name of the field which was not provided.
159        name: String,
160    },
161    /// A custom error.
162    #[from]
163    #[display("Custom error: {_0}")]
164    Custom(Box<dyn CustomError>),
165}
166
167/// Anything implementing this trait can be used in [`ErrorKind::Custom`].
168#[cfg(feature = "std")]
169pub trait CustomError: std::error::Error + Send + Sync + 'static {}
170#[cfg(feature = "std")]
171impl<T: std::error::Error + Send + Sync + 'static> CustomError for T {}
172
173/// Anything implementing this trait can be used in [`ErrorKind::Custom`].
174#[cfg(not(feature = "std"))]
175pub trait CustomError: core::fmt::Debug + core::fmt::Display + Send + Sync + 'static {}
176#[cfg(not(feature = "std"))]
177impl<T: core::fmt::Debug + core::fmt::Display + Send + Sync + 'static> CustomError for T {}

179/// The kind of type that we're trying to encode.
180#[allow(missing_docs)]
181#[derive(Copy, Clone, PartialEq, Eq, Debug)]
182pub enum Kind {
183    Struct,
184    Tuple,
185    Variant,
186    Array,
187    BitSequence,
188    Bool,
189    Char,
190    Str,
191    Number,
192}
