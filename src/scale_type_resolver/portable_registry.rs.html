22/// An error that can occur when we try to encode or decode to a SCALE bit sequence type.
23#[derive(Debug, Clone, PartialEq, Eq)]
24pub enum Error {
25    /// The registry did not contain the bit order type listed.
26    OrderFormatNotFound(u32),
27    /// The registry did not contain the bit store type listed.
28    StoreFormatNotFound(u32),
29    /// The bit order type did not have a valid identifier/name.
30    NoBitOrderIdent,
31    /// The bit store type that we found was not what we expected (a primitive u8/u16/u32/u64).
32    UnsupportedBitStoreFormatEncountered,
33    /// The bit order type name that we found was not what we expected ("Lsb0" or "Msb0").
34    UnsupportedBitOrderFormatEncountered,
35}
36
37impl core::fmt::Display for Error {
38    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
39        match self {
40            Error::OrderFormatNotFound(n) => {
41                write!(f, "Bit order type {n} not found in registry")
42            }
43            Error::StoreFormatNotFound(n) => {
44                write!(f, "Bit store type {n} not found in registry")
45            }
46            Error::NoBitOrderIdent => {
47                write!(f, "Bit order cannot be identified")
48            }
49            Error::UnsupportedBitStoreFormatEncountered => {
50                write!(f, "Unsupported bit store format encountered")
51            }
52            Error::UnsupportedBitOrderFormatEncountered => {
53                write!(f, "Unsupported bit order format encountered")
54            }
55        }
56    }
57}

59impl TypeResolver for PortableRegistry {
60    type TypeId = u32;
61    type Error = Error;
62
63    fn resolve_type<'this, V: ResolvedTypeVisitor<'this, TypeId = Self::TypeId>>(
64        &'this self,
65        type_id: Self::TypeId,
66        visitor: V,
67    ) -> Result<V::Value, Self::Error> {
68        let Some(ty) = self.resolve(type_id) else {
69            return Ok(visitor.visit_not_found());
70        };
71
72        let path_iter = ty.path.segments.iter().map(|s| s.as_ref());
73
74        let val = match &ty.type_def {
75            scale_info::TypeDef::Composite(composite) => {
76                visitor.visit_composite(path_iter, iter_fields(&composite.fields))
77            }
78            scale_info::TypeDef::Variant(variant) => {
79                visitor.visit_variant(path_iter, iter_variants(&variant.variants))
80            }
81            scale_info::TypeDef::Sequence(seq) => {
82                visitor.visit_sequence(path_iter, seq.type_param.id)
83            }
84            scale_info::TypeDef::Array(arr) => {
85                visitor.visit_array(arr.type_param.id, arr.len as usize)
86            }
87            scale_info::TypeDef::Tuple(tuple) => {
88                let ids = tuple.fields.iter().map(|f| f.id);
89                visitor.visit_tuple(ids)
90            }
91            scale_info::TypeDef::Primitive(prim) => {
92                let primitive = into_primitive(prim);
93                visitor.visit_primitive(primitive)
94            }
95            scale_info::TypeDef::Compact(compact) => visitor.visit_compact(compact.type_param.id),
96            scale_info::TypeDef::BitSequence(bitseq) => {
97                let (order, store) = bits_from_metadata(bitseq, self)?;
98                visitor.visit_bit_sequence(store, order)
99            }
100        };
101
102        Ok(val)
103    }
104}

106fn into_primitive(primitive: &scale_info::TypeDefPrimitive) -> Primitive {
107    match primitive {
108        scale_info::TypeDefPrimitive::Bool => Primitive::Bool,
109        scale_info::TypeDefPrimitive::Char => Primitive::Char,
110        scale_info::TypeDefPrimitive::Str => Primitive::Str,
111        scale_info::TypeDefPrimitive::U8 => Primitive::U8,
112        scale_info::TypeDefPrimitive::U16 => Primitive::U16,
113        scale_info::TypeDefPrimitive::U32 => Primitive::U32,
114        scale_info::TypeDefPrimitive::U64 => Primitive::U64,
115        scale_info::TypeDefPrimitive::U128 => Primitive::U128,
116        scale_info::TypeDefPrimitive::U256 => Primitive::U256,
117        scale_info::TypeDefPrimitive::I8 => Primitive::I8,
118        scale_info::TypeDefPrimitive::I16 => Primitive::I16,
119        scale_info::TypeDefPrimitive::I32 => Primitive::I32,
120        scale_info::TypeDefPrimitive::I64 => Primitive::I64,
121        scale_info::TypeDefPrimitive::I128 => Primitive::I128,
122        scale_info::TypeDefPrimitive::I256 => Primitive::I256,
123    }
124}

126fn iter_variants(
127    variants: &'_ [scale_info::Variant<PortableForm>],
128) -> impl ExactSizeIterator<Item = Variant<'_, impl ExactSizeIterator<Item = Field<'_, u32>>>> {
129    variants.iter().map(|v| Variant {
130        index: v.index,
131        name: &v.name,
132        fields: iter_fields(&v.fields),
133    })
134}
135
136fn iter_fields(
137    fields: &'_ [scale_info::Field<PortableForm>],
138) -> impl ExactSizeIterator<Item = Field<'_, u32>> {
139    fields.iter().map(|f| Field {
140        name: f.name.as_deref(),
141        id: f.ty.id,
142    })
143}

145/// Given some type information in the form of a [`scale_info::PortableRegistry`],
146/// and a [`scale_info::TypeDefBitSequence`], this returns information about the
147/// order and store format of the bit sequence.
148pub fn bits_from_metadata(
149    ty: &scale_info::TypeDefBitSequence<scale_info::form::PortableForm>,
150    types: &scale_info::PortableRegistry,
151) -> Result<(BitsOrderFormat, BitsStoreFormat), Error> {
152    let bit_store_ty = ty.bit_store_type.id;
153    let bit_order_ty = ty.bit_order_type.id;
154
155    // What is the backing store type expected?
156    let bit_store_def = &types
157        .resolve(bit_store_ty)
158        .ok_or(Error::StoreFormatNotFound(bit_store_ty))?
159        .type_def;
160
161    // What is the bit order type expected?
162    let bit_order_def = types
163        .resolve(bit_order_ty)
164        .ok_or(Error::OrderFormatNotFound(bit_order_ty))?
165        .path
166        .ident()
167        .ok_or(Error::NoBitOrderIdent)?;
168
169    use scale_info::{TypeDef, TypeDefPrimitive};
170    let bit_store_out = match bit_store_def {
171        TypeDef::Primitive(TypeDefPrimitive::U8) => Some(BitsStoreFormat::U8),
172        TypeDef::Primitive(TypeDefPrimitive::U16) => Some(BitsStoreFormat::U16),
173        TypeDef::Primitive(TypeDefPrimitive::U32) => Some(BitsStoreFormat::U32),
174        TypeDef::Primitive(TypeDefPrimitive::U64) => Some(BitsStoreFormat::U64),
175        _ => None,
176    }
177    .ok_or(Error::UnsupportedBitStoreFormatEncountered)?;
178
179    let bit_order_out = match &*bit_order_def {
180        "Lsb0" => Some(BitsOrderFormat::Lsb0),
181        "Msb0" => Some(BitsOrderFormat::Msb0),
182        _ => None,
183    }
184    .ok_or(Error::UnsupportedBitOrderFormatEncountered)?;
185
186    Ok((bit_order_out, bit_store_out))
187}

196    fn make_type<T: scale_info::TypeInfo + 'static>() -> (u32, scale_info::PortableRegistry) {
197        let m = scale_info::MetaType::new::<T>();
198        let mut types = scale_info::Registry::new();
199        let id = types.register_type(&m);
200        let portable_registry: scale_info::PortableRegistry = types.into();
201
202        (id.id, portable_registry)
203    }

205    fn assert_type<T: scale_info::TypeInfo + 'static>(info: ResolvedTypeInfo) {
206        let (id, types) = make_type::<T>();
207        let resolved_info = to_resolved_info(id, &types);
208        assert_eq!(info, resolved_info);
209    }

211    fn to_resolved_info(type_id: u32, types: &PortableRegistry) -> ResolvedTypeInfo {
212        use crate::visitor;
213
214        // Build a quick visitor which turns resolved type info
215        // into a concrete type for us to check.
216        let visitor = visitor::new((), |_, _| panic!("all methods implemented"))
217            .visit_not_found(|_| ResolvedTypeInfo::NotFound)
218            .visit_composite(|_, _, fields| {
219                let fs = fields
220                    .map(|f| {
221                        let inner_ty = to_resolved_info(f.id, types);
222                        (f.name.map(|n| n.to_owned()), inner_ty)
223                    })
224                    .collect();
225                ResolvedTypeInfo::CompositeOf(fs)
226            })
227            .visit_variant(|_, _, variants| {
228                let vs = variants
229                    .map(|v| {
230                        let fs: Vec<_> = v
231                            .fields
232                            .map(|f| {
233                                let inner_ty = to_resolved_info(f.id, types);
234                                (f.name.map(|n| n.to_owned()), inner_ty)
235                            })
236                            .collect();
237                        (v.name.to_owned(), fs)
238                    })
239                    .collect();
240                ResolvedTypeInfo::VariantOf(vs)
241            })
242            .visit_sequence(|_, _, type_id| {
243                ResolvedTypeInfo::SequenceOf(Box::new(to_resolved_info(type_id, types)))
244            })
245            .visit_array(|_, type_id, len| {
246                ResolvedTypeInfo::ArrayOf(Box::new(to_resolved_info(type_id, types)), len)
247            })
248            .visit_tuple(|_, type_ids| {
249                let ids = type_ids.map(|id| to_resolved_info(id, types)).collect();
250                ResolvedTypeInfo::TupleOf(ids)
251            })
252            .visit_primitive(|_, primitive| ResolvedTypeInfo::Primitive(primitive))
253            .visit_compact(|_, type_id| {
254                ResolvedTypeInfo::Compact(Box::new(to_resolved_info(type_id, types)))
255            })
256            .visit_bit_sequence(|_, store_format, order_format| {
257                ResolvedTypeInfo::BitSequence(store_format, order_format)
258            });
259
260        match types.resolve_type(type_id, visitor) {
261            Err(e) => ResolvedTypeInfo::Err(e),
262            Ok(info) => info,
263        }
264    }

267    #[derive(Clone, Debug, PartialEq, Eq)]
268    enum ResolvedTypeInfo {
269        Err(Error),
270        NotFound,
271        CompositeOf(Vec<(Option<String>, ResolvedTypeInfo)>),
272        VariantOf(Vec<(String, Vec<(Option<String>, ResolvedTypeInfo)>)>),
273        SequenceOf(Box<ResolvedTypeInfo>),
274        ArrayOf(Box<ResolvedTypeInfo>, usize),
275        TupleOf(Vec<ResolvedTypeInfo>),
276        Primitive(Primitive),
277        Compact(Box<ResolvedTypeInfo>),
278        BitSequence(BitsStoreFormat, BitsOrderFormat),
279    }
