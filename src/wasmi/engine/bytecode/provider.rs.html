8/// A light-weight reference to a [`Register`] slice.
9///
10/// # Dev. Note
11///
12/// We use [`AnyConst32`] instead of a simple `u32` here to
13/// reduce the alignment requirement of this type so that
14/// it can be used by variants of [`Instruction`] without
15/// bloating up the [`Instruction`] type due to alignment
16/// constraints.
17#[derive(Debug, Copy, Clone, PartialEq, Eq)]
18pub struct ProviderSliceRef(AnyConst32);
19
20impl ProviderSliceRef {
21    /// Returns a new [`ProviderSliceRef`] from the given `usize` index.
22    fn from_index(index: usize) -> Result<Self, Error> {
23        u32::try_from(index)
24            .map_err(|_| Error::from(TranslationError::ProviderSliceOverflow))
25            .map(AnyConst32::from)
26            .map(Self)
27    }
28}

30/// A provider for an input to an [`Instruction`].
31#[derive(Debug, Copy, Clone, PartialEq, Eq)]
32pub enum Provider<T> {
33    /// A [`Register`] value.
34    Register(Register),
35    /// An immediate (or constant) value.
36    Const(T),
37}
38
39impl<T> Provider<T> {
40    /// Returns `Some` if `self` is a [`Provider::Register`].
41    pub fn into_register(self) -> Option<Register> {
42        match self {
43            Self::Register(register) => Some(register),
44            Self::Const(_) => None,
45        }
46    }
47}

49/// An untyped [`Provider`].
50///
51/// # Note
52///
53/// The [`UntypedProvider`] is primarily used for execution of
54/// Wasmi bytecode where typing usually no longer plays a role.
55pub type UntypedProvider = Provider<UntypedVal>;
56
57impl From<Register> for UntypedProvider {
58    fn from(register: Register) -> Self {
59        Self::Register(register)
60    }
61}
62
63impl From<UntypedVal> for UntypedProvider {
64    fn from(register: UntypedVal) -> Self {
65        Self::Const(register)
66    }
67}
68
69impl UntypedProvider {
70    /// Creates a new immediate value [`UntypedProvider`].
71    pub fn immediate(value: impl Into<UntypedVal>) -> Self {
72        Self::from(value.into())
73    }
74}

76/// A [`Provider`] slice stack.
77#[derive(Debug)]
78pub struct ProviderSliceStack<T> {
79    /// The end indices of each [`ProviderSliceRef`].
80    ends: Vec<usize>,
81    /// All [`Provider`] of all allocated [`Provider`] slices.
82    providers: Vec<Provider<T>>,
83}
84
85impl<T> Default for ProviderSliceStack<T> {
86    fn default() -> Self {
87        Self {
88            ends: Vec::new(),
89            providers: Vec::new(),
90        }
91    }
92}
93
94impl<T> ProviderSliceStack<T> {
95    /// Resets the [`ProviderSliceStack`] to allow for reuse.
96    pub fn reset(&mut self) {
97        self.ends.clear();
98        self.providers.clear();
99    }
100
101    /// Pushes a new [`Provider`] slice and returns its [`ProviderSliceRef`].
102    pub fn push<I>(&mut self, providers: I) -> Result<ProviderSliceRef, Error>
103    where
104        I: IntoIterator<Item = Provider<T>>,
105    {
106        self.providers.extend(providers);
107        let end = self.providers.len();
108        let index = self.ends.len();
109        self.ends.push(end);
110        ProviderSliceRef::from_index(index)
111    }
112
113    /// Pops the top-most [`Register`] slice from the [`ProviderSliceStack`] and returns it.
114    pub fn pop(&mut self) -> Option<Drain<Provider<T>>> {
115        let end = self.ends.pop()?;
116        let start = self.ends.last().copied().unwrap_or(0);
117        Some(self.providers.drain(start..end))
118    }
119}
