8/// The type of a Wasm control flow block.
9#[derive(Debug, Copy, Clone)]
10pub struct BlockType {
11    inner: BlockTypeInner,
12}
13
14/// The inner workings of the [`BlockType`].
15#[derive(Debug, Copy, Clone)]
16pub enum BlockTypeInner {
17    /// A block type with no parameters and no results.
18    Empty,
19    /// A block type with no parameters and exactly one result.
20    Returns,
21    /// A general block type with parameters and results.
22    FuncType(DedupFuncType),
23}
24
25impl BlockType {
26    /// Creates a new [`BlockType`] from the given [`wasmparser::BlockType`].
27    ///
28    /// # Errors
29    ///
30    /// If the conversion is not valid or unsupported.
31    pub fn new(block_type: wasmparser::BlockType, res: &ModuleHeader) -> Self {
32        match block_type {
33            wasmparser::BlockType::Empty => Self::empty(),
34            wasmparser::BlockType::Type(return_type) => {
35                let return_type = WasmiValueType::from(return_type).into_inner();
36                Self::returns(return_type)
37            }
38            wasmparser::BlockType::FuncType(func_type_idx) => {
39                let dedup_func_type = res.get_func_type(FuncTypeIdx::from(func_type_idx));
40                Self::func_type(dedup_func_type)
41            }
42        }
43    }
44
45    /// Creates a [`BlockType`] from the underlying type.
46    fn from_inner(inner: BlockTypeInner) -> Self {
47        Self { inner }
48    }
49
50    /// Creates a [`BlockType`] with no parameter and no results.
51    fn empty() -> Self {
52        Self::from_inner(BlockTypeInner::Empty)
53    }
54
55    /// Creates a [`BlockType`] with no parameters and a single result type.
56    fn returns(_return_type: ValType) -> Self {
57        Self::from_inner(BlockTypeInner::Returns)
58    }
59
60    /// Creates a [`BlockType`] with parameters and results.
61    pub(crate) fn func_type(func_type: &DedupFuncType) -> Self {
62        Self::from_inner(BlockTypeInner::FuncType(*func_type))
63    }
64
65    /// Returns the number of parameters of the [`BlockType`].
66    pub fn len_params(&self, engine: &Engine) -> usize {
67        match &self.inner {
68            BlockTypeInner::Empty | BlockTypeInner::Returns => 0,
69            BlockTypeInner::FuncType(func_type) => {
70                engine.resolve_func_type(func_type, |func_type| func_type.params().len())
71            }
72        }
73    }
74
75    /// Returns the number of results of the [`BlockType`].
76    pub fn len_results(&self, engine: &Engine) -> usize {
77        match &self.inner {
78            BlockTypeInner::Empty => 0,
79            BlockTypeInner::Returns => 1,
80            BlockTypeInner::FuncType(func_type) => {
81                engine.resolve_func_type(func_type, |func_type| func_type.results().len())
82            }
83        }
84    }
85}
