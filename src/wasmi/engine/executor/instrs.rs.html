52/// Executes compiled function instructions until either
53///
54/// - returning from the root function
55/// - calling a host function
56/// - encountering a trap
57///
58/// # Errors
59///
60/// If the execution traps.
61#[inline(never)]
62pub fn execute_instrs<'engine, T>(
63    store: &mut Store<T>,
64    cache: &'engine mut InstanceCache,
65    value_stack: &'engine mut ValueStack,
66    call_stack: &'engine mut CallStack,
67    code_map: &'engine CodeMap,
68    func_types: &'engine FuncTypeRegistry,
69) -> Result<(), Error> {
70    Executor::new(cache, value_stack, call_stack, code_map, func_types).execute(store)
71}

110    /// Creates a new [`Executor`] for executing a Wasmi function frame.
111    #[inline(always)]
112    pub fn new(
113        cache: &'engine mut InstanceCache,
114        value_stack: &'engine mut ValueStack,
115        call_stack: &'engine mut CallStack,
116        code_map: &'engine CodeMap,
117        func_types: &'engine FuncTypeRegistry,
118    ) -> Self {
119        let frame = call_stack
120            .peek()
121            .expect("must have call frame on the call stack");
122        // Safety: We are using the frame's own base offset as input because it is
123        //         guaranteed by the Wasm validation and translation phase to be
124        //         valid for all register indices used by the associated function body.
125        let sp = unsafe { value_stack.stack_ptr_at(frame.base_offset()) };
126        let ip = frame.instr_ptr();
127        Self {
128            sp,
129            ip,
130            cache,
131            value_stack,
132            call_stack,
133            code_map,
134            func_types,
135        }
136    }

1204            .fuel_mut()
1205            .consume_fuel_unchecked(block_fuel.to_u64())?;
