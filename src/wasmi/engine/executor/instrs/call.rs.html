24/// Dispatches and executes the host function.
25///
26/// Returns the number of parameters and results of the called host function.
27///
28/// # Errors
29///
30/// Returns the error of the host function if an error occurred.
31#[inline(always)]
32pub fn dispatch_host_func<T>(
33    store: &mut Store<T>,
34    func_types: &FuncTypeRegistry,
35    value_stack: &mut ValueStack,
36    host_func: HostFuncEntity,
37    instance: Option<&Instance>,
38) -> Result<(usize, usize), Error> {
39    let (input_types, output_types) = func_types
40        .resolve_func_type(host_func.ty_dedup())
41        .params_results();
42    let len_inputs = input_types.len();
43    let len_outputs = output_types.len();
44    let max_inout = len_inputs.max(len_outputs);
45    let values = value_stack.as_slice_mut();
46    let params_results = FuncParams::new(
47        values.split_at_mut(values.len() - max_inout).1,
48        len_inputs,
49        len_outputs,
50    );
51    let trampoline = store.resolve_trampoline(host_func.trampoline()).clone();
52    trampoline
53        .call(store, instance, params_results)
54        .map_err(|error| {
55            // Note: We drop the values that have been temporarily added to
56            //       the stack to act as parameter and result buffer for the
57            //       called host function. Since the host function failed we
58            //       need to clean up the temporary buffer values here.
59            //       This is required for resumable calls to work properly.
60            value_stack.drop(max_inout);
61            error
62        })?;
63    Ok((len_inputs, len_outputs))
64}

156    /// Updates the [`InstructionPtr`] of the caller [`CallFrame`] before dispatching a call.
157    ///
158    /// # Note
159    ///
160    /// The `offset` denotes how many [`Instruction`] words make up the call instruction.
161    #[inline(always)]
162    fn update_instr_ptr_at(&mut self, offset: usize) {
163        // Note: we explicitly do not mutate `self.ip` since that would make
164        // other parts of the code more fragile with respect to instruction ordering.
165        self.ip.add(offset);
166        let caller = self
167            .call_stack
168            .peek_mut()
169            .expect("caller call frame must be on the stack");
170        caller.update_instr_ptr(self.ip);
171    }

173    /// Fetches the [`Instruction::CallIndirectParams`] parameter for a call [`Instruction`].
174    ///
175    /// # Note
176    ///
177    /// - This advances the [`InstructionPtr`] to the next [`Instruction`].
178    /// - This is done by encoding an [`Instruction::TableGet`] instruction
179    ///   word following the actual instruction where the [`TableIdx`]
180    ///   paremeter belongs to.
181    /// - This is required for some instructions that do not fit into
182    ///   a single instruction word and store a [`TableIdx`] value in
183    ///   another instruction word.
184    fn pull_call_indirect_params(&mut self) -> (u32, TableIdx) {
185        self.ip.add(1);
186        match self.ip.get() {
187            Instruction::CallIndirectParams(call_params) => {
188                let index = u32::from(self.get_register(call_params.index));
189                let table = call_params.table;
190                (index, table)
191            }
192            Instruction::CallIndirectParamsImm16(call_params) => {
193                let index = u32::from(call_params.index);
194                let table = call_params.table;
195                (index, table)
196            }
197            unexpected => unreachable!(
198                "expected `Instruction::CallIndirectParams[Imm16]` but found {unexpected:?}"
199            ),
200        }
201    }

203    /// Creates a [`CallFrame`] for calling the [`CompiledFunc`].
204    #[inline(always)]
205    fn dispatch_compiled_func<C: CallContext>(
206        &mut self,
207        results: RegisterSpan,
208        func: &CompiledFuncEntity,
209    ) -> Result<CallFrame, Error> {
210        let (base_ptr, frame_ptr) = self.value_stack.alloc_call_frame(func)?;
211        // We have to reinstantiate the `self.sp` [`FrameRegisters`] since we just called
212        // [`ValueStack::alloc_call_frame`] which might invalidate all live [`FrameRegisters`].
213        let caller = self
214            .call_stack
215            .peek()
216            .expect("need to have a caller on the call stack");
217        // Safety: We use the base offset of a live call frame on the call stack.
218        self.sp = unsafe { self.value_stack.stack_ptr_at(caller.base_offset()) };
219        let instance = caller.instance();
220        let instr_ptr = InstructionPtr::new(func.instrs().as_ptr());
221        let frame = CallFrame::new(instr_ptr, frame_ptr, base_ptr, results, *instance);
222        if <C as CallContext>::HAS_PARAMS {
223            let called_sp = self.frame_stack_ptr(&frame);
224            self.copy_call_params(called_sp);
225        }
226        Ok(frame)
227    }

229    /// Copies the parameters from caller for the callee [`CallFrame`].
230    ///
231    /// This will also adjust the instruction pointer to point to the
232    /// last call parameter [`Instruction`] if any.
233    #[inline(always)]
234    fn copy_call_params(&mut self, mut callee_regs: FrameRegisters) {
235        let mut dst = Register::from_i16(0);
236        self.ip.add(1);
237        if let Instruction::RegisterList(_) = self.ip.get() {
238            self.copy_call_params_list(&mut dst, &mut callee_regs);
239        }
240        match self.ip.get() {
241            Instruction::Register(value) => {
242                self.copy_regs(&mut dst, &mut callee_regs, array::from_ref(value));
243            }
244            Instruction::Register2(values) => {
245                self.copy_regs(&mut dst, &mut callee_regs, values);
246            }
247            Instruction::Register3(values) => {
248                self.copy_regs(&mut dst, &mut callee_regs, values);
249            }
250            unexpected => {
251                unreachable!(
252                    "unexpected Instruction found while copying call parameters: {unexpected:?}"
253                )
254            }
255        }
256    }

258    /// Copies an array of [`Register`] to the `dst` [`Register`] span.
259    #[inline(always)]
260    fn copy_regs<const N: usize>(
261        &self,
262        dst: &mut Register,
263        callee_regs: &mut FrameRegisters,
264        regs: &[Register; N],
265    ) {
266        for value in regs {
267            let value = self.get_register(*value);
268            // Safety: The `callee.results()` always refer to a span of valid
269            //         registers of the `caller` that does not overlap with the
270            //         registers of the callee since they reside in different
271            //         call frames. Therefore this access is safe.
272            unsafe { callee_regs.set(*dst, value) }
273            *dst = dst.next();
274        }
275    }
276
277    /// Copies a list of [`Instruction::RegisterList`] to the `dst` [`Register`] span.
278    /// Copies the parameters from `src` for the called [`CallFrame`].
279    ///
280    /// This will make the [`InstructionPtr`] point to the [`Instruction`] following the
281    /// last [`Instruction::RegisterList`] if any.
282    #[inline(always)]
283    #[cold]
284    fn copy_call_params_list(&mut self, dst: &mut Register, callee_regs: &mut FrameRegisters) {
285        while let Instruction::RegisterList(values) = self.ip.get() {
286            self.copy_regs(dst, callee_regs, values);
287            self.ip.add(1);
288        }
289    }

291    /// Prepares a [`CompiledFunc`] call with optional call parameters.
292    #[inline(always)]
293    fn prepare_compiled_func_call<C: CallContext>(
294        &mut self,
295        store: &mut StoreInner,
296        results: RegisterSpan,
297        func: CompiledFunc,
298    ) -> Result<(), Error> {
299        let func = self.code_map.get(Some(store.fuel_mut()), func)?;
300        let mut called = self.dispatch_compiled_func::<C>(results, func)?;
301        match <C as CallContext>::KIND {
302            CallKind::Nested => {
303                // We need to update the instruction pointer of the caller call frame.
304                self.update_instr_ptr_at(1);
305            }
306            CallKind::Tail => {
307                // In case of a tail call we have to remove the caller call frame after
308                // allocating the callee call frame. This moves all cells of the callee frame
309                // and may invalidate pointers to it.
310                //
311                // Safety:
312                //
313                // We provide `merge_call_frames` properly with `frame` that has just been allocated
314                // on the value stack which is what the function expects. After this operation we ensure
315                // that `self.sp` is adjusted via a call to `init_call_frame` since it may have been
316                // invalidated by this method.
317                unsafe { Stack::merge_call_frames(self.call_stack, self.value_stack, &mut called) };
318            }
319        }
320        self.init_call_frame(&called);
321        self.call_stack.push(called)?;
322        Ok(())
323    }

447    /// Executes an imported or indirect (tail) call instruction.
448    fn execute_call_imported_impl<C: CallContext, T>(
449        &mut self,
450        store: &mut Store<T>,
451        results: RegisterSpan,
452        func: &Func,
453    ) -> Result<(), Error> {
454        match store.inner.resolve_func(func) {
455            FuncEntity::Wasm(func) => {
456                let instance = *func.instance();
457                let func_body = func.func_body();
458                self.prepare_compiled_func_call::<C>(&mut store.inner, results, func_body)?;
459                self.cache.update_instance(&instance);
460                Ok(())
461            }
462            FuncEntity::Host(host_func) => {
463                self.execute_host_func::<C, T>(store, results, func, *host_func)
464            }
465        }
466    }

468    /// Executes a host function.
469    ///
470    /// # Note
471    ///
472    /// This uses the value stack to store paramters and results of the host function call.
473    /// Returns an [`ErrorKind::ResumableHost`] variant if the host function returned an error
474    /// and there are still call frames on the call stack making it possible to resume the
475    /// execution at a later point in time.
476    ///
477    /// [`ErrorKind::ResumableHost`]: crate::error::ErrorKind::ResumableHost
478    #[inline(always)]
479    fn execute_host_func<C: CallContext, T>(
480        &mut self,
481        store: &mut Store<T>,
482        results: RegisterSpan,
483        func: &Func,
484        host_func: HostFuncEntity,
485    ) -> Result<(), Error> {
486        let (input_types, output_types) = self
487            .func_types
488            .resolve_func_type(host_func.ty_dedup())
489            .params_results();
490        let len_params = input_types.len();
491        let len_results = output_types.len();
492        let max_inout = len_params.max(len_results);
493        self.value_stack.reserve(max_inout)?;
494        // Safety: we just called reserve to fit the new values.
495        let offset = unsafe { self.value_stack.extend_zeros(max_inout) };
496        let offset_sp = unsafe { self.value_stack.stack_ptr_at(offset) };
497        // We have to reinstantiate the `self.sp` [`FrameRegisters`] since we just called
498        // [`ValueStack::reserve`] which might invalidate all live [`FrameRegisters`].
499        let caller = match <C as CallContext>::KIND {
500            CallKind::Nested => self.call_stack.peek().copied(),
501            CallKind::Tail => self.call_stack.pop(),
502        }
503        .expect("need to have a caller on the call stack");
504        // Safety: we use the base offset of a live call frame on the call stack.
505        self.sp = unsafe { self.value_stack.stack_ptr_at(caller.base_offset()) };
506        if <C as CallContext>::HAS_PARAMS {
507            self.copy_call_params(offset_sp);
508        }
509        if matches!(<C as CallContext>::KIND, CallKind::Nested) {
510            self.update_instr_ptr_at(1);
511        }
512        self.cache.reset();
513        let (len_inputs, len_outputs) = self
514            .dispatch_host_func::<T>(store, host_func, caller)
515            .map_err(|error| match self.call_stack.is_empty() {
516                true => error,
517                false => ResumableHostError::new(error, *func, results).into(),
518            })?;
519        // # Safety (1)
520        //
521        // We can safely acquire the stack pointer to the caller's and callee's (host)
522        // call frames because we just allocated the host call frame and can be sure that
523        // they are different.
524        // In the following we make sure to not access registers out of bounds of each
525        // call frame since we rely on Wasm validation and proper Wasm translation to
526        // provide us with valid result registers.
527        let mut caller_sp = unsafe { self.value_stack.stack_ptr_at(caller.base_offset()) };
528        // # Safety: See Safety (1) above.
529        let callee_sp = unsafe {
530            self.value_stack
531                .stack_ptr_last_n(len_inputs.max(len_outputs))
532        };
533        let results = results.iter(len_outputs);
534        let values = RegisterSpan::new(Register::from_i16(0)).iter(len_outputs);
535        for (result, value) in results.zip(values) {
536            // # Safety: See Safety (1) above.
537            unsafe { caller_sp.set(result, callee_sp.get(value)) };
538        }
539        // Finally, the value stack needs to be truncated to its original size.
540        self.value_stack.drop(max_inout);
541        Ok(())
542    }

617    /// Executes an [`Instruction::CallIndirect`] and [`Instruction::CallIndirect0`].
618    fn execute_call_indirect_impl<C: CallContext, T>(
619        &mut self,
620        store: &mut Store<T>,
621        results: RegisterSpan,
622        func_type: SignatureIdx,
623        index: u32,
624        table: TableIdx,
625    ) -> Result<(), Error> {
626        let table = self.cache.get_table(&store.inner, table);
627        let funcref = store
628            .inner
629            .resolve_table(&table)
630            .get_untyped(index)
631            .map(FuncRef::from)
632            .ok_or(TrapCode::TableOutOfBounds)?;
633        let func = funcref.func().ok_or(TrapCode::IndirectCallToNull)?;
634        let actual_signature = store.inner.resolve_func(func).ty_dedup();
635        let expected_signature = store
636            .inner
637            .resolve_instance(self.cache.instance())
638            .get_signature(func_type.to_u32())
639            .unwrap_or_else(|| {
640                panic!("missing signature for call_indirect at index: {func_type:?}")
641            });
642        if actual_signature != expected_signature {
643            return Err(Error::from(TrapCode::BadSignature));
644        }
645        self.execute_call_imported_impl::<C, T>(store, results, func)
646    }
