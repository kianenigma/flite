20/// The stack of nested function calls.
21#[derive(Debug, Default)]
22pub struct CallStack {
23    /// The stack of nested function calls.
24    calls: Vec<CallFrame>,
25    /// The maximum allowed recursion depth.
26    ///
27    /// # Note
28    ///
29    /// A [`TrapCode::StackOverflow`] is raised if the recursion limit is exceeded.
30    recursion_limit: usize,
31}
32
33impl CallStack {
34    /// Default value for the maximum recursion depth.
35    pub const DEFAULT_MAX_RECURSION_DEPTH: usize = 1024;
36
37    /// Creates a new [`CallStack`] using the given recursion limit.
38    pub fn new(recursion_limit: usize) -> Self {
39        Self {
40            calls: Vec::new(),
41            recursion_limit,
42        }
43    }
44
45    /// Clears the [`CallStack`] entirely.
46    ///
47    /// # Note
48    ///
49    /// The [`CallStack`] can sometimes be left in a non-empty state upon
50    /// executing a function, for example when a trap is encountered. We
51    /// reset the [`CallStack`] before executing the next function to
52    /// provide a clean slate for all executions.
53    pub fn reset(&mut self) {
54        self.calls.clear();
55    }
56
57    /// Returns the number of [`CallFrame`] on the [`CallStack`].
58    #[inline(always)]
59    fn len(&self) -> usize {
60        self.calls.len()
61    }
62
63    /// Returns `true` if the [`CallStack`] is empty.
64    #[inline(always)]
65    pub fn is_empty(&self) -> bool {
66        self.len() == 0
67    }
68
69    /// Pushes a [`CallFrame`] onto the [`CallStack`].
70    ///
71    /// # Errors
72    ///
73    /// If the recursion limit has been reached.
74    #[inline(always)]
75    pub fn push(&mut self, call: CallFrame) -> Result<(), TrapCode> {
76        if self.len() == self.recursion_limit {
77            return Err(err_stack_overflow());
78        }
79        self.calls.push(call);
80        Ok(())
81    }
82
83    /// Pops the last [`CallFrame`] from the [`CallStack`] if any.
84    #[inline(always)]
85    pub fn pop(&mut self) -> Option<CallFrame> {
86        self.calls.pop()
87    }
88
89    /// Peeks the last [`CallFrame`] of the [`CallStack`] if any.
90    #[inline(always)]
91    pub fn peek(&self) -> Option<&CallFrame> {
92        self.calls.last()
93    }
94
95    /// Peeks the last [`CallFrame`] of the [`CallStack`] if any.
96    #[inline(always)]
97    pub fn peek_mut(&mut self) -> Option<&mut CallFrame> {
98        self.calls.last_mut()
99    }
100
101    /// Peeks the two top-most [`CallFrame`] on the [`CallStack`] if any.
102    ///
103    /// # Note
104    ///
105    /// - The top-most [`CallFrame`] on the [`CallStack`] is referred to as the `callee`.
106    /// - The second top-most [`CallFrame`] on the [`CallStack`] is referred to as the `caller`.
107    ///
108    /// So this function returns a pair of `(callee, caller?)`.
109    pub fn peek_2(&self) -> Option<(&CallFrame, Option<&CallFrame>)> {
110        let (callee, remaining) = self.calls.split_last()?;
111        let caller = remaining.last();
112        Some((callee, caller))
113    }
114}

116/// A single frame of a called [`CompiledFunc`].
117#[derive(Debug, Copy, Clone)]
118pub struct CallFrame {
119    /// The pointer to the [`Instruction`] that is executed next.
120    instr_ptr: InstructionPtr,
121    /// Pointer to the first mutable cell of a [`CallFrame`].
122    base_ptr: BaseValueStackOffset,
123    /// Pointer to the first cell of a [`CallFrame`].
124    frame_ptr: FrameValueStackOffset,
125    /// Span of registers were the caller expects them in its [`CallFrame`].
126    results: RegisterSpan,
127    /// The instance in which the function has been defined.
128    ///
129    /// # Note
130    ///
131    /// The [`Instance`] is used to inspect and manipulate data that is
132    /// non-local to the function such as [`Memory`], [`Global`] and [`Table`].
133    instance: Instance,
134}
135
136impl CallFrame {
137    /// Creates a new [`CallFrame`].
138    pub fn new(
139        instr_ptr: InstructionPtr,
140        frame_ptr: FrameValueStackOffset,
141        base_ptr: BaseValueStackOffset,
142        results: RegisterSpan,
143        instance: Instance,
144    ) -> Self {
145        Self {
146            instr_ptr,
147            base_ptr,
148            frame_ptr,
149            results,
150            instance,
151        }
152    }
153
154    /// Moves the [`ValueStack`] offsets of the [`CallFrame`] down by `delta`.
155    ///
156    /// # Note
157    ///
158    /// This is used for the implementation of tail calls.
159    pub fn move_down(&mut self, delta: usize) {
160        let base_index = usize::from(self.base_offset());
161        let frame_index = usize::from(self.frame_offset());
162        debug_assert!(delta <= base_index);
163        debug_assert!(delta <= frame_index);
164        self.base_ptr = BaseValueStackOffset::new(base_index - delta);
165        self.frame_ptr = FrameValueStackOffset::new(frame_index - delta);
166    }
167
168    /// Updates the [`InstructionPtr`] of the [`CallFrame`].
169    ///
170    /// This is required before dispatching a nested function call to update
171    /// the instruction pointer of the caller so that it can continue at that
172    /// position when the called function returns.
173    pub fn update_instr_ptr(&mut self, new_instr_ptr: InstructionPtr) {
174        self.instr_ptr = new_instr_ptr;
175    }
176
177    /// Returns the [`InstructionPtr`] of the [`CallFrame`].
178    pub fn instr_ptr(&self) -> InstructionPtr {
179        self.instr_ptr
180    }
181
182    /// Returns the [`FrameValueStackOffset`] of the [`CallFrame`].
183    pub fn frame_offset(&self) -> FrameValueStackOffset {
184        self.frame_ptr
185    }
186
187    /// Returns the [`BaseValueStackOffset`] of the [`CallFrame`].
188    pub fn base_offset(&self) -> BaseValueStackOffset {
189        self.base_ptr
190    }
191
192    /// Returns the [`RegisterSpan`] of the [`CallFrame`].
193    ///
194    /// # Note
195    ///
196    /// The registers yielded by the returned [`RegisterSpan`]
197    /// refer to the [`CallFrame`] of the caller of this [`CallFrame`].
198    pub fn results(&self) -> RegisterSpan {
199        self.results
200    }
201
202    /// Returns the [`Instance`] of the [`CallFrame`].
203    pub fn instance(&self) -> &Instance {
204        &self.instance
205    }
206}
