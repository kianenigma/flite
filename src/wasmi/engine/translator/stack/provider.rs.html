12/// Tagged providers are inputs to Wasmi bytecode instructions.
13///
14/// Either a [`Register`] or a constant [`UntypedVal`].
15#[derive(Debug, Copy, Clone)]
16pub enum TaggedProvider {
17    /// A register referring to a function local constant value.
18    ConstLocal(Register),
19    /// A register referring to a function parameter or local variable.
20    Local(Register),
21    /// A register referring to a dynamically allocated register.
22    Dynamic(Register),
23    /// A register referring to a preservation allocated register.
24    Preserved(Register),
25    /// An untyped constant value.
26    ConstValue(TypedVal),
27}
28
29/// The stack of providers.
30///
31/// # Note
32///
33/// This partially emulates the Wasm value stack during Wasm translation phase.
34#[derive(Debug, Default)]
35pub struct ProviderStack {
36    /// The internal stack of providers.
37    providers: Vec<TaggedProvider>,
38    /// The number of locals on the `providers` stack.
39    ///
40    /// # Note
41    ///
42    /// This is used to act as fast bailout for local preservation in the common
43    /// case that no locals are on the stack.
44    len_locals: usize,
45    /// Indicates whether to use `locals` to store `locals` on the provider stack.
46    ///
47    /// # Note
48    ///
49    /// This is an optimization since using [`LocalRefs`] is expensive.
50    /// We mainly use [`LocalRefs`] to mitigate some attack surfaces of malicious inputs.
51    /// We flip this `bool` flag once `providers` grow beyond a certain threshold.
52    /// This way linear operations on `providers` can be seen as constant and
53    /// thus not attackable.
54    use_locals: bool,
55    /// Used to store indices of `local.get` on the `providers` stack.
56    locals: LocalRefs,
57}
58
59impl ProviderStack {
60    /// Resets the [`ProviderStack`].
61    pub fn reset(&mut self) {
62        self.providers.clear();
63        self.use_locals = false;
64        self.locals.reset();
65        self.len_locals = 0;
66    }
67
68    /// Maximum provider stack height before switching to attack-immune
69    /// [`LocalRefs`] implementation for `local.get` preservation.
70    const PRESERVE_THRESHOLD: usize = 16;
71
72    /// Synchronizes [`LocalRefs`] with the current state of the `providers` stack.
73    ///
74    /// This is required to initialize usage of the attack-immune [`LocalRefs`] before first use.
75    fn sync_local_refs(&mut self) {
76        if self.use_locals || self.providers.len() < Self::PRESERVE_THRESHOLD {
77            return;
78        }
79        self.use_locals = true;
80        for (index, provider) in self.providers.iter().enumerate() {
81            let TaggedProvider::Local(local) = provider else {
82                continue;
83            };
84            self.locals.push_at(*local, index);
85        }
86        self.use_locals = true;
87    }
88
89    /// Preserves `local.get` on the [`ProviderStack`] by shifting to the preservation space.
90    ///
91    /// In case there are `local.get n` with `n == preserve_index` on the [`ProviderStack`]
92    /// there is a [`Register`] on the preservation space allocated for them. The [`Register`]
93    /// allocated this way is returned. Otherwise `None` is returned.
94    pub fn preserve_locals(
95        &mut self,
96        preserve_index: u32,
97        reg_alloc: &mut RegisterAlloc,
98    ) -> Result<Option<Register>, Error> {
99        if self.len_locals == 0 {
100            return Ok(None);
101        }
102        self.sync_local_refs();
103        let local = i16::try_from(preserve_index)
104            .map(Register::from_i16)
105            .unwrap_or_else(|_| {
106                panic!("encountered invalid local register index: {preserve_index}")
107            });
108        match self.use_locals {
109            false => self.preserve_locals_inplace(local, reg_alloc),
110            true => self.preserve_locals_extern(local, reg_alloc),
111        }
112    }
113
114    /// Preserves all locals on the [`ProviderStack`] by shifting them to the preservation space.
115    ///
116    /// # Note
117    ///
118    /// - Calls `f(local_index, preserved_register)` for each local preserved this way with its `local_index`
119    ///   and the newly allocated `preserved_register` on the presevation register space.
120    /// - If the same local appears multiple  times on the provider stack `f` is only called once
121    ///   representing all of them.
122    pub fn preserve_all_locals(
123        &mut self,
124        reg_alloc: &mut RegisterAlloc,
125        f: impl FnMut(PreservedLocal) -> Result<(), Error>,
126    ) -> Result<(), Error> {
127        if self.len_locals == 0 {
128            return Ok(());
129        }
130        self.sync_local_refs();
131        match self.use_locals {
132            false => self.preserve_all_locals_inplace(reg_alloc, f),
133            true => self.preserve_all_locals_extern(reg_alloc, f),
134        }
135    }
136
137    /// Preserves the `local` [`Register`] on the provider stack in-place.
138    ///
139    /// # Note
140    ///
141    /// - This is the efficient case which is susceptible to malicious inputs
142    ///   since it needs to iterate over the entire provider stack and might be
143    ///   called roughly once per Wasm instruction in the worst-case.
144    /// - Therefore we only use it behind a safety guard to remove the attack surface.
145    fn preserve_locals_inplace(
146        &mut self,
147        local: Register,
148        reg_alloc: &mut RegisterAlloc,
149    ) -> Result<Option<Register>, Error> {
150        debug_assert!(!self.use_locals);
151        let mut preserved = None;
152        for provider in &mut self.providers {
153            let TaggedProvider::Local(register) = *provider else {
154                continue;
155            };
156            debug_assert!(reg_alloc.is_local(register));
157            if register != local {
158                continue;
159            }
160            let preserved_register = match preserved {
161                None => {
162                    let register = reg_alloc.push_preserved()?;
163                    preserved = Some(register);
164                    register
165                }
166                Some(register) => {
167                    reg_alloc.bump_preserved(register);
168                    register
169                }
170            };
171            *provider = TaggedProvider::Preserved(preserved_register);
172            self.len_locals -= 1;
173            if self.len_locals == 0 {
174                break;
175            }
176        }
177        Ok(preserved)
178    }
179
180    /// Preserves all locals on the [`ProviderStack`] by shifting them to the preservation space.
181    ///
182    /// # Note
183    ///
184    /// - Calls `f(local_index, preserved_register)` for each local preserved this way with its `local_index`
185    ///   and the newly allocated `preserved_register` on the presevation register space.
186    /// - If the same local appears multiple  times on the provider stack `f` is only called once
187    ///   representing all of them.
188    ///
189    /// # Dev. Note
190    ///
191    /// - This is the efficient case which is susceptible to malicious inputs
192    ///   since it needs to iterate over the entire provider stack and might be
193    ///   called roughly once per Wasm instruction in the worst-case.
194    /// - Therefore we only use it behind a safety guard to remove the attack surface.
195    fn preserve_all_locals_inplace(
196        &mut self,
197        reg_alloc: &mut RegisterAlloc,
198        mut f: impl FnMut(PreservedLocal) -> Result<(), Error>,
199    ) -> Result<(), Error> {
200        debug_assert!(!self.use_locals);
201        let mut preserved = <ArrayVec<PreservedLocal, { Self::PRESERVE_THRESHOLD }>>::new();
202        for provider in self.providers.iter_mut().rev() {
203            let TaggedProvider::Local(local_register) = *provider else {
204                continue;
205            };
206            debug_assert!(reg_alloc.is_local(local_register));
207            let (preserved_register, is_new) =
208                // Note: linear search is fine since we operate only on very small sets of data.
209                match preserved.iter().find(|p| p.local == local_register) {
210                    Some(preserved_local) => {
211                        let preserved_register = preserved_local.preserved;
212                        reg_alloc.bump_preserved(preserved_register);
213                        (preserved_register, false)
214                    }
215                    None => {
216                        let preserved_register = reg_alloc.push_preserved()?;
217                        preserved.push(PreservedLocal::new(local_register, preserved_register));
218                        (preserved_register, true)
219                    }
220                };
221            *provider = TaggedProvider::Preserved(preserved_register);
222            self.len_locals -= 1;
223            if is_new {
224                f(PreservedLocal::new(local_register, preserved_register))?;
225            }
226            if self.len_locals == 0 {
227                break;
228            }
229        }
230        debug_assert_eq!(self.len_locals, 0);
231        Ok(())
232    }

234    /// Preserves the `local` [`Register`] on the provider stack out-of-place.
235    ///
236    /// # Note
237    ///
238    /// - This is the inefficient case which is immune to malicious inputs
239    ///   since it only iterates over the locals required for preservation
240    ///   which are stored out-of-place of the provider stack.
241    /// - Since this is slower we only use it when necessary.
242    fn preserve_locals_extern(
243        &mut self,
244        local: Register,
245        reg_alloc: &mut RegisterAlloc,
246    ) -> Result<Option<Register>, Error> {
247        debug_assert!(self.use_locals);
248        let mut preserved = None;
249        self.locals.drain_at(local, |provider_index| {
250            let provider = &mut self.providers[provider_index];
251            debug_assert!(matches!(provider, TaggedProvider::Local(_)));
252            let preserved_register = match preserved {
253                Some(register) => {
254                    reg_alloc.bump_preserved(register);
255                    register
256                }
257                None => {
258                    let register = reg_alloc.push_preserved()?;
259                    preserved = Some(register);
260                    register
261                }
262            };
263            *provider = TaggedProvider::Preserved(preserved_register);
264            self.len_locals -= 1;
265            Ok(())
266        })?;
267        Ok(preserved)
268    }

270    /// Preserves all locals on the [`ProviderStack`] by shifting them to the preservation space.
271    ///
272    /// # Note
273    ///
274    /// - Calls `f(local_index, preserved_register)` for each local preserved this way with its `local_index`
275    ///   and the newly allocated `preserved_register` on the presevation register space.
276    /// - If the same local appears multiple  times on the provider stack `f` is only called once
277    ///   representing all of them.
278    ///
279    /// # Dev. Note
280    ///
281    /// - This is the inefficient case which is immune to malicious inputs
282    ///   since it only iterates over the locals required for preservation
283    ///   which are stored out-of-place of the provider stack.
284    /// - Since this is slower we only use it when necessary.
285    fn preserve_all_locals_extern(
286        &mut self,
287        reg_alloc: &mut RegisterAlloc,
288        mut f: impl FnMut(PreservedLocal) -> Result<(), Error>,
289    ) -> Result<(), Error> {
290        debug_assert!(self.use_locals);
291        let mut group = None;
292        self.locals.drain_all(|local, index| {
293            let preserved = match group {
294                Some((group, preserved)) if group == local => {
295                    reg_alloc.bump_preserved(preserved);
296                    preserved
297                }
298                _ => {
299                    let preserved = reg_alloc.push_preserved()?;
300                    group = Some((local, preserved));
301                    f(PreservedLocal::new(local, preserved))?;
302                    preserved
303                }
304            };
305            self.providers[index] = TaggedProvider::Preserved(preserved);
306            self.len_locals -= 1;
307            Ok(())
308        })?;
309        debug_assert_eq!(self.len_locals, 0);
310        Ok(())
311    }

313    /// Registers an `amount` of function inputs or local variables.
314    ///
315    /// # Errors
316    ///
317    /// If too many registers have been registered.
318    pub fn register_locals(&mut self, amount: u32) {
319        self.locals.register_locals(amount)
320    }

322    /// Returns the number of [`TaggedProvider`] on the [`ProviderStack`].
323    pub fn len(&self) -> usize {
324        self.providers.len()
325    }
326
327    /// Pushes a provider to the [`ProviderStack`].
328    #[inline(always)]
329    fn push(&mut self, provider: TaggedProvider) {
330        let index = self.providers.len();
331        self.providers.push(provider);
332        if let TaggedProvider::Local(reg) = provider {
333            self.len_locals += 1;
334            if self.use_locals {
335                self.locals.push_at(reg, index);
336            }
337        }
338    }

340    /// Pushes a [`Register`] to the [`ProviderStack`] referring to a function parameter or local variable.
341    pub fn push_local(&mut self, reg: Register) {
342        debug_assert!(!reg.is_const());
343        self.push(TaggedProvider::Local(reg));
344    }
345
346    /// Pushes a dynamically allocated [`Register`] to the [`ProviderStack`].
347    pub fn push_dynamic(&mut self, reg: Register) {
348        debug_assert!(!reg.is_const());
349        self.push(TaggedProvider::Dynamic(reg));
350    }
351
352    /// Pushes a preservation allocated [`Register`] to the [`ProviderStack`].
353    pub fn push_preserved(&mut self, reg: Register) {
354        debug_assert!(!reg.is_const());
355        self.push(TaggedProvider::Preserved(reg));
356    }
357
358    /// Pushes a [`Register`] to the [`ProviderStack`] referring to a function parameter or local variable.
359    pub fn push_const_local(&mut self, reg: Register) {
360        debug_assert!(reg.is_const());
361        self.push(TaggedProvider::ConstLocal(reg));
362    }
363
364    /// Pushes a constant value to the [`ProviderStack`].
365    pub fn push_const_value<T>(&mut self, value: T)
366    where
367        T: Into<TypedVal>,
368    {
369        self.push(TaggedProvider::ConstValue(value.into()));
370    }

372    /// Pops the top-most [`TaggedProvider`] from the [`ProviderStack`].
373    ///
374    /// # Panics
375    ///
376    /// If the [`ProviderStack`] is empty.
377    pub fn peek(&self) -> TaggedProvider {
378        self.providers
379            .last()
380            .copied()
381            .unwrap_or_else(|| panic!("tried to peek provider from empty provider stack"))
382    }
383
384    /// Pops the top-most [`TaggedProvider`] from the [`ProviderStack`].
385    ///
386    /// # Panics
387    ///
388    /// If the [`ProviderStack`] is empty.
389    pub fn pop(&mut self) -> TaggedProvider {
390        let popped = self
391            .providers
392            .pop()
393            .unwrap_or_else(|| panic!("tried to pop provider from empty provider stack"));
394        if let TaggedProvider::Local(register) = popped {
395            self.len_locals -= 1;
396            if self.use_locals {
397                // If a `local.get` was popped from the provider stack we
398                // also need to pop it from the `local.get` indices stack.
399                let stack_index = self.locals.pop_at(register);
400                debug_assert_eq!(self.providers.len(), stack_index);
401            }
402        }
403        popped
404    }
405
406    /// Peeks the `n` top-most [`TaggedProvider`] items of the [`ProviderStack`].
407    ///
408    /// # Panics
409    ///
410    /// If the [`ProviderStack`] does not contain at least `n` [`TaggedProvider`] items.
411    pub fn peek_n(&self, n: usize) -> &[TaggedProvider] {
412        let len = self.providers.len();
413        assert!(
414            n <= len,
415            "tried to peek {n} items from provider stack with only {len} items"
416        );
417        &self.providers[(len - n)..]
418    }
