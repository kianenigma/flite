12pub type StackIndex = usize;

14/// The index of an entry in the [`LocalRefs`] data structure.
15type EntryIndex = usize;
16
17/// Data structure to store local indices on the compilation stack for large stacks.
18///
19/// # Note
20///
21/// - The main purpose is to provide efficient implementations to preserve locals on
22///   the compilation stack for single local and mass local preservation.
23/// - Also this data structure is critical to not be attackable by malicious actors
24///   when operating on very large stacks or local variable quantities.
25#[derive(Debug, Default)]
26pub struct LocalRefs {
27    /// The last local added to [`LocalRefs`] per local variable if any.
28    locals_last: BTreeMap<Register, EntryIndex>,
29    /// The entries of the [`LocalRefs`] data structure.
30    entries: LocalRefsEntries,
31}
32
33/// The entries of the [`LocalRefs`] data structure.
34///
35/// # Note
36///
37/// This type mostly exists to gracefully resolve some borrow-checking issues
38/// when operating on parts of the fields of the [`LocalRefs`] while `locals_last`
39/// is borrowed.
40#[derive(Debug, Default)]
41pub struct LocalRefsEntries {
42    /// The index of the next free (vacant) entry.
43    next_free: Option<EntryIndex>,
44    /// All entries of the [`LocalRefs`] data structure.
45    entries: Vec<LocalRefEntry>,
46}
47
48impl LocalRefsEntries {
49    /// Resets the [`LocalRefs`].
50    pub fn reset(&mut self) {
51        self.next_free = None;
52        self.entries.clear();
53    }
54
55    /// Returns the next free [`EntryIndex`] for reusing vacant entries.
56    #[inline]
57    pub fn next_free(&self) -> Option<EntryIndex> {
58        self.next_free
59    }
60
61    /// Returns the next [`EntryIndex`] for the next new non-reused entry.
62    #[inline]
63    pub fn next_index(&self) -> EntryIndex {
64        self.entries.len()
65    }
66
67    /// Pushes an occupied entry to the [`LocalRefsEntries`].
68    #[inline]
69    pub fn push_occupied(&mut self, slot: StackIndex, prev: Option<EntryIndex>) -> EntryIndex {
70        let index = self.next_index();
71        self.entries.push(LocalRefEntry::Occupied { slot, prev });
72        index
73    }
74
75    /// Reuses the vacant entry at `index` for a new occupied entry.
76    ///
77    /// # Panics
78    ///
79    /// If the entry at `index` is not vacant.
80    #[inline]
81    pub fn reuse_vacant(&mut self, index: EntryIndex, slot: StackIndex, prev: Option<EntryIndex>) {
82        let old_entry = mem::replace(
83            &mut self.entries[index],
84            LocalRefEntry::Occupied { slot, prev },
85        );
86        self.next_free = match old_entry {
87            LocalRefEntry::Vacant { next_free } => next_free,
88            occupied @ LocalRefEntry::Occupied { .. } => {
89                panic!("tried to reuse occupied entry at index {index}: {occupied:?}")
90            }
91        };
92    }
93
94    /// Removes the entry at the given `index`.
95    ///
96    /// Returns the entry index of the next entry in the list and the
97    /// [`StackIndex`] associated to the removed entry.
98    #[inline]
99    fn remove_entry(&mut self, index: EntryIndex) -> (Option<EntryIndex>, StackIndex) {
100        let next_free = self.next_free();
101        let old_entry = mem::replace(
102            &mut self.entries[index],
103            LocalRefEntry::Vacant { next_free },
104        );
105        let LocalRefEntry::Occupied { prev, slot } = old_entry else {
106            panic!("expected occupied entry but found vacant: {old_entry:?}");
107        };
108        self.next_free = Some(index);
109        (prev, slot)
110    }
111}
112
113/// An entry representing a local variable on the compilation stack or a vacant entry.
114#[derive(Debug, Copy, Clone)]
115enum LocalRefEntry {
116    Vacant {
117        /// The next free slot of the [`LocalRefs`] data structure.
118        next_free: Option<EntryIndex>,
119    },
120    Occupied {
121        /// The slot index of the local variable on the compilation stack.
122        slot: StackIndex,
123        /// The next [`LocalRefEntry`] referencing the same local variable if any.
124        prev: Option<EntryIndex>,
125    },
126}
127
128impl LocalRefs {
129    /// Resets the [`LocalRefs`].
130    pub fn reset(&mut self) {
131        self.locals_last.clear();
132        self.entries.reset();
133    }
134
135    /// Registers an `amount` of function inputs or local variables.
136    ///
137    /// # Errors
138    ///
139    /// If too many registers have been registered.
140    pub fn register_locals(&mut self, _amount: u32) {
141        // Nothing to do here.
142    }
143
144    /// Updates the last index for `local` to `index` and returns the previous last index.
145    fn update_last(&mut self, index: EntryIndex, local: Register) -> Option<EntryIndex> {
146        match self.locals_last.entry(local) {
147            btree_map::Entry::Vacant(entry) => {
148                entry.insert(index);
149                None
150            }
151            btree_map::Entry::Occupied(mut entry) => {
152                let prev = *entry.get();
153                entry.insert(index);
154                Some(prev)
155            }
156        }
157    }
158
159    /// Pushes the stack index of a `local.get` on the [`ProviderStack`].
160    ///
161    /// # Panics
162    ///
163    /// If the `local` index is out of bounds.
164    pub fn push_at(&mut self, local: Register, slot: StackIndex) {
165        match self.entries.next_free() {
166            Some(index) => {
167                let prev = self.update_last(index, local);
168                self.entries.reuse_vacant(index, slot, prev);
169            }
170            None => {
171                let index = self.entries.next_index();
172                let prev = self.update_last(index, local);
173                let pushed = self.entries.push_occupied(slot, prev);
174                debug_assert_eq!(pushed, index);
175            }
176        };
177    }
178
179    /// Returns `true` if `self` is empty.
180    #[inline]
181    fn is_empty(&self) -> bool {
182        self.locals_last.is_empty()
183    }
184
185    /// Reset `self` if `self` is empty.
186    #[inline]
187    fn reset_if_empty(&mut self) {
188        if self.is_empty() {
189            self.entries.reset();
190        }
191    }
192
193    /// Pops the stack index of a `local.get` on the [`ProviderStack`].
194    ///
195    /// # Panics
196    ///
197    /// - If the `local` index is out of bounds.
198    /// - If there is no `local.get` stack index on the stack.
199    pub fn pop_at(&mut self, local: Register) -> StackIndex {
200        let btree_map::Entry::Occupied(mut last) = self.locals_last.entry(local) else {
201            panic!("missing stack index for local on the provider stack: {local:?}")
202        };
203        let index = *last.get();
204        let (prev, slot) = self.entries.remove_entry(index);
205        match prev {
206            Some(prev) => last.insert(prev),
207            None => last.remove(),
208        };
209        self.reset_if_empty();
210        slot
211    }
212
213    /// Drains all local indices of the `local` variable on the [`ProviderStack`].
214    ///
215    /// # Note
216    ///
217    /// Calls `f` with the index of each local on the [`ProviderStack`] that matches `local`.
218    pub fn drain_at(
219        &mut self,
220        local: Register,
221        f: impl FnMut(StackIndex) -> Result<(), Error>,
222    ) -> Result<(), Error> {
223        let Some(last) = self.locals_last.remove(&local) else {
224            return Ok(());
225        };
226        self.drain_list_at(last, f)?;
227        self.reset_if_empty();
228        Ok(())
229    }
230
231    /// Drains all local indices on the [`ProviderStack`].
232    ///
233    /// # Note
234    ///
235    /// Calls `f` with the pair of local and its index of each local on the [`ProviderStack`].
236    pub fn drain_all(
237        &mut self,
238        mut f: impl FnMut(Register, StackIndex) -> Result<(), Error>,
239    ) -> Result<(), Error> {
240        let local_last = mem::take(&mut self.locals_last);
241        for (local, last) in &local_last {
242            let local = *local;
243            self.drain_list_at(*last, |index| f(local, index))?;
244        }
245        self.locals_last = local_last;
246        self.locals_last.clear();
247        self.entries.reset();
248        Ok(())
249    }
250
251    /// Drains the list of locals starting at `index` at the entries array.
252    #[inline]
253    fn drain_list_at(
254        &mut self,
255        index: EntryIndex,
256        mut f: impl FnMut(StackIndex) -> Result<(), Error>,
257    ) -> Result<(), Error> {
258        let mut last = Some(index);
259        while let Some(index) = last {
260            let (prev, slot) = self.entries.remove_entry(index);
261            last = prev;
262            f(slot)?;
263        }
264        Ok(())
265    }
266}

270    use super::*;
271
272    fn reg(index: i16) -> Register {
273        Register::from_i16(index)
274    }
275
276    #[test]
277    fn push_pop_works() {
278        let mut locals = LocalRefs::default();
279        locals.push_at(reg(0), 2);
280        locals.push_at(reg(0), 4);
281        locals.push_at(reg(1), 6);
282        locals.push_at(reg(2), 8);
283        locals.push_at(reg(5), 10);
284        locals.push_at(reg(1), 12);
285        locals.push_at(reg(0), 14);
286        assert_eq!(locals.pop_at(reg(0)), 14);
287        assert_eq!(locals.pop_at(reg(0)), 4);
288        assert_eq!(locals.pop_at(reg(0)), 2);
289        assert_eq!(locals.pop_at(reg(1)), 12);
290        assert_eq!(locals.pop_at(reg(1)), 6);
291        assert_eq!(locals.pop_at(reg(2)), 8);
292        assert_eq!(locals.pop_at(reg(5)), 10);
293    }
