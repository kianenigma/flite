1use core::mem::ManuallyDrop;
2use std::{slice, vec, vec::Vec};

4/// A byte buffer implementation.
5///
6/// # Note
7///
8/// This is less efficient than the byte buffer implementation that is
9/// based on actual OS provided virtual memory but it is a safe fallback
10/// solution fitting any platform.
11#[derive(Debug)]
12pub struct ByteBuffer {
13    /// The pointer to the underlying byte buffer.
14    ptr: *mut u8,
15    /// The current length of the byte buffer.
16    ///
17    /// # Note
18    ///
19    /// - **Vec:** `vec.len()`
20    /// - **Static:** The accessible subslice of the entire underlying static byte buffer.
21    len: usize,
22    /// The capacity of the current allocation.
23    ///
24    /// # Note
25    ///
26    /// - **Vec**: `vec.capacity()`
27    /// - **Static:** The total length of the underlying static byte buffer.
28    capacity: usize,
29    /// Whether the [`ByteBuffer`] was initialized from a `&'static [u8]` or a `Vec<u8>`.
30    is_static: bool,
31}
32
33// # Safety
34//
35// `ByteBuffer` is essentially an `enum`` of `Vec<u8>` or `&'static mut [u8]`.
36// Both of them are `Send` so this is sound.
37unsafe impl Send for ByteBuffer {}
38
39/// Decomposes the `Vec<u8>` into its raw components.
40///
41/// Returns the raw pointer to the underlying data, the length of
42/// the vector (in bytes), and the allocated capacity of the
43/// data (in bytes). These are the same arguments in the same
44/// order as the arguments to [`Vec::from_raw_parts`].
45///
46/// # Safety
47///
48/// After calling this function, the caller is responsible for the
49/// memory previously managed by the `Vec`. The only way to do
50/// this is to convert the raw pointer, length, and capacity back
51/// into a `Vec` with the [`Vec::from_raw_parts`] function, allowing
52/// the destructor to perform the cleanup.
53///
54/// # Note
55///
56/// This utility method is required since [`Vec::into_raw_parts`] is
57/// not yet stable unfortunately. (Date: 2024-03-14)
58fn vec_into_raw_parts(vec: Vec<u8>) -> (*mut u8, usize, usize) {
59    let mut vec = ManuallyDrop::new(vec);
60    (vec.as_mut_ptr(), vec.len(), vec.capacity())
61}
62
63impl ByteBuffer {
64    /// Creates a new byte buffer with the given initial length.
65    ///
66    /// # Panics
67    ///
68    /// If there is not enough memory to initialize `initial_len` bytes.
69    pub fn new(initial_len: usize) -> Self {
70        let vec = vec![0x00_u8; initial_len];
71        let (ptr, len, capacity) = vec_into_raw_parts(vec);
72        Self {
73            ptr,
74            len,
75            capacity,
76            is_static: false,
77        }
78    }
79
80    /// Creates a new byte buffer with the given initial length.
81    ///
82    /// This will zero all the bytes in `buffer[0..initial_len`].
83    ///
84    /// # Panics
85    ///
86    /// If `initial_len` is greater than the length of `buffer`.
87    pub fn new_static(buffer: &'static mut [u8], initial_len: usize) -> Self {
88        assert!(initial_len <= buffer.len());
89        buffer[..initial_len].fill(0x00_u8);
90        Self {
91            ptr: buffer.as_mut_ptr(),
92            len: initial_len,
93            capacity: buffer.len(),
94            is_static: true,
95        }
96    }
97
98    /// Grows the byte buffer to the given `new_size`.
99    ///
100    /// The newly added bytes will be zero initialized.
101    ///
102    /// # Panics
103    ///
104    /// - If the current size of the [`ByteBuffer`] is larger than `new_size`.
105    /// - If backed by static buffer and `new_size` is larger than it's capacity.
106    pub fn grow(&mut self, new_size: usize) {
107        assert!(new_size >= self.len());
108        match self.get_vec() {
109            Some(mut vec) => {
110                // Case: the byte buffer is backed by a `Vec<u8>`.
111                vec.resize(new_size, 0x00_u8);
112                let (ptr, len, capacity) = vec_into_raw_parts(vec);
113                self.ptr = ptr;
114                self.len = len;
115                self.capacity = capacity;
116            }
117            None => {
118                // Case: the byte buffer is backed by a `&'static [u8]`.
119                if self.capacity < new_size {
120                    panic!("cannot grow a byte buffer backed by `&'static mut [u8]` beyond its capacity")
121                }
122                let len = self.len();
123                self.len = new_size;
124                self.data_mut()[len..new_size].fill(0x00_u8);
125            }
126        }
127    }
128
129    /// Returns the length of the byte buffer in bytes.
130    pub fn len(&self) -> usize {
131        self.len
132    }
133
134    /// Returns a shared slice to the bytes underlying to the byte buffer.
135    pub fn data(&self) -> &[u8] {
136        // # Safety
137        //
138        // The byte buffer is either backed by a `Vec<u8>` or a &'static [u8]`
139        // which are both valid byte slices in the range `self.ptr[0..self.len]`.
140        unsafe { slice::from_raw_parts(self.ptr, self.len) }
141    }
142
143    /// Returns an exclusive slice to the bytes underlying to the byte buffer.
144    pub fn data_mut(&mut self) -> &mut [u8] {
145        // # Safety
146        //
147        // The byte buffer is either backed by a `Vec<u8>` or a &'static [u8]`
148        // which are both valid byte slices in the range `self.ptr[0..self.len]`.
149        unsafe { slice::from_raw_parts_mut(self.ptr, self.len) }
150    }
151
152    /// Returns the underlying `Vec<u8>` if the byte buffer is not backed by a static buffer.
153    ///
154    /// Otherwise returns `None`.
155    ///
156    /// # Note
157    ///
158    /// The returned `Vec` will free its memory and thus the memory of the [`ByteBuffer`] if dropped.
159    fn get_vec(&mut self) -> Option<Vec<u8>> {
160        if self.is_static {
161            return None;
162        }
163        // Safety
164        //
165        // - At this point we are guaranteed that the byte buffer is backed by a `Vec`
166        //   so it is safe to reconstruct the `Vec` by its raw parts.
167        Some(unsafe { Vec::from_raw_parts(self.ptr, self.len, self.capacity) })
168    }
169}
170
171impl Drop for ByteBuffer {
172    fn drop(&mut self) {
173        self.get_vec();
174    }
175}

177#[cfg(test)]
178mod test {
179    use super::*;
180    #[test]
181    fn test_basic_allocation_deallocation() {
182        let buffer = ByteBuffer::new(10);
183        assert_eq!(buffer.len(), 10);
184        // Dropping the buffer should not cause UB.
185    }
186
187    #[test]
188    fn test_basic_data_manipulation() {
189        let mut buffer = ByteBuffer::new(10);
190        assert_eq!(buffer.len(), 10);
191        let data = buffer.data(); // test we can read the data
192        assert_eq!(data, &[0; 10]);
193        let data = buffer.data_mut(); // test we can take a mutable reference to the data
194        data[4] = 4; // test we can write to the data and it is not UB
195        let data = buffer.data(); // test we can take a new reference to the data
196        assert_eq!(data, &[0, 0, 0, 0, 4, 0, 0, 0, 0, 0]); // test we can read the data
197                                                           // test drop is okay
198    }
199
200    #[test]
201    fn test_static_buffer_initialization() {
202        static mut BUF: [u8; 10] = [7; 10];
203        let buf = unsafe { &mut *core::ptr::addr_of_mut!(BUF) };
204        let mut buffer = ByteBuffer::new_static(buf, 5);
205        assert_eq!(buffer.len(), 5);
206        // Modifying the static buffer through ByteBuffer and checking its content.
207        let data = buffer.data_mut();
208        data[0] = 1;
209        unsafe {
210            assert_eq!(BUF[0], 1);
211        }
212    }
213
214    #[test]
215    fn test_growing_buffer() {
216        let mut buffer = ByteBuffer::new(5);
217        buffer.grow(10);
218        assert_eq!(buffer.len(), 10);
219        assert_eq!(buffer.data(), &[0; 10]);
220    }
221
222    #[test]
223    fn test_growing_static() {
224        static mut BUF: [u8; 10] = [7; 10];
225        let buf = unsafe { &mut *core::ptr::addr_of_mut!(BUF) };
226        let mut buffer = ByteBuffer::new_static(buf, 5);
227        assert_eq!(buffer.len(), 5);
228        assert_eq!(buffer.data(), &[0; 5]);
229        buffer.grow(8);
230        assert_eq!(buffer.len(), 8);
231        assert_eq!(buffer.data(), &[0; 8]);
232        buffer.grow(10);
233        assert_eq!(buffer.len(), 10);
234        assert_eq!(buffer.data(), &[0; 10]);
235    }
