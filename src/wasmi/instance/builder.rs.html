18/// A module instance entity builder.
19#[derive(Debug)]
20pub struct InstanceEntityBuilder {
21    func_types: Arc<[DedupFuncType]>,
22    tables: Vec<Table>,
23    funcs: Vec<Func>,
24    memories: Vec<Memory>,
25    globals: Vec<Global>,
26    start_fn: Option<FuncIdx>,
27    exports: Map<Box<str>, Extern>,
28    data_segments: Vec<DataSegment>,
29    elem_segments: Vec<ElementSegment>,
30}
31
32impl InstanceEntityBuilder {
33    /// Creates a new [`InstanceEntityBuilder`] optimized for the [`Module`].
34    pub fn new(module: &Module) -> Self {
35        fn vec_with_capacity_exact<T>(capacity: usize) -> Vec<T> {
36            let mut v = Vec::new();
37            v.reserve_exact(capacity);
38            v
39        }
40        let mut len_funcs = module.len_funcs();
41        let mut len_globals = module.len_globals();
42        let mut len_tables = module.len_tables();
43        let mut len_memories = module.len_memories();
44        for import in module.imports() {
45            match import.ty() {
46                ExternType::Func(_) => {
47                    len_funcs += 1;
48                }
49                ExternType::Table(_) => {
50                    len_tables += 1;
51                }
52                ExternType::Memory(_) => {
53                    len_memories += 1;
54                }
55                ExternType::Global(_) => {
56                    len_globals += 1;
57                }
58            }
59        }
60        Self {
61            func_types: module.func_types_cloned(),
62            tables: vec_with_capacity_exact(len_tables),
63            funcs: vec_with_capacity_exact(len_funcs),
64            memories: vec_with_capacity_exact(len_memories),
65            globals: vec_with_capacity_exact(len_globals),
66            start_fn: None,
67            exports: Map::default(),
68            data_segments: Vec::new(),
69            elem_segments: Vec::new(),
70        }
71    }
72
73    /// Sets the start function of the built instance.
74    ///
75    /// # Panics
76    ///
77    /// If the start function has already been set.
78    pub fn set_start(&mut self, start_fn: FuncIdx) {
79        match &mut self.start_fn {
80            Some(index) => panic!("already set start function {index:?}"),
81            None => {
82                self.start_fn = Some(start_fn);
83            }
84        }
85    }
86
87    /// Returns the start function index if any.
88    pub fn get_start(&self) -> Option<FuncIdx> {
89        self.start_fn
90    }
91
92    /// Returns the [`Memory`] at the `index`.
93    ///
94    /// # Panics
95    ///
96    /// If there is no [`Memory`] at the given `index.
97    pub fn get_memory(&self, index: u32) -> Memory {
98        self.memories
99            .get(index as usize)
100            .copied()
101            .unwrap_or_else(|| panic!("missing `Memory` at index: {index}"))
102    }
103
104    /// Returns the [`Table`] at the `index`.

106    /// # Panics

108    /// If there is no [`Table`] at the given `index.
109    pub fn get_table(&self, index: u32) -> Table {
110        self.tables
111            .get(index as usize)
112            .copied()
113            .unwrap_or_else(|| panic!("missing `Table` at index: {index}"))
114    }
115
116    /// Returns the [`Global`] at the `index`.
117    ///
118    /// # Panics
119    ///
120    /// If there is no [`Global`] at the given `index.
121    pub fn get_global(&self, index: u32) -> Global {
122        self.globals
123            .get(index as usize)
124            .copied()
125            .unwrap_or_else(|| panic!("missing `Global` at index: {index}"))
126    }

128    /// Returns the function at the `index`.

131    ///

134        self.funcs
135            .get(index as usize)
136            .copied()
137            .unwrap_or_else(|| panic!("missing `Func` at index: {index}"))

139
140    /// Pushes a new [`Memory`] to the [`InstanceEntity`] under construction.
141    pub fn push_memory(&mut self, memory: Memory) {
142        self.memories.push(memory);
143    }
144
145    /// Pushes a new [`Table`] to the [`InstanceEntity`] under construction.

149

155    /// Pushes a new [`Func`] to the [`InstanceEntity`] under construction.

157        self.funcs.push(func);
158    }
159
160    /// Pushes a new [`Extern`] under the given `name` to the [`InstanceEntity`] under construction.
161    ///
162    /// # Panics
163    ///
164    /// If the name has already been used by an already pushed [`Extern`].
165    pub fn push_export(&mut self, name: &str, new_value: Extern) {
166        if let Some(old_value) = self.exports.get(name) {
167            panic!(
168                "tried to register {new_value:?} for name {name} \
169                but name is already used by {old_value:?}",
170            )
171        }
172        self.exports.insert(name.into(), new_value);
173    }
174
175    /// Pushes the [`DataSegment`] to the [`InstanceEntity`] under construction.
176    pub fn push_data_segment(&mut self, segment: DataSegment) {
177        self.data_segments.push(segment);
178    }
179
180    /// Pushes the [`ElementSegment`] to the [`InstanceEntity`] under construction.
181    pub fn push_element_segment(&mut self, segment: ElementSegment) {
182        self.elem_segments.push(segment);
183    }
184
185    /// Finishes constructing the [`InstanceEntity`].
186    pub fn finish(self) -> InstanceEntity {
187        InstanceEntity {
188            initialized: true,
189            func_types: self.func_types,
190            tables: self.tables.into(),
191            funcs: self.funcs.into(),
192            memories: self.memories.into(),
193            globals: self.globals.into(),
194            exports: self.exports,
195            data_segments: self.data_segments.into(),
196            elem_segments: self.elem_segments.into(),
197        }
198    }
