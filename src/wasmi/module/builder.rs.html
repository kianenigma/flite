99#[derive(Debug, Default)]

107impl ModuleImportsBuilder {
108    /// Finishes construction of [`ModuleImports`].
109    pub fn finish(self) -> ModuleImports {
110        let len_funcs = self.funcs.len();
111        let len_globals = self.globals.len();
112        let len_memories = self.memories.len();
113        let len_tables = self.tables.len();
114        let funcs = self.funcs.into_iter().map(Imported::Func);
115        let tables = self.tables.into_iter().map(Imported::Table);
116        let memories = self.memories.into_iter().map(Imported::Memory);
117        let globals = self.globals.into_iter().map(Imported::Global);
118        let items = funcs
119            .chain(tables)
120            .chain(memories)
121            .chain(globals)
122            .collect::<Box<[_]>>();
123        ModuleImports {
124            items,
125            len_funcs,
126            len_globals,
127            len_memories,
128            len_tables,
129        }
130    }
131}

143impl ModuleHeaderBuilder {
144    /// Pushes the given function types to the [`Module`] under construction.
145    ///
146    /// # Errors
147    ///
148    /// If a function type fails to validate.
149    ///
150    /// # Panics
151    ///
152    /// If this function has already been called on the same [`ModuleBuilder`].
153    pub fn push_func_types<T>(&mut self, func_types: T) -> Result<(), Error>
154    where
155        T: IntoIterator<Item = Result<FuncType, Error>>,
156        <T as IntoIterator>::IntoIter: ExactSizeIterator,
157    {
158        assert!(
159            self.func_types.is_empty(),
160            "tried to initialize module function types twice"
161        );
162        let func_types = func_types.into_iter();
163        // Note: we use `reserve_exact` instead of `reserve` because this
164        //       is the last extension of the vector during the build process
165        //       and optimizes conversion to boxed slice.
166        self.func_types.reserve_exact(func_types.len());
167        for func_type in func_types {
168            let func_type = func_type?;
169            let dedup = self.engine.alloc_func_type(func_type);
170            self.func_types.push(dedup)
171        }
172        Ok(())
173    }
174
175    /// Pushes the given imports to the [`Module`] under construction.
176    ///
177    /// # Errors
178    ///
179    /// If an import fails to validate.
180    ///
181    /// # Panics
182    ///
183    /// If this function has already been called on the same [`ModuleBuilder`].
184    pub fn push_imports<T>(&mut self, imports: T) -> Result<(), Error>
185    where
186        T: IntoIterator<Item = Result<Import, Error>>,
187    {
188        for import in imports {
189            let import = import?;
190            let (name, kind) = import.into_name_and_type();
191            match kind {
192                ExternTypeIdx::Func(func_type_idx) => {
193                    self.imports.funcs.push(name);
194                    let func_type = self.func_types[func_type_idx.into_u32() as usize];
195                    self.funcs.push(func_type);
196                }
197                ExternTypeIdx::Table(table_type) => {
198                    self.imports.tables.push(name);
199                    self.tables.push(table_type);
200                }
201                ExternTypeIdx::Memory(memory_type) => {
202                    self.imports.memories.push(name);
203                    self.memories.push(memory_type);
204                }
205                ExternTypeIdx::Global(global_type) => {
206                    self.imports.globals.push(name);
207                    self.globals.push(global_type);
208                }
209            }
210        }
211        Ok(())
212    }

214    /// Pushes the given function declarations to the [`Module`] under construction.
215    ///
216    /// # Errors
217    ///
218    /// If a function declaration fails to validate.
219    ///
220    /// # Panics
221    ///
222    /// If this function has already been called on the same [`ModuleBuilder`].
223    pub fn push_funcs<T>(&mut self, funcs: T) -> Result<(), Error>
224    where
225        T: IntoIterator<Item = Result<FuncTypeIdx, Error>>,
226        <T as IntoIterator>::IntoIter: ExactSizeIterator,
227    {
228        assert_eq!(
229            self.funcs.len(),
230            self.imports.funcs.len(),
231            "tried to initialize module function declarations twice"
232        );
233        let funcs = funcs.into_iter();
234        // Note: we use `reserve_exact` instead of `reserve` because this
235        //       is the last extension of the vector during the build process
236        //       and optimizes conversion to boxed slice.
237        self.funcs.reserve_exact(funcs.len());
238        self.compiled_funcs.reserve_exact(funcs.len());
239        for func in funcs {
240            let func_type_idx = func?;
241            let func_type = self.func_types[func_type_idx.into_u32() as usize];
242            let Ok(func_index) = u32::try_from(self.funcs.len()) else {
243                panic!("function index out of bounds: {}", self.funcs.len())
244            };
245            self.funcs.push(func_type);
246            let compiled_func = self.engine.alloc_func();
247            self.compiled_funcs.push(compiled_func);
248            self.compiled_funcs_idx
249                .insert(compiled_func, FuncIdx::from(func_index));
250        }
251        Ok(())
252    }

317    ///
318    /// # Errors
319    ///
320    /// If a global variable declaration fails to validate.
321    ///
322    /// # Panics
323    ///
324    /// If this function has already been called on the same [`ModuleBuilder`].
325    pub fn push_globals<T>(&mut self, globals: T) -> Result<(), Error>
326    where
327        T: IntoIterator<Item = Result<Global, Error>>,
328        <T as IntoIterator>::IntoIter: ExactSizeIterator,
329    {
330        assert_eq!(
331            self.globals.len(),
332            self.imports.globals.len(),
333            "tried to initialize module global variable declarations twice"
334        );
335        let globals = globals.into_iter();
336        // Note: we use `reserve_exact` instead of `reserve` because this
337        //       is the last extension of the vector during the build process
338        //       and optimizes conversion to boxed slice.
339        self.globals.reserve_exact(globals.len());
340        for global in globals {
341            let global = global?;
342            let (global_decl, global_init) = global.into_type_and_init();
343            self.globals.push(global_decl);
344            self.globals_init.push(global_init);
345        }
346        Ok(())

405impl ModuleBuilder {
406    /// Reserve space for at least `additional` new data segments.
