37    /// Starts parsing and validating the Wasm bytecode stream.
38    ///
39    /// Returns the compiled and validated Wasm [`Module`] upon success.
40    ///
41    /// # Safety
42    ///
43    /// The caller is responsible to either
44    ///
45    /// 1) Populate the [`ModuleParser`] with a [`Validator`] prior to calling this method, OR;
46    /// 2) Make sure that the provided `stream` yields valid WebAssembly bytecode.
47    ///
48    /// Otherwise this method has undefined behavior.
49    ///
50    /// # Errors
51    ///
52    /// If the Wasm bytecode stream fails to validate.
53    unsafe fn parse_buffered_impl(mut self, mut buffer: &[u8]) -> Result<Module, Error> {
54        let header = Self::parse_buffered_header(&mut self, &mut buffer)?;
55        let builder = Self::parse_buffered_code(&mut self, &mut buffer, header)?;
56        let module = Self::parse_buffered_data(&mut self, &mut buffer, builder)?;
57        Ok(module)
58    }

60    /// Fetch next Wasm module payload and adust the `buffer`.
61    ///
62    /// # Errors
63    ///
64    /// If the parsed Wasm is malformed.
65    fn next_payload<'a>(&mut self, buffer: &mut &'a [u8]) -> Result<(usize, Payload<'a>), Error> {
66        match self.parser.parse(&buffer[..], true)? {
67            Chunk::Parsed { consumed, payload } => Ok((consumed, payload)),
68            Chunk::NeedMoreData(_hint) => {
69                // This is not possible since `eof` is always true.
70                unreachable!()
71            }
72        }
73    }
74
75    /// Consumes the parts of the buffer that have been processed.
76    fn consume_buffer<'a>(consumed: usize, buffer: &mut &'a [u8]) -> &'a [u8] {
77        let (consumed, remaining) = buffer.split_at(consumed);
78        *buffer = remaining;
79        consumed
80    }

82    /// Parse the Wasm module header.
83    ///
84    /// - The Wasm module header is the set of all sections that appear before
85    ///   the Wasm code section.
86    /// - We separate parsing of the Wasm module header since the information of
87    ///   the Wasm module header is required for translating the Wasm code section.
88    ///
89    /// # Errors
90    ///
91    /// If the Wasm bytecode stream fails to parse or validate.
92    fn parse_buffered_header(&mut self, buffer: &mut &[u8]) -> Result<ModuleHeader, Error> {
93        let mut header = ModuleHeaderBuilder::new(&self.engine);
94        loop {
95            let (consumed, payload) = self.next_payload(buffer)?;
96            match payload {
97                Payload::Version {
98                    num,
99                    encoding,
100                    range,
101                } => self.process_version(num, encoding, range),
102                Payload::TypeSection(section) => self.process_types(section, &mut header),
103                Payload::ImportSection(section) => self.process_imports(section, &mut header),
104                Payload::FunctionSection(section) => self.process_functions(section, &mut header),
105                Payload::TableSection(section) => self.process_tables(section, &mut header),
106                Payload::MemorySection(section) => self.process_memories(section, &mut header),
107                Payload::GlobalSection(section) => self.process_globals(section, &mut header),
108                Payload::ExportSection(section) => self.process_exports(section, &mut header),
109                Payload::StartSection { func, range } => {
110                    self.process_start(func, range, &mut header)
111                }
112                Payload::ElementSection(section) => self.process_element(section, &mut header),
113                Payload::DataCountSection { count, range } => self.process_data_count(count, range),
114                Payload::CodeSectionStart { count, range, size } => {
115                    self.process_code_start(count, range, size)?;
116                    Self::consume_buffer(consumed, buffer);
117                    break;
118                }
119                Payload::DataSection(_) => break,
120                Payload::End(_) => break,
121                Payload::CustomSection { .. } => Ok(()),
122                unexpected => self.process_invalid_payload(unexpected),
123            }?;
124            Self::consume_buffer(consumed, buffer);
125        }
126        Ok(header.finish())
127    }

129    /// Parse the Wasm code section entries.
130    ///
131    /// We separate parsing of the Wasm code section since most of a Wasm module
132    /// is made up of code section entries which we can parse and validate more efficiently
133    /// by serving them with a specialized routine.
134    ///
135    /// # Errors
136    ///
137    /// If the Wasm bytecode stream fails to parse or validate.
138    fn parse_buffered_code(
139        &mut self,
140        buffer: &mut &[u8],
141        header: ModuleHeader,
142    ) -> Result<ModuleBuilder, Error> {
143        loop {
144            let (consumed, payload) = self.next_payload(buffer)?;
145            match payload {
146                Payload::CodeSectionEntry(func_body) => {
147                    // Note: Unfortunately the `wasmparser` crate is missing an API
148                    //       to return the byte slice for the respective code section
149                    //       entry payload. Please remove this work around as soon as
150                    //       such an API becomes available.
151                    let bytes = Self::consume_buffer(consumed, buffer);
152                    let remaining = func_body.get_binary_reader().bytes_remaining();
153                    let start = consumed - remaining;
154                    let bytes = &bytes[start..];
155                    self.process_code_entry(func_body, bytes, &header)?;
156                }
157                _ => break,
158            }
159        }
160        Ok(ModuleBuilder::new(header))
161    }
162
163    /// Parse the Wasm data section and finalize parsing.
164    ///
165    /// We separate parsing of the Wasm data section since it is the only Wasm
166    /// section that comes after the Wasm code section that we have to separate
167    /// out for technical reasons.
168    ///
169    /// # Errors
170    ///
171    /// If the Wasm bytecode stream fails to parse or validate.
172    fn parse_buffered_data(
173        &mut self,
174        buffer: &mut &[u8],
175        mut builder: ModuleBuilder,
176    ) -> Result<Module, Error> {
177        loop {
178            let (consumed, payload) = self.next_payload(buffer)?;
179            match payload {
180                Payload::DataSection(section) => {
181                    self.process_data(section, &mut builder)?;
182                }
183                Payload::End(offset) => {
184                    self.process_end(offset)?;
185                    break;
186                }
187                Payload::CustomSection { .. } => {}
188                invalid => self.process_invalid_payload(invalid)?,
189            }
190            Self::consume_buffer(consumed, buffer);
191        }
192        Ok(builder.finish(&self.engine))
193    }
