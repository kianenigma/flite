16//! Weight-related utilities.
17
18use crate::weights::WeightInfo;
19
20use frame_support::weights::Weight;
21use sp_runtime::SaturatedConversion;
22
23/// Extended weight info.
24pub trait WeightInfoExt: WeightInfo {
25	fn enqueue_xcmp_message(size_in_bytes: usize, new_page: bool) -> Weight {
26		let size_in_bytes = size_in_bytes.saturated_into();
27		// The first message for a certain origin consumes some extra reads and writes.
28		// Also, the first message on a page consumes 1 extra write.
29		// For simplicity let's just consider that the first message on a page is also
30		// the first message for that origin.
31		match new_page {
32			true => Self::enqueue_n_bytes_xcmp_message(size_in_bytes),
33			false => {
34				let size_overhead = Self::enqueue_n_bytes_xcmp_message(size_in_bytes) -
35					Self::enqueue_n_bytes_xcmp_message(0);
36				Self::enqueue_2_empty_xcmp_messages()
37					.saturating_sub(Self::enqueue_n_bytes_xcmp_message(0))
38					.saturating_add(size_overhead)
39			},
40		}
41	}
42}
43
44impl<T: WeightInfo> WeightInfoExt for T {}
