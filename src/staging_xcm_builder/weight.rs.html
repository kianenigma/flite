37pub struct FixedWeightBounds<T, C, M>(PhantomData<(T, C, M)>);
38impl<T: Get<Weight>, C: Decode + GetDispatchInfo, M: Get<u32>> WeightBounds<C>
39	for FixedWeightBounds<T, C, M>
40{
41	fn weight(message: &mut Xcm<C>) -> Result<Weight, ()> {
42		tracing::trace!(target: "xcm::weight", ?message, "FixedWeightBounds");
43		let mut instructions_left = M::get();
44		Self::weight_with_limit(message, &mut instructions_left)
45	}
46	fn instr_weight(instruction: &mut Instruction<C>) -> Result<Weight, ()> {
47		Self::instr_weight_with_limit(instruction, &mut u32::max_value())
48	}
49}
50
51impl<T: Get<Weight>, C: Decode + GetDispatchInfo, M> FixedWeightBounds<T, C, M> {
52	fn weight_with_limit(message: &mut Xcm<C>, instrs_limit: &mut u32) -> Result<Weight, ()> {
53		let mut r: Weight = Weight::zero();
54		*instrs_limit = instrs_limit.checked_sub(message.0.len() as u32).ok_or(())?;
55		for instruction in message.0.iter_mut() {
56			r = r
57				.checked_add(&Self::instr_weight_with_limit(instruction, instrs_limit)?)
58				.ok_or(())?;
59		}
60		Ok(r)
61	}
62	fn instr_weight_with_limit(
63		instruction: &mut Instruction<C>,
64		instrs_limit: &mut u32,
65	) -> Result<Weight, ()> {
66		let instr_weight = match instruction {
67			Transact { ref mut call, .. } => call.ensure_decoded()?.get_dispatch_info().call_weight,
68			SetErrorHandler(xcm) | SetAppendix(xcm) | ExecuteWithOrigin { xcm, .. } =>
69				Self::weight_with_limit(xcm, instrs_limit)?,
70			_ => Weight::zero(),
71		};
72		T::get().checked_add(&instr_weight).ok_or(())
73	}
74}

121	}

124/// Function trait for handling some revenue. Similar to a negative imbalance (credit) handler, but
125/// for a `Asset`. Sensible implementations will deposit the asset in some known treasury or
126/// block-author account.
127pub trait TakeRevenue {
128	/// Do something with the given `revenue`, which is a single non-wildcard `Asset`.
129	fn take_revenue(revenue: Asset);
130}
131
132/// Null implementation just burns the revenue.
133impl TakeRevenue for () {
134	fn take_revenue(_revenue: Asset) {}
135}

137/// Simple fee calculator that requires payment in a single fungible at a fixed rate.
138///
139/// The constant `Get` type parameter should be the fungible ID, the amount of it required for one
140/// second of weight and the amount required for 1 MB of proof.
141pub struct FixedRateOfFungible<T: Get<(AssetId, u128, u128)>, R: TakeRevenue>(
142	Weight,
143	u128,
144	PhantomData<(T, R)>,
145);
146impl<T: Get<(AssetId, u128, u128)>, R: TakeRevenue> WeightTrader for FixedRateOfFungible<T, R> {
147	fn new() -> Self {
148		Self(Weight::zero(), 0, PhantomData)
149	}
150
151	fn buy_weight(
152		&mut self,
153		weight: Weight,
154		payment: AssetsInHolding,
155		context: &XcmContext,
156	) -> Result<AssetsInHolding, XcmError> {
157		let (id, units_per_second, units_per_mb) = T::get();
158		tracing::trace!(
159			target: "xcm::weight",
160			?id, ?weight, ?payment, ?context,
161			"FixedRateOfFungible::buy_weight",
162		);
163		let amount = (units_per_second * (weight.ref_time() as u128) /
164			(WEIGHT_REF_TIME_PER_SECOND as u128)) +
165			(units_per_mb * (weight.proof_size() as u128) / (WEIGHT_PROOF_SIZE_PER_MB as u128));
166		if amount == 0 {
167			return Ok(payment)
168		}
169		let unused = payment.checked_sub((id, amount).into()).map_err(|error| {
170			tracing::error!(target: "xcm::weight", ?amount, ?error, "FixedRateOfFungible::buy_weight Failed to substract from payment");
171			XcmError::TooExpensive
172		})?;
173		self.0 = self.0.saturating_add(weight);
174		self.1 = self.1.saturating_add(amount);
175		Ok(unused)
176	}
177
178	fn refund_weight(&mut self, weight: Weight, context: &XcmContext) -> Option<Asset> {
179		let (id, units_per_second, units_per_mb) = T::get();
180		tracing::trace!(target: "xcm::weight", ?id, ?weight, ?context, "FixedRateOfFungible::refund_weight");
181		let weight = weight.min(self.0);
182		let amount = (units_per_second * (weight.ref_time() as u128) /
183			(WEIGHT_REF_TIME_PER_SECOND as u128)) +
184			(units_per_mb * (weight.proof_size() as u128) / (WEIGHT_PROOF_SIZE_PER_MB as u128));
185		self.0 -= weight;
186		self.1 = self.1.saturating_sub(amount);
187		if amount > 0 {
188			Some((id, amount).into())
189		} else {
190			None
191		}
192	}
193}
194
195impl<T: Get<(AssetId, u128, u128)>, R: TakeRevenue> Drop for FixedRateOfFungible<T, R> {
196	fn drop(&mut self) {
197		if self.1 > 0 {
198			R::take_revenue((T::get().0, self.1).into());
199		}
200	}
201}

203/// Weight trader which uses the configured `WeightToFee` to set the right price for weight and then
204/// places any weight bought into the right account.
205pub struct UsingComponents<
206	WeightToFee: WeightToFeeT<Balance = <Fungible as Inspect<AccountId>>::Balance>,
207	AssetIdValue: Get<Location>,
208	AccountId,
209	Fungible: Balanced<AccountId> + Inspect<AccountId>,
210	OnUnbalanced: OnUnbalancedT<Credit<AccountId, Fungible>>,
211>(
212	Weight,
213	Fungible::Balance,
214	PhantomData<(WeightToFee, AssetIdValue, AccountId, Fungible, OnUnbalanced)>,
215);
216impl<
217		WeightToFee: WeightToFeeT<Balance = <Fungible as Inspect<AccountId>>::Balance>,
218		AssetIdValue: Get<Location>,
219		AccountId,
220		Fungible: Balanced<AccountId> + Inspect<AccountId>,
221		OnUnbalanced: OnUnbalancedT<Credit<AccountId, Fungible>>,
222	> WeightTrader for UsingComponents<WeightToFee, AssetIdValue, AccountId, Fungible, OnUnbalanced>
223{
224	fn new() -> Self {
225		Self(Weight::zero(), Zero::zero(), PhantomData)
226	}
227
228	fn buy_weight(
229		&mut self,
230		weight: Weight,
231		payment: AssetsInHolding,
232		context: &XcmContext,
233	) -> Result<AssetsInHolding, XcmError> {
234		tracing::trace!(target: "xcm::weight", ?weight, ?payment, ?context, "UsingComponents::buy_weight");
235		let amount = WeightToFee::weight_to_fee(&weight);
236		let u128_amount: u128 = amount.try_into().map_err(|_| {
237			tracing::debug!(target: "xcm::weight", ?amount, "Weight fee could not be converted");
238			XcmError::Overflow
239		})?;
240		let required = Asset { id: AssetId(AssetIdValue::get()), fun: Fungible(u128_amount) };
241		let unused = payment.checked_sub(required).map_err(|error| {
242			tracing::debug!(target: "xcm::weight", ?error, "Failed to substract from payment");
243			XcmError::TooExpensive
244		})?;
245		self.0 = self.0.saturating_add(weight);
246		self.1 = self.1.saturating_add(amount);
247		Ok(unused)
248	}
249
250	fn refund_weight(&mut self, weight: Weight, context: &XcmContext) -> Option<Asset> {
251		tracing::trace!(target: "xcm::weight", ?weight, ?context, available_weight = ?self.0, available_amount = ?self.1, "UsingComponents::refund_weight");
252		let weight = weight.min(self.0);
253		let amount = WeightToFee::weight_to_fee(&weight);
254		self.0 -= weight;
255		self.1 = self.1.saturating_sub(amount);
256		let amount: u128 = amount.saturated_into();
257		tracing::trace!(target: "xcm::weight", ?amount, "UsingComponents::refund_weight");
258		if amount > 0 {
259			Some((AssetIdValue::get(), amount).into())
260		} else {
261			None
262		}
263	}
264}
265impl<
266		WeightToFee: WeightToFeeT<Balance = <Fungible as Inspect<AccountId>>::Balance>,
267		AssetId: Get<Location>,
268		AccountId,
269		Fungible: Balanced<AccountId> + Inspect<AccountId>,
270		OnUnbalanced: OnUnbalancedT<Credit<AccountId, Fungible>>,
271	> Drop for UsingComponents<WeightToFee, AssetId, AccountId, Fungible, OnUnbalanced>
272{
273	fn drop(&mut self) {
274		OnUnbalanced::on_unbalanced(Fungible::issue(self.1));
275	}
