16/// Bytes which represent a public key.
17///
18/// Includes an extra byte for an SEC1 tag.
19pub type PublicKeyBytes = [u8; KEY_SIZE + 1];
20
21/// Trait for key types which can be derived using BIP32.
22pub trait PublicKey: Sized {
23    /// Initialize this key from bytes.
24    fn from_bytes(bytes: PublicKeyBytes) -> Result<Self>;
25
26    /// Serialize this key as bytes.
27    fn to_bytes(&self) -> PublicKeyBytes;
28
29    /// Derive a child key from a parent key and a provided tweak value.
30    fn derive_child(&self, other: PrivateKeyBytes) -> Result<Self>;
31
32    /// Compute a 4-byte key fingerprint for this public key.
33    ///
34    /// Default implementation uses `RIPEMD160(SHA256(public_key))`.
35    fn fingerprint(&self) -> KeyFingerprint {
36        let digest = Ripemd160::digest(Sha256::digest(self.to_bytes()));
37        digest[..4].try_into().expect("digest truncated")
38    }
39
40    /// Derive a tweak value that can be used to generate the child key (see [`derive_child`]).
41    ///
42    /// The `chain_code` is either a newly initialized one,
43    /// or one obtained from the previous invocation of `derive_tweak()`
44    /// (for a multi-level derivation).
45    ///
46    /// **Warning:** make sure that if you are creating a new `chain_code`, you are doing so
47    /// in a cryptographically safe way.
48    /// Normally this would be done according to BIP-39 (within [`ExtendedPrivateKey::new`]).
49    ///
50    /// **Note:** `child_number` cannot be a hardened one (will result in an error).
51    fn derive_tweak(
52        &self,
53        chain_code: &ChainCode,
54        child_number: ChildNumber,
55    ) -> Result<(PrivateKeyBytes, ChainCode)> {
56        if child_number.is_hardened() {
57            // Cannot derive child public keys for hardened `ChildNumber`s
58            return Err(Error::ChildNumber);
59        }
60
61        let mut hmac = HmacSha512::new_from_slice(chain_code).map_err(|_| Error::Crypto)?;
62
63        hmac.update(&self.to_bytes());
64        hmac.update(&child_number.to_bytes());
65
66        let result = hmac.finalize().into_bytes();
67        let (tweak_bytes, chain_code_bytes) = result.split_at(KEY_SIZE);
68
69        // Note that at this point we are only asserting that `tweak_bytes` have the expected size.
70        // Checking if it actually fits the curve scalar happens in `derive_child()`.
71        let tweak = tweak_bytes.try_into()?;
72
73        let chain_code = chain_code_bytes.try_into()?;
74
75        Ok((tweak, chain_code))
76    }
77}

79#[cfg(feature = "secp256k1")]
80impl PublicKey for k256::PublicKey {
81    fn from_bytes(bytes: PublicKeyBytes) -> Result<Self> {
82        Ok(k256::PublicKey::from_sec1_bytes(&bytes)?)
83    }
84
85    fn to_bytes(&self) -> PublicKeyBytes {
86        self.to_encoded_point(true)
87            .as_bytes()
88            .try_into()
89            .expect("malformed public key")
90    }
91
92    fn derive_child(&self, other: PrivateKeyBytes) -> Result<Self> {
93        let child_scalar =
94            Option::<k256::NonZeroScalar>::from(k256::NonZeroScalar::from_repr(other.into()))
95                .ok_or(Error::Crypto)?;
96
97        let child_point = self.to_projective() + (k256::AffinePoint::generator() * *child_scalar);
98        Self::from_affine(child_point.into()).map_err(|_| Error::Crypto)
99    }
100}

102#[cfg(feature = "secp256k1")]
103impl PublicKey for k256::ecdsa::VerifyingKey {
104    fn from_bytes(bytes: PublicKeyBytes) -> Result<Self> {
105        Ok(k256::ecdsa::VerifyingKey::from_sec1_bytes(&bytes)?)
106    }
107
108    fn to_bytes(&self) -> PublicKeyBytes {
109        k256::CompressedPoint::from(self)
110            .as_slice()
111            .try_into()
112            .expect("malformed key")
113    }
114
115    fn derive_child(&self, other: PrivateKeyBytes) -> Result<Self> {
116        k256::PublicKey::from(self)
117            .derive_child(other)
118            .map(Into::into)
119    }
120}

136#[cfg(feature = "secp256k1-ffi")]
137impl PublicKey for secp256k1_ffi::PublicKey {
138    fn from_bytes(bytes: PublicKeyBytes) -> Result<Self> {
139        Ok(secp256k1_ffi::PublicKey::from_slice(&bytes)?)
140    }
141
142    fn to_bytes(&self) -> PublicKeyBytes {
143        self.serialize()
144    }
145
146    fn derive_child(&self, bytes: PrivateKeyBytes) -> Result<Self> {
147        use secp256k1_ffi::{Secp256k1, VerifyOnly};
148
149        let engine = Secp256k1::<VerifyOnly>::verification_only();
150        let scalar = secp256k1_ffi::Scalar::from_be_bytes(bytes)?;
151        Ok(self.add_exp_tweak(&engine, &scalar)?)
152    }
153}
