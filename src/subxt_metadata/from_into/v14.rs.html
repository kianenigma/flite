16impl TryFrom<v14::RuntimeMetadataV14> for Metadata {
17    type Error = TryFromError;
18    fn try_from(value: v14::RuntimeMetadataV14) -> Result<Self, Self::Error> {
19        // Convert to v15 and then convert that into Metadata.
20        v14_to_v15(value)?.try_into()
21    }
22}
23
24impl From<Metadata> for v14::RuntimeMetadataV14 {
25    fn from(val: Metadata) -> Self {
26        let v15 = val.into();
27        v15_to_v14(v15)
28    }
29}

31fn v15_to_v14(mut metadata: v15::RuntimeMetadataV15) -> v14::RuntimeMetadataV14 {
32    let types = &mut metadata.types;
33
34    // In subxt we care about the `Address`, `Call`, `Signature` and `Extra` types.
35    let extrinsic_type = scale_info::Type {
36        path: scale_info::Path {
37            segments: vec![
38                "primitives".to_owned(),
39                "runtime".to_owned(),
40                "generic".to_owned(),
41                "UncheckedExtrinsic".to_owned(),
42            ],
43        },
44        type_params: vec![
45            scale_info::TypeParameter::<scale_info::form::PortableForm> {
46                name: "Address".to_owned(),
47                ty: Some(metadata.extrinsic.address_ty),
48            },
49            scale_info::TypeParameter::<scale_info::form::PortableForm> {
50                name: "Call".to_owned(),
51                ty: Some(metadata.extrinsic.call_ty),
52            },
53            scale_info::TypeParameter::<scale_info::form::PortableForm> {
54                name: "Signature".to_owned(),
55                ty: Some(metadata.extrinsic.signature_ty),
56            },
57            scale_info::TypeParameter::<scale_info::form::PortableForm> {
58                name: "Extra".to_owned(),
59                ty: Some(metadata.extrinsic.extra_ty),
60            },
61        ],
62        type_def: scale_info::TypeDef::Composite(scale_info::TypeDefComposite { fields: vec![] }),
63        docs: vec![],
64    };
65    let extrinsic_type_id = types.types.len() as u32;
66
67    types.types.push(scale_info::PortableType {
68        id: extrinsic_type_id,
69        ty: extrinsic_type,
70    });
71
72    v14::RuntimeMetadataV14 {
73        types: metadata.types,
74        pallets: metadata
75            .pallets
76            .into_iter()
77            .map(|pallet| frame_metadata::v14::PalletMetadata {
78                name: pallet.name,
79                storage: pallet
80                    .storage
81                    .map(|storage| frame_metadata::v14::PalletStorageMetadata {
82                        prefix: storage.prefix,
83                        entries: storage
84                            .entries
85                            .into_iter()
86                            .map(|entry| {
87                                let modifier = match entry.modifier {
88                                    frame_metadata::v15::StorageEntryModifier::Optional => {
89                                        frame_metadata::v14::StorageEntryModifier::Optional
90                                    }
91                                    frame_metadata::v15::StorageEntryModifier::Default => {
92                                        frame_metadata::v14::StorageEntryModifier::Default
93                                    }
94                                };
95
96                                let ty = match entry.ty {
97                                    frame_metadata::v15::StorageEntryType::Plain(ty) => {
98                                        frame_metadata::v14::StorageEntryType::Plain(ty)
99                                    },
100                                    frame_metadata::v15::StorageEntryType::Map {
101                                        hashers,
102                                        key,
103                                        value,
104                                    } => frame_metadata::v14::StorageEntryType::Map {
105                                        hashers: hashers.into_iter().map(|hasher| match hasher {
106                                            frame_metadata::v15::StorageHasher::Blake2_128 => frame_metadata::v14::StorageHasher::Blake2_128,
107                                            frame_metadata::v15::StorageHasher::Blake2_256 => frame_metadata::v14::StorageHasher::Blake2_256,
108                                            frame_metadata::v15::StorageHasher::Blake2_128Concat  => frame_metadata::v14::StorageHasher::Blake2_128Concat ,
109                                            frame_metadata::v15::StorageHasher::Twox128 => frame_metadata::v14::StorageHasher::Twox128,
110                                            frame_metadata::v15::StorageHasher::Twox256 => frame_metadata::v14::StorageHasher::Twox256,
111                                            frame_metadata::v15::StorageHasher::Twox64Concat => frame_metadata::v14::StorageHasher::Twox64Concat,
112                                            frame_metadata::v15::StorageHasher::Identity=> frame_metadata::v14::StorageHasher::Identity,
113                                        }).collect(),
114                                        key,
115                                        value,
116                                    },
117                                };
118
119                                frame_metadata::v14::StorageEntryMetadata {
120                                    name: entry.name,
121                                    modifier,
122                                    ty,
123                                    default: entry.default,
124                                    docs: entry.docs,
125                                }
126                            })
127                            .collect(),
128                    }),
129                calls: pallet.calls.map(|calls| frame_metadata::v14::PalletCallMetadata { ty: calls.ty } ),
130                event: pallet.event.map(|event| frame_metadata::v14::PalletEventMetadata { ty: event.ty } ),
131                constants: pallet.constants.into_iter().map(|constant| frame_metadata::v14::PalletConstantMetadata {
132                    name: constant.name,
133                    ty: constant.ty,
134                    value: constant.value,
135                    docs: constant.docs,
136                } ).collect(),
137                error: pallet.error.map(|error| frame_metadata::v14::PalletErrorMetadata { ty: error.ty } ),
138                index: pallet.index,
139            })
140            .collect(),
141        extrinsic: frame_metadata::v14::ExtrinsicMetadata {
142            ty: extrinsic_type_id.into(),
143            version: metadata.extrinsic.version,
144            signed_extensions: metadata.extrinsic.signed_extensions.into_iter().map(|ext| {
145                frame_metadata::v14::SignedExtensionMetadata {
146                    identifier: ext.identifier,
147                    ty: ext.ty,
148                    additional_signed: ext.additional_signed,
149                }
150            }).collect()
151        },
152        ty: metadata.ty,
153    }
154}

174                        prefix: storage.prefix,

178                            .map(|entry| {
179                                let modifier = match entry.modifier {
180                                    frame_metadata::v14::StorageEntryModifier::Optional => {
181                                        frame_metadata::v15::StorageEntryModifier::Optional
182                                    }
183                                    frame_metadata::v14::StorageEntryModifier::Default => {
184                                        frame_metadata::v15::StorageEntryModifier::Default
185                                    }
186                                };
187
188                                let ty = match entry.ty {
189                                    frame_metadata::v14::StorageEntryType::Plain(ty) => {
190                                        frame_metadata::v15::StorageEntryType::Plain(ty)
191                                    },
192                                    frame_metadata::v14::StorageEntryType::Map {
193                                        hashers,
194                                        key,
195                                        value,
196                                    } => frame_metadata::v15::StorageEntryType::Map {
197                                        hashers: hashers.into_iter().map(|hasher| match hasher {
198                                            frame_metadata::v14::StorageHasher::Blake2_128 => frame_metadata::v15::StorageHasher::Blake2_128,
199                                            frame_metadata::v14::StorageHasher::Blake2_256 => frame_metadata::v15::StorageHasher::Blake2_256,
200                                            frame_metadata::v14::StorageHasher::Blake2_128Concat  => frame_metadata::v15::StorageHasher::Blake2_128Concat ,
201                                            frame_metadata::v14::StorageHasher::Twox128 => frame_metadata::v15::StorageHasher::Twox128,
202                                            frame_metadata::v14::StorageHasher::Twox256 => frame_metadata::v15::StorageHasher::Twox256,
203                                            frame_metadata::v14::StorageHasher::Twox64Concat => frame_metadata::v15::StorageHasher::Twox64Concat,
204                                            frame_metadata::v14::StorageHasher::Identity=> frame_metadata::v15::StorageHasher::Identity,
205                                        }).collect(),
206                                        key,
207                                        value,
208                                    },
209                                };
210
211                                frame_metadata::v15::StorageEntryMetadata {
212                                    name: entry.name,
213                                    modifier,
214                                    ty,
215                                    default: entry.default,
216                                    docs: entry.docs,
217                                }
218                            })
219                            .collect(),
220                    }),
221                calls: pallet.calls.map(|calls| frame_metadata::v15::PalletCallMetadata { ty: calls.ty } ),
222                event: pallet.event.map(|event| frame_metadata::v15::PalletEventMetadata { ty: event.ty } ),
223                constants: pallet.constants.into_iter().map(|constant| frame_metadata::v15::PalletConstantMetadata {
224                    name: constant.name,
225                    ty: constant.ty,
226                    value: constant.value,

229                error: pallet.error.map(|error| frame_metadata::v15::PalletErrorMetadata { ty: error.ty } ),

231                docs: Default::default(),
232            })
233            .collect(),
234        extrinsic: frame_metadata::v15::ExtrinsicMetadata {
235            version: metadata.extrinsic.version,
236            signed_extensions: metadata.extrinsic.signed_extensions.into_iter().map(|ext| {
237                frame_metadata::v15::SignedExtensionMetadata {
238                    identifier: ext.identifier,
239                    ty: ext.ty,
240                    additional_signed: ext.additional_signed,
241                }
242            }).collect(),
243            address_ty: extrinsic_parts.address.into(),
244            call_ty: extrinsic_parts.call.into(),
245            signature_ty: extrinsic_parts.signature.into(),
246            extra_ty: extrinsic_parts.extra.into(),
247        },
248        ty: metadata.ty,
249        apis: Default::default(),
250        outer_enums,
251        custom: v15::CustomMetadata {
252            map: Default::default(),
253        },
254    })

257/// The type IDs extracted from the metadata that represent the
258/// generic type parameters passed to the `UncheckedExtrinsic` from
259/// the substrate-based chain.
260struct ExtrinsicPartTypeIds {
261    address: u32,
262    call: u32,
263    signature: u32,
264    extra: u32,
265}
266
267impl ExtrinsicPartTypeIds {
268    /// Extract the generic type parameters IDs from the extrinsic type.
269    fn new(metadata: &v14::RuntimeMetadataV14) -> Result<Self, TryFromError> {
270        const ADDRESS: &str = "Address";
271        const CALL: &str = "Call";
272        const SIGNATURE: &str = "Signature";
273        const EXTRA: &str = "Extra";
274
275        let extrinsic_id = metadata.extrinsic.ty.id;
276        let Some(extrinsic_ty) = metadata.types.resolve(extrinsic_id) else {
277            return Err(TryFromError::TypeNotFound(extrinsic_id));
278        };
279
280        let params: HashMap<_, _> = extrinsic_ty
281            .type_params
282            .iter()
283            .map(|ty_param| {
284                let Some(ty) = ty_param.ty else {
285                    return Err(TryFromError::TypeNameNotFound(ty_param.name.clone()));
286                };
287
288                Ok((ty_param.name.as_str(), ty.id))
289            })
290            .collect::<Result<_, _>>()?;
291
292        let Some(address) = params.get(ADDRESS) else {
293            return Err(TryFromError::TypeNameNotFound(ADDRESS.into()));
294        };
295        let Some(call) = params.get(CALL) else {
296            return Err(TryFromError::TypeNameNotFound(CALL.into()));
297        };
298        let Some(signature) = params.get(SIGNATURE) else {
299            return Err(TryFromError::TypeNameNotFound(SIGNATURE.into()));
300        };
301        let Some(extra) = params.get(EXTRA) else {
302            return Err(TryFromError::TypeNameNotFound(EXTRA.into()));
303        };
304
305        Ok(ExtrinsicPartTypeIds {
306            address: *address,
307            call: *call,
308            signature: *signature,
309            extra: *extra,
310        })
311    }
312}

314fn generate_outer_enums(
315    metadata: &mut v14::RuntimeMetadataV14,
316) -> Result<v15::OuterEnums<scale_info::form::PortableForm>, TryFromError> {
317    let find_type = |name: &str| {
318        metadata.types.types.iter().find_map(|ty| {
319            let ident = ty.ty.path.ident()?;
320
321            if ident != name {
322                return None;
323            }
324
325            let TypeDef::Variant(_) = &ty.ty.type_def else {
326                return None;
327            };
328
329            Some((ty.id, ty.ty.path.segments.clone()))
330        })
331    };
332
333    let Some((call_enum, mut call_path)) = find_type("RuntimeCall") else {
334        return Err(TryFromError::TypeNameNotFound("RuntimeCall".into()));
335    };
336
337    let Some((event_enum, _)) = find_type("RuntimeEvent") else {
338        return Err(TryFromError::TypeNameNotFound("RuntimeEvent".into()));
339    };
340
341    let error_enum = if let Some((error_enum, _)) = find_type("RuntimeError") {
342        error_enum
343    } else {
344        let Some(last) = call_path.last_mut() else {
345            return Err(TryFromError::InvalidTypePath("RuntimeCall".into()));
346        };
347        "RuntimeError".clone_into(last);
348        generate_outer_error_enum_type(metadata, call_path)
349    };
350
351    Ok(v15::OuterEnums {
352        call_enum_ty: call_enum.into(),
353        event_enum_ty: event_enum.into(),
354        error_enum_ty: error_enum.into(),
355    })
356}

358/// Generates an outer `RuntimeError` enum type and adds it to the metadata.
359///
360/// Returns the id of the generated type from the registry.
361fn generate_outer_error_enum_type(
362    metadata: &mut v14::RuntimeMetadataV14,
363    path_segments: Vec<String>,
364) -> u32 {
365    let variants: Vec<_> = metadata
366        .pallets
367        .iter()
368        .filter_map(|pallet| {
369            let error = pallet.error.as_ref()?;
370
371            // Note:  using the `alloc::format!` macro like in `let path = format!("{}Error", pallet.name);`
372            // leads to linker errors about extern function `_Unwind_Resume` not being defined.
373            let mut path = String::new();
374            write!(path, "{}Error", pallet.name).expect("Cannot panic, qed;");
375            let ty = error.ty.id.into();
376
377            Some(scale_info::Variant {
378                name: pallet.name.clone(),
379                fields: vec![scale_info::Field {
380                    name: None,
381                    ty,
382                    type_name: Some(path),
383                    docs: vec![],
384                }],
385                index: pallet.index,
386                docs: vec![],
387            })
388        })
389        .collect();
390
391    let enum_type = scale_info::Type {
392        path: scale_info::Path {
393            segments: path_segments,
394        },
395        type_params: vec![],
396        type_def: scale_info::TypeDef::Variant(scale_info::TypeDefVariant { variants }),
397        docs: vec![],
398    };
399
400    let enum_type_id = metadata.types.types.len() as u32;
401
402    metadata.types.types.push(scale_info::PortableType {
403        id: enum_type_id,
404        ty: enum_type,
405    });
406
407    enum_type_id
408}

603            }
