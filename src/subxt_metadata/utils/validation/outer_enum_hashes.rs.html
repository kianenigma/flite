3use hashbrown::HashMap;
4
5use scale_info::{PortableRegistry, TypeDef};

7use crate::{
8    utils::{

10        validation::{get_type_def_variant_hash, get_type_hash},
11    },
12    Metadata,
13};

15use super::{concat_and_hash3, Hash, HASH_LEN};

17/// Hash representations of the `frame_metadata::v15::OuterEnums`.
18pub struct OuterEnumHashes {
19    call_hash: (u32, Hash),
20    error_hash: (u32, Hash),
21    event_hash: (u32, Hash),
22}
23
24impl OuterEnumHashes {
25    /// Constructs new `OuterEnumHashes` from metadata. If `only_these_variants` is set, the enums are stripped down to only these variants, before their hashes are calculated.
26    pub fn new(
27        metadata: &Metadata,
28        specific_pallets: Option<&[&str]>,
29        specific_runtimes: Option<&[&str]>,
30    ) -> Self {
31        let filter = |names: Option<&[&str]>, name: &str| match names {
32            Some(names) => names.contains(&name),
33            None => true,
34        };
35        let mut check_enum_type_id = retain::keep_outer_enum(
36            metadata,
37            &mut |name| filter(specific_pallets, name),
38            &mut |name| filter(specific_runtimes, name),
39        );
40
41        let variants = |filter: bool| {
42            if !filter {
43                specific_pallets
44            } else {
45                None
46            }
47        };
48        fn get_enum_hash(
49            registry: &PortableRegistry,
50            id: u32,
51            only_these_variants: Option<&[&str]>,
52        ) -> Hash {
53            let ty = registry
54                .types
55                .get(id as usize)
56                .expect("Metadata should contain enum type in registry");
57
58            if let TypeDef::Variant(variant) = &ty.ty.type_def {
59                get_type_def_variant_hash(
60                    registry,
61                    variant,
62                    only_these_variants,
63                    &mut HashMap::new(),
64                    // ignored, because not computed yet...
65                    &OuterEnumHashes::empty(),
66                )
67            } else {
68                get_type_hash(registry, id, &OuterEnumHashes::empty())
69            }
70        }
71        let enums = &metadata.outer_enums;
72
73        let call_variants = variants(check_enum_type_id(enums.call_enum_ty));
74        let call_hash = get_enum_hash(metadata.types(), enums.call_enum_ty, call_variants);
75        let event_variants = variants(check_enum_type_id(enums.event_enum_ty));
76        let event_hash = get_enum_hash(metadata.types(), enums.event_enum_ty, event_variants);
77        let error_variants = variants(check_enum_type_id(enums.error_enum_ty));
78        let error_hash = get_enum_hash(metadata.types(), enums.error_enum_ty, error_variants);
79
80        Self {
81            call_hash: (enums.call_enum_ty, call_hash),
82            error_hash: (enums.error_enum_ty, error_hash),
83            event_hash: (enums.event_enum_ty, event_hash),
84        }
85    }
86
87    /// Constructs empty `OuterEnumHashes` with type ids that are never a real type id.
88    /// Can be used as a placeholder when outer enum hashes are required but should be ignored.
89    pub fn empty() -> Self {
90        Self {
91            call_hash: (u32::MAX, [0; HASH_LEN]),
92            error_hash: (u32::MAX, [0; HASH_LEN]),
93            event_hash: (u32::MAX, [0; HASH_LEN]),
94        }
95    }
96
97    /// Returns a combined hash of the top level enums.
98    pub fn combined_hash(&self) -> Hash {
99        concat_and_hash3(&self.call_hash.1, &self.error_hash.1, &self.event_hash.1)
100    }
101
102    /// Checks if a type is one of the 3 top level enum types. If so, returns Some(hash).
103    ///
104    /// This is useful, because top level enums are sometimes stripped down to only certain pallets.
105    /// The hashes of these stripped down types are stored in this struct.
106    pub fn resolve(&self, id: u32) -> Option<[u8; HASH_LEN]> {
107        match id {
108            e if e == self.error_hash.0 => Some(self.error_hash.1),
109            e if e == self.event_hash.0 => Some(self.event_hash.1),
110            e if e == self.call_hash.0 => Some(self.call_hash.1),
111            _ => None,
112        }
113    }
114}
