19// The number of bytes our `hash` function produces.
20pub(crate) const HASH_LEN: usize = 32;
21pub type Hash = [u8; HASH_LEN];
22
23/// Internal byte representation for various metadata types utilized for
24/// generating deterministic hashes between different rust versions.
25#[repr(u8)]
26enum TypeBeingHashed {
27    Composite,
28    Variant,
29    Sequence,
30    Array,
31    Tuple,
32    Primitive,
33    Compact,
34    BitSequence,
35}
36
37/// Hashing function utilized internally.
38fn hash(data: &[u8]) -> Hash {
39    sp_crypto_hashing::twox_256(data)
40}
41
42/// XOR two hashes together. Only use this when you don't care about the order
43/// of the things you're hashing together.
44fn xor(a: Hash, b: Hash) -> Hash {
45    let mut out = [0u8; HASH_LEN];
46    for (idx, (a, b)) in a.into_iter().zip(b).enumerate() {
47        out[idx] = a ^ b;
48    }
49    out
50}
51
52// Combine some number of HASH_LEN byte hashes and output a single HASH_LEN
53// byte hash to uniquely represent the inputs.
54macro_rules! count_idents {
55    () => { 0 };
56    ($n:ident $($rest:ident)*) => { 1 + count_idents!($($rest)*) }
57}
58macro_rules! concat_and_hash_n {
59    ($name:ident($($arg:ident)+)) => {
60        fn $name($($arg: &Hash),+) -> Hash {
61            let mut out = [0u8; HASH_LEN * count_idents!($($arg)+)];
62            let mut start = 0;
63            $(
64                out[start..start+HASH_LEN].copy_from_slice(&$arg[..]);
65                #[allow(unused_assignments)]
66                { start += HASH_LEN; }
67            )+
68            hash(&out)
69        }
70    }
71}
72concat_and_hash_n!(concat_and_hash2(a b));
73concat_and_hash_n!(concat_and_hash3(a b c));
74concat_and_hash_n!(concat_and_hash4(a b c d));
75concat_and_hash_n!(concat_and_hash5(a b c d e));
76concat_and_hash_n!(concat_and_hash6(a b c d e f));

78/// Obtain the hash representation of a `scale_info::Field`.
79fn get_field_hash(
80    registry: &PortableRegistry,
81    field: &Field<PortableForm>,
82    cache: &mut HashMap<u32, CachedHash>,
83    outer_enum_hashes: &OuterEnumHashes,
84) -> Hash {
85    let field_name_bytes = match &field.name {
86        Some(name) => hash(name.as_bytes()),
87        None => [0u8; HASH_LEN],
88    };
89
90    concat_and_hash2(
91        &field_name_bytes,
92        &get_type_hash_recurse(registry, field.ty.id, cache, outer_enum_hashes),
93    )
94}

96/// Obtain the hash representation of a `scale_info::Variant`.
97fn get_variant_hash(
98    registry: &PortableRegistry,
99    var: &Variant<PortableForm>,
100    cache: &mut HashMap<u32, CachedHash>,
101    outer_enum_hashes: &OuterEnumHashes,
102) -> Hash {
103    let variant_name_bytes = hash(var.name.as_bytes());
104    let variant_field_bytes = var.fields.iter().fold([0u8; HASH_LEN], |bytes, field| {
105        // EncodeAsType and DecodeAsType don't care about variant field ordering,
106        // so XOR the fields to ensure that it doesn't matter.
107        xor(
108            bytes,
109            get_field_hash(registry, field, cache, outer_enum_hashes),
110        )
111    });
112
113    concat_and_hash2(&variant_name_bytes, &variant_field_bytes)
114}

116fn get_type_def_variant_hash(
117    registry: &PortableRegistry,
118    variant: &TypeDefVariant<PortableForm>,
119    only_these_variants: Option<&[&str]>,
120    cache: &mut HashMap<u32, CachedHash>,
121    outer_enum_hashes: &OuterEnumHashes,
122) -> Hash {
123    let variant_id_bytes = [TypeBeingHashed::Variant as u8; HASH_LEN];
124    let variant_field_bytes = variant.variants.iter().fold([0u8; HASH_LEN], |bytes, var| {
125        // With EncodeAsType and DecodeAsType we no longer care which order the variants are in,
126        // as long as all of the names+types are there. XOR to not care about ordering.
127        let should_hash = only_these_variants
128            .as_ref()
129            .map(|only_these_variants| only_these_variants.contains(&var.name.as_str()))
130            .unwrap_or(true);
131        if should_hash {
132            xor(
133                bytes,
134                get_variant_hash(registry, var, cache, outer_enum_hashes),
135            )
136        } else {
137            bytes
138        }
139    });
140    concat_and_hash2(&variant_id_bytes, &variant_field_bytes)
141}

143/// Obtain the hash representation of a `scale_info::TypeDef`.
144fn get_type_def_hash(
145    registry: &PortableRegistry,
146    ty_def: &TypeDef<PortableForm>,
147    cache: &mut HashMap<u32, CachedHash>,
148    outer_enum_hashes: &OuterEnumHashes,
149) -> Hash {
150    match ty_def {
151        TypeDef::Composite(composite) => {
152            let composite_id_bytes = [TypeBeingHashed::Composite as u8; HASH_LEN];
153            let composite_field_bytes =
154                composite
155                    .fields
156                    .iter()
157                    .fold([0u8; HASH_LEN], |bytes, field| {
158                        // With EncodeAsType and DecodeAsType we no longer care which order the fields are in,
159                        // as long as all of the names+types are there. XOR to not care about ordering.
160                        xor(
161                            bytes,
162                            get_field_hash(registry, field, cache, outer_enum_hashes),
163                        )
164                    });
165            concat_and_hash2(&composite_id_bytes, &composite_field_bytes)
166        }
167        TypeDef::Variant(variant) => {
168            get_type_def_variant_hash(registry, variant, None, cache, outer_enum_hashes)
169        }
170        TypeDef::Sequence(sequence) => concat_and_hash2(
171            &[TypeBeingHashed::Sequence as u8; HASH_LEN],
172            &get_type_hash_recurse(registry, sequence.type_param.id, cache, outer_enum_hashes),
173        ),
174        TypeDef::Array(array) => {
175            // Take length into account too; different length must lead to different hash.
176            let array_id_bytes = {
177                let mut a = [0u8; HASH_LEN];
178                a[0] = TypeBeingHashed::Array as u8;
179                a[1..5].copy_from_slice(&array.len.to_be_bytes());
180                a
181            };
182            concat_and_hash2(
183                &array_id_bytes,
184                &get_type_hash_recurse(registry, array.type_param.id, cache, outer_enum_hashes),
185            )
186        }
187        TypeDef::Tuple(tuple) => {
188            let mut bytes = hash(&[TypeBeingHashed::Tuple as u8]);
189            for field in &tuple.fields {
190                bytes = concat_and_hash2(
191                    &bytes,
192                    &get_type_hash_recurse(registry, field.id, cache, outer_enum_hashes),
193                );
194            }
195            bytes
196        }
197        TypeDef::Primitive(primitive) => {
198            // Cloning the 'primitive' type should essentially be a copy.
199            hash(&[TypeBeingHashed::Primitive as u8, primitive.clone() as u8])
200        }
201        TypeDef::Compact(compact) => concat_and_hash2(
202            &[TypeBeingHashed::Compact as u8; HASH_LEN],
203            &get_type_hash_recurse(registry, compact.type_param.id, cache, outer_enum_hashes),
204        ),
205        TypeDef::BitSequence(bitseq) => concat_and_hash3(
206            &[TypeBeingHashed::BitSequence as u8; HASH_LEN],
207            &get_type_hash_recurse(registry, bitseq.bit_order_type.id, cache, outer_enum_hashes),
208            &get_type_hash_recurse(registry, bitseq.bit_store_type.id, cache, outer_enum_hashes),
209        ),
210    }
211}

213/// indicates whether a hash has been fully computed for a type or not
214#[derive(Clone, Debug)]
215pub enum CachedHash {
216    /// hash not known yet, but computation has already started
217    Recursive,
218    /// hash of the type, computation was finished
219    Hash(Hash),
220}
221
222impl CachedHash {
223    fn hash(&self) -> Hash {
224        match &self {
225            CachedHash::Hash(hash) => *hash,
226            CachedHash::Recursive => [123; HASH_LEN], // some magical value
227        }
228    }
229}

231/// Obtain the hash representation of a `scale_info::Type` identified by id.
232///
233/// Hashes of the outer enums (call, event, error) should be computed prior to this
234/// and passed in as the `outer_enum_hashes` argument. Whenever a type is encountered that
235/// is one of the outer enums, the procomputed hash is used instead of computing a new one.
236///
237/// The reason for this unintuitive behavior is that we sometimes want to trim the outer enum types
238/// beforehand to only include certain pallets, which affects their hash values.
239pub fn get_type_hash(
240    registry: &PortableRegistry,
241    id: u32,
242    outer_enum_hashes: &OuterEnumHashes,
243) -> Hash {
244    get_type_hash_recurse(registry, id, &mut HashMap::new(), outer_enum_hashes)
245}
246
247/// Obtain the hash representation of a `scale_info::Type` identified by id.
248fn get_type_hash_recurse(
249    registry: &PortableRegistry,
250    id: u32,
251    cache: &mut HashMap<u32, CachedHash>,
252    outer_enum_hashes: &OuterEnumHashes,
253) -> Hash {
254    // If the type is part of precomputed outer enum hashes, the respective hash is used instead:
255    if let Some(hash) = outer_enum_hashes.resolve(id) {
256        return hash;
257    }
258
259    // Guard against recursive types, with a 2 step caching approach:
260    //    if the cache has an entry for the id, just return a hash derived from it.
261    //    if the type has not been seen yet, mark it with `CachedHash::Recursive` in the cache and proceed to `get_type_def_hash()`.
262    //        -> During the execution of get_type_def_hash() we might get into get_type_hash(id) again for the original id
263    //            -> in this case the `CachedHash::Recursive` provokes an early return.
264    //        -> Once we return from `get_type_def_hash()` we need to update the cache entry:
265    //            -> We set the cache value to `CachedHash::Hash(type_hash)`, where `type_hash` was returned from `get_type_def_hash()`
266    //            -> It makes sure, that different types end up with different cache values.
267    //
268    // Values in the cache can be thought of as a mapping like this:
269    // type_id ->  not contained           = We haven't seen the type yet.
270    //         -> `CachedHash::Recursive`  = We have seen the type but hash calculation for it hasn't finished yet.
271    //         -> `CachedHash::Hash(hash)` = Hash calculation for the type was completed.
272    if let Some(cached_hash) = cache.get(&id) {
273        return cached_hash.hash();
274    }
275    cache.insert(id, CachedHash::Recursive);
276    let ty = registry
277        .resolve(id)
278        .expect("Type ID provided by the metadata is registered; qed");
279    let type_hash = get_type_def_hash(registry, &ty.type_def, cache, outer_enum_hashes);
280    cache.insert(id, CachedHash::Hash(type_hash));
281    type_hash
282}

284/// Obtain the hash representation of a `frame_metadata::v15::ExtrinsicMetadata`.
285fn get_extrinsic_hash(
286    registry: &PortableRegistry,
287    extrinsic: &ExtrinsicMetadata,
288    outer_enum_hashes: &OuterEnumHashes,
289) -> Hash {
290    // Get the hashes of the extrinsic type.
291    let address_hash = get_type_hash(registry, extrinsic.address_ty, outer_enum_hashes);
292    // The `RuntimeCall` type is intentionally omitted and hashed by the outer enums instead.
293    let signature_hash = get_type_hash(registry, extrinsic.signature_ty, outer_enum_hashes);
294    let extra_hash = get_type_hash(registry, extrinsic.extra_ty, outer_enum_hashes);
295
296    let mut bytes = concat_and_hash4(
297        &address_hash,
298        &signature_hash,
299        &extra_hash,
300        &[extrinsic.version; 32],
301    );
302
303    for signed_extension in extrinsic.signed_extensions.iter() {
304        bytes = concat_and_hash4(
305            &bytes,
306            &hash(signed_extension.identifier.as_bytes()),
307            &get_type_hash(registry, signed_extension.extra_ty, outer_enum_hashes),
308            &get_type_hash(registry, signed_extension.additional_ty, outer_enum_hashes),
309        )
310    }
311
312    bytes
313}

315/// Get the hash corresponding to a single storage entry.
316fn get_storage_entry_hash(
317    registry: &PortableRegistry,
318    entry: &StorageEntryMetadata,
319    outer_enum_hashes: &OuterEnumHashes,
320) -> Hash {
321    let mut bytes = concat_and_hash3(
322        &hash(entry.name.as_bytes()),
323        // Cloning 'entry.modifier' should essentially be a copy.
324        &[entry.modifier as u8; HASH_LEN],
325        &hash(&entry.default),
326    );
327
328    match &entry.entry_type {
329        StorageEntryType::Plain(ty) => {
330            concat_and_hash2(&bytes, &get_type_hash(registry, *ty, outer_enum_hashes))
331        }
332        StorageEntryType::Map {
333            hashers,
334            key_ty,
335            value_ty,
336        } => {
337            for hasher in hashers {
338                // Cloning the hasher should essentially be a copy.
339                bytes = concat_and_hash2(&bytes, &[*hasher as u8; HASH_LEN]);
340            }
341            concat_and_hash3(
342                &bytes,
343                &get_type_hash(registry, *key_ty, outer_enum_hashes),
344                &get_type_hash(registry, *value_ty, outer_enum_hashes),
345            )
346        }
347    }
348}

350/// Get the hash corresponding to a single runtime API method.
351fn get_runtime_method_hash(
352    registry: &PortableRegistry,
353    trait_name: &str,
354    method_metadata: &RuntimeApiMethodMetadata,
355    outer_enum_hashes: &OuterEnumHashes,
356) -> Hash {
357    // The trait name is part of the runtime API call that is being
358    // generated for this method. Therefore the trait name is strongly
359    // connected to the method in the same way as a parameter is
360    // to the method.
361    let mut bytes = concat_and_hash2(
362        &hash(trait_name.as_bytes()),
363        &hash(method_metadata.name.as_bytes()),
364    );
365
366    for input in &method_metadata.inputs {
367        bytes = concat_and_hash3(
368            &bytes,
369            &hash(input.name.as_bytes()),
370            &get_type_hash(registry, input.ty, outer_enum_hashes),
371        );
372    }
373
374    bytes = concat_and_hash2(
375        &bytes,
376        &get_type_hash(registry, method_metadata.output_ty, outer_enum_hashes),
377    );
378
379    bytes
380}

382/// Obtain the hash of all of a runtime API trait, including all of its methods.
383pub fn get_runtime_trait_hash(
384    trait_metadata: RuntimeApiMetadata,
385    outer_enum_hashes: &OuterEnumHashes,
386) -> Hash {
387    let trait_name = &*trait_metadata.inner.name;
388    let method_bytes = trait_metadata
389        .methods()
390        .fold([0u8; HASH_LEN], |bytes, method_metadata| {
391            // We don't care what order the trait methods exist in, and want the hash to
392            // be identical regardless. For this, we can just XOR the hashes for each method
393            // together; we'll get the same output whichever order they are XOR'd together in,
394            // so long as each individual method is the same.
395            xor(
396                bytes,
397                get_runtime_method_hash(
398                    trait_metadata.types,
399                    trait_name,
400                    method_metadata,
401                    outer_enum_hashes,
402                ),
403            )
404        });
405
406    concat_and_hash2(&hash(trait_name.as_bytes()), &method_bytes)
407}

413    custom_metadata

423/// Obtain the hash of some custom value in the metadata including it's name/key.
424///
425/// If the `custom_value` has a type id that is not present in the metadata,
426/// only the name and bytes are used for hashing.
427pub fn get_custom_value_hash(
428    custom_value: &CustomValueMetadata,
429    outer_enum_hashes: &OuterEnumHashes,
430) -> Hash {
431    let name_hash = hash(custom_value.name.as_bytes());
432    if custom_value.types.resolve(custom_value.type_id()).is_none() {
433        hash(&name_hash)
434    } else {
435        concat_and_hash2(
436            &name_hash,
437            &get_type_hash(
438                custom_value.types,
439                custom_value.type_id(),
440                outer_enum_hashes,
441            ),
442        )
443    }
444}

490/// Obtain the hash representation of a `frame_metadata::v15::PalletMetadata`.
491pub fn get_pallet_hash(pallet: PalletMetadata, outer_enum_hashes: &OuterEnumHashes) -> Hash {
492    let registry = pallet.types;
493
494    let call_bytes = match pallet.call_ty_id() {
495        Some(calls) => get_type_hash(registry, calls, outer_enum_hashes),
496        None => [0u8; HASH_LEN],
497    };
498    let event_bytes = match pallet.event_ty_id() {
499        Some(event) => get_type_hash(registry, event, outer_enum_hashes),
500        None => [0u8; HASH_LEN],
501    };
502    let error_bytes = match pallet.error_ty_id() {
503        Some(error) => get_type_hash(registry, error, outer_enum_hashes),
504        None => [0u8; HASH_LEN],
505    };
506    let constant_bytes = pallet.constants().fold([0u8; HASH_LEN], |bytes, constant| {
507        // We don't care what order the constants occur in, so XOR together the combinations
508        // of (constantName, constantType) to make the order we see them irrelevant.
509        let constant_hash = concat_and_hash2(
510            &hash(constant.name.as_bytes()),
511            &get_type_hash(registry, constant.ty(), outer_enum_hashes),
512        );
513        xor(bytes, constant_hash)
514    });
515    let storage_bytes = match pallet.storage() {
516        Some(storage) => {
517            let prefix_hash = hash(storage.prefix().as_bytes());
518            let entries_hash = storage
519                .entries()
520                .iter()
521                .fold([0u8; HASH_LEN], |bytes, entry| {
522                    // We don't care what order the storage entries occur in, so XOR them together
523                    // to make the order irrelevant.
524                    xor(
525                        bytes,
526                        get_storage_entry_hash(registry, entry, outer_enum_hashes),
527                    )
528                });
529            concat_and_hash2(&prefix_hash, &entries_hash)
530        }
531        None => [0u8; HASH_LEN],
532    };
533
534    // Hash all of the above together:
535    concat_and_hash5(
536        &call_bytes,
537        &event_bytes,
538        &error_bytes,
539        &constant_bytes,
540        &storage_bytes,
541    )
542}

544/// Obtain a hash representation of our metadata or some part of it.
545/// This is obtained by calling [`crate::Metadata::hasher()`].
546pub struct MetadataHasher<'a> {
547    metadata: &'a Metadata,
548    specific_pallets: Option<Vec<&'a str>>,
549    specific_runtime_apis: Option<Vec<&'a str>>,
550    include_custom_values: bool,
551}
552
553impl<'a> MetadataHasher<'a> {
554    /// Create a new [`MetadataHasher`]
555    pub(crate) fn new(metadata: &'a Metadata) -> Self {
556        Self {
557            metadata,
558            specific_pallets: None,
559            specific_runtime_apis: None,
560            include_custom_values: true,
561        }
562    }
563
564    /// Only hash the provided pallets instead of hashing every pallet.
565    pub fn only_these_pallets<S: AsRef<str>>(&mut self, specific_pallets: &'a [S]) -> &mut Self {
566        self.specific_pallets = Some(specific_pallets.iter().map(|n| n.as_ref()).collect());
567        self
568    }
569
570    /// Only hash the provided runtime APIs instead of hashing every runtime API
571    pub fn only_these_runtime_apis<S: AsRef<str>>(
572        &mut self,
573        specific_runtime_apis: &'a [S],
574    ) -> &mut Self {
575        self.specific_runtime_apis =
576            Some(specific_runtime_apis.iter().map(|n| n.as_ref()).collect());
577        self
578    }
579
580    /// Do not hash the custom values
581    pub fn ignore_custom_values(&mut self) -> &mut Self {
582        self.include_custom_values = false;
583        self
584    }
585
586    /// Hash the given metadata.
587    pub fn hash(&self) -> Hash {
588        let metadata = self.metadata;
589
590        // Get the hashes of outer enums, considering only `specific_pallets` (if any are set).
591        // If any of the typed that represent outer enums are encountered later, hashes from `top_level_enum_hashes` can be substituted.
592        let outer_enum_hashes = OuterEnumHashes::new(
593            metadata,
594            self.specific_pallets.as_deref(),
595            self.specific_runtime_apis.as_deref(),
596        );
597
598        let pallet_hash = metadata.pallets().fold([0u8; HASH_LEN], |bytes, pallet| {
599            // If specific pallets are given, only include this pallet if it is in the specific pallets.
600            let should_hash = self
601                .specific_pallets
602                .as_ref()
603                .map(|specific_pallets| specific_pallets.contains(&pallet.name()))
604                .unwrap_or(true);
605            // We don't care what order the pallets are seen in, so XOR their
606            // hashes together to be order independent.
607            if should_hash {
608                xor(bytes, get_pallet_hash(pallet, &outer_enum_hashes))
609            } else {
610                bytes
611            }
612        });
613
614        let apis_hash = metadata
615            .runtime_api_traits()
616            .fold([0u8; HASH_LEN], |bytes, api| {
617                // If specific runtime APIs are given, only include this pallet if it is in the specific runtime APIs.
618                let should_hash = self
619                    .specific_runtime_apis
620                    .as_ref()
621                    .map(|specific_runtime_apis| specific_runtime_apis.contains(&api.name()))
622                    .unwrap_or(true);
623                // We don't care what order the runtime APIs are seen in, so XOR their
624                // hashes together to be order independent.
625                if should_hash {
626                    xor(bytes, get_runtime_trait_hash(api, &outer_enum_hashes))
627                } else {
628                    bytes
629                }
630            });
631
632        let extrinsic_hash =
633            get_extrinsic_hash(&metadata.types, &metadata.extrinsic, &outer_enum_hashes);
634        let runtime_hash =
635            get_type_hash(&metadata.types, metadata.runtime_ty(), &outer_enum_hashes);
636        let custom_values_hash = self
637            .include_custom_values
638            .then(|| get_custom_metadata_hash(&metadata.custom(), &outer_enum_hashes))
639            .unwrap_or_default();
640
641        concat_and_hash6(
642            &pallet_hash,
643            &apis_hash,
644            &extrinsic_hash,
645            &runtime_hash,
646            &outer_enum_hashes.combined_hash(),
647            &custom_values_hash,
648        )
649    }
