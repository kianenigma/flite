5use alloc::vec::Vec;
6use core::mem;
7use hashbrown::HashMap;
8
9/// A minimal ordered map to let one search for
10/// things by key or get the values in insert order.
11#[derive(Debug, Clone)]
12pub struct OrderedMap<K, V> {
13    values: Vec<V>,
14    map: HashMap<K, usize>,
15}
16
17impl<K, V> Default for OrderedMap<K, V> {
18    fn default() -> Self {
19        Self {
20            values: Default::default(),
21            map: Default::default(),
22        }
23    }
24}
25
26impl<K, V> OrderedMap<K, V>
27where
28    K: PartialEq + Eq + core::hash::Hash,
29{
30    /// Create a new, empty [`OrderedMap`].
31    pub fn new() -> Self {
32        Self::default()
33    }
34
35    /// Number of entries in the map.
36    #[allow(dead_code)]
37    pub fn len(&self) -> usize {
38        self.values.len()
39    }
40
41    /// Is the map empty.
42    #[allow(dead_code)]
43    pub fn is_empty(&self) -> bool {
44        self.values.is_empty()
45    }
46
47    /// Retain specific entries.
48    pub fn retain<F>(&mut self, mut f: F)
49    where
50        F: FnMut(&V) -> bool,
51    {
52        let values = mem::take(&mut self.values);
53        let map = mem::take(&mut self.map);
54
55        // Filter the values, storing a map from old to new positions:
56        let mut new_values = Vec::new();
57        let mut old_pos_to_new_pos = HashMap::new();
58        for (pos, value) in values.into_iter().enumerate().filter(|(_, v)| f(v)) {
59            old_pos_to_new_pos.insert(pos, new_values.len());
60            new_values.push(value);
61        }
62
63        // Update the values now we've filtered them:
64        self.values = new_values;
65
66        // Rebuild the map using the new positions:
67        self.map = map
68            .into_iter()
69            .filter_map(|(k, v)| old_pos_to_new_pos.get(&v).map(|v2| (k, *v2)))
70            .collect();
71    }
72
73    /// Push/insert an item to the end of the map.
74    pub fn push_insert(&mut self, key: K, value: V) {
75        let idx = self.values.len();
76        self.values.push(value);
77        self.map.insert(key, idx);
78    }
79
80    /// Get an item by its key.
81    pub fn get_by_key<Q>(&self, key: &Q) -> Option<&V>
82    where
83        K: alloc::borrow::Borrow<Q>,
84        Q: core::hash::Hash + Eq + ?Sized,
85    {
86        self.map.get(key).and_then(|&v| self.values.get(v))
87    }
88
89    /// Get an item by its index.
90    pub fn get_by_index(&self, i: usize) -> Option<&V> {
91        self.values.get(i)
92    }
93
94    /// Access the underlying values.
95    pub fn values(&self) -> &[V] {
96        &self.values
97    }
98
99    /// Mutable access to the underlying values.
100    pub fn values_mut(&mut self) -> &mut [V] {
101        &mut self.values
102    }
103
104    /// Return the underlying values.
105    pub fn into_values(self) -> Vec<V> {
106        self.values
107    }
108}
109
110impl<K, V> FromIterator<(K, V)> for OrderedMap<K, V>
111where
112    K: PartialEq + Eq + core::hash::Hash,
113{
114    fn from_iter<T: IntoIterator<Item = (K, V)>>(iter: T) -> Self {
115        let mut map = OrderedMap::new();
116        for (k, v) in iter {
117            map.push_insert(k, v)
118        }
119        map
120    }
121}

123#[cfg(test)]
124mod test {
125    use super::*;
126
127    #[test]
128    fn retain() {
129        let mut m = OrderedMap::from_iter([(1, 'a'), (2, 'b'), (3, 'c')]);
130
131        m.retain(|v| *v != 'b');
132
133        assert_eq!(m.get_by_key(&1), Some(&'a'));
134        assert_eq!(m.get_by_key(&2), None);
135        assert_eq!(m.get_by_key(&3), Some(&'c'));
136
137        assert_eq!(m.values(), &['a', 'c'])
138    }
139}
