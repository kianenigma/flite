5//! This module contains a trait which controls the parameters that must
6//! be provided in order to successfully construct an extrinsic.
7//! [`crate::config::DefaultExtrinsicParams`] provides a general-purpose
8//! implementation of this that will work in many cases.
9
10use super::refine_params::RefineParams;
11use crate::{client::ClientState, error::ExtrinsicParamsError, Config};
12use alloc::vec::Vec;
13
14/// This trait allows you to configure the "signed extra" and
15/// "additional" parameters that are a part of the transaction payload
16/// or the signer payload respectively.
17pub trait ExtrinsicParams<T: Config>: ExtrinsicParamsEncoder + Sized + Send + 'static {
18    /// These parameters can be provided to the constructor along with
19    /// some default parameters that `subxt` understands, in order to
20    /// help construct your [`ExtrinsicParams`] object.
21    type Params: RefineParams<T>;
22
23    /// Construct a new instance of our [`ExtrinsicParams`].
24    fn new(client: &ClientState<T>, params: Self::Params) -> Result<Self, ExtrinsicParamsError>;
25}
26
27/// This trait is expected to be implemented for any [`ExtrinsicParams`], and
28/// defines how to encode the "additional" and "extra" params. Both functions
29/// are optional and will encode nothing by default.
30pub trait ExtrinsicParamsEncoder: 'static {
31    /// This is expected to SCALE encode the "signed extra" parameters
32    /// to some buffer that has been provided. These are the parameters
33    /// which are sent along with the transaction, as well as taken into
34    /// account when signing the transaction.
35    fn encode_extra_to(&self, _v: &mut Vec<u8>) {}
36
37    /// This is expected to SCALE encode the "additional" parameters
38    /// to some buffer that has been provided. These parameters are _not_
39    /// sent along with the transaction, but are taken into account when
40    /// signing it, meaning the client and node must agree on their values.
41    fn encode_additional_to(&self, _v: &mut Vec<u8>) {}
42}
