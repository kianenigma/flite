5//! The "default" Substrate/Polkadot Address type. This is used in codegen, as well as signing related bits.
6//! This doesn't contain much functionality itself, but is easy to convert to/from an `sp_runtime::MultiAddress`
7//! for instance, to gain functionality without forcing a dependency on Substrate crates here.
8
9use alloc::vec::Vec;
10use codec::{Decode, Encode};
11
12/// A multi-format address wrapper for on-chain accounts. This is a simplified version of Substrate's
13/// `sp_runtime::MultiAddress`. To obtain more functionality, convert this into that type (this conversion
14/// functionality is provided via `From` impls if the `substrate-compat` feature is enabled).
15#[derive(
16    Clone,
17    Eq,
18    PartialEq,
19    Ord,
20    PartialOrd,
21    Encode,
22    Decode,
23    Debug,
24    scale_encode::EncodeAsType,
25    scale_decode::DecodeAsType,
26    scale_info::TypeInfo,
27)]
28pub enum MultiAddress<AccountId, AccountIndex> {
29    /// It's an account ID (pubkey).
30    Id(AccountId),
31    /// It's an account index.
32    Index(#[codec(compact)] AccountIndex),
33    /// It's some arbitrary raw bytes.
34    Raw(Vec<u8>),
35    /// It's a 32 byte representation.
36    Address32([u8; 32]),
37    /// Its a 20 byte representation.
38    Address20([u8; 20]),
39}
40
41impl<AccountId, AccountIndex> From<AccountId> for MultiAddress<AccountId, AccountIndex> {
42    fn from(a: AccountId) -> Self {
43        Self::Id(a)
44    }
45}
46
47// Improve compat with the substrate version if we're using those crates:
48#[cfg(feature = "substrate-compat")]
49mod substrate_impls {
50    use super::{super::AccountId32, *};
51    use polkadot_sdk::sp_runtime;
52
53    impl<N> From<sp_runtime::AccountId32> for MultiAddress<AccountId32, N> {
54        fn from(value: sp_runtime::AccountId32) -> Self {
55            let val: AccountId32 = value.into();
56            val.into()
57        }
58    }
59
60    impl<Id, N> From<sp_runtime::MultiAddress<Id, N>> for MultiAddress<AccountId32, N>
61    where
62        Id: Into<AccountId32>,
63    {
64        fn from(value: sp_runtime::MultiAddress<Id, N>) -> Self {
65            match value {
66                sp_runtime::MultiAddress::Id(v) => Self::Id(v.into()),
67                sp_runtime::MultiAddress::Index(v) => Self::Index(v),
68                sp_runtime::MultiAddress::Raw(v) => Self::Raw(v),
69                sp_runtime::MultiAddress::Address32(v) => Self::Address32(v),
70                sp_runtime::MultiAddress::Address20(v) => Self::Address20(v),
71            }
72        }
73    }
74}
