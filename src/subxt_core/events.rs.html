50/// Create a new [`Events`] instance from the given bytes.

52/// This is a shortcut for [`Events::decode_from`].
53pub fn decode_from<T: Config>(event_bytes: Vec<u8>, metadata: Metadata) -> Events<T> {
54    Events::decode_from(event_bytes, metadata)

57/// Trait to uniquely identify the events's identity from the runtime metadata.
58///
59/// Generated API structures that represent an event implement this trait.
60///
61/// The trait is utilized to decode emitted events from a block, via obtaining the
62/// form of the `Event` from the metadata.
63pub trait StaticEvent: DecodeAsFields {
64    /// Pallet name.
65    const PALLET: &'static str;
66    /// Event name.
67    const EVENT: &'static str;
68
69    /// Returns true if the given pallet and event names match this event.
70    fn is_event(pallet: &str, event: &str) -> bool {
71        Self::PALLET == pallet && Self::EVENT == event
72    }
73}

75/// A collection of events obtained from a block, bundled with the necessary
76/// information needed to decode and iterate over them.
77#[derive_where(Clone)]
78pub struct Events<T: Config> {
79    metadata: Metadata,
80    // Note; raw event bytes are prefixed with a Compact<u32> containing
81    // the number of events to be decoded. The start_idx reflects that, so
82    // that we can skip over those bytes when decoding them
83    event_bytes: Arc<[u8]>,
84    start_idx: usize,
85    num_events: u32,
86    marker: core::marker::PhantomData<T>,
87}
88
89// Ignore the Metadata when debug-logging events; it's big and distracting.
90impl<T: Config> core::fmt::Debug for Events<T> {
91    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
92        f.debug_struct("Events")
93            .field("event_bytes", &self.event_bytes)
94            .field("start_idx", &self.start_idx)
95            .field("num_events", &self.num_events)
96            .finish()
97    }
98}

100impl<T: Config> Events<T> {
101    /// Create a new [`Events`] instance from the given bytes.
102    pub fn decode_from(event_bytes: Vec<u8>, metadata: Metadata) -> Self {
103        // event_bytes is a SCALE encoded vector of events. So, pluck the
104        // compact encoded length from the front, leaving the remaining bytes
105        // for our iterating to decode.
106        //
107        // Note: if we get no bytes back, avoid an error reading vec length
108        // and default to 0 events.
109        let cursor = &mut &*event_bytes;
110        let num_events = <Compact<u32>>::decode(cursor).unwrap_or(Compact(0)).0;
111
112        // Start decoding after the compact encoded bytes.
113        let start_idx = event_bytes.len() - cursor.len();
114
115        Self {
116            metadata,
117            event_bytes: event_bytes.into(),
118            start_idx,
119            num_events,
120            marker: core::marker::PhantomData,
121        }
122    }
123
124    /// The number of events.
125    pub fn len(&self) -> u32 {
126        self.num_events
127    }
128
129    /// Are there no events in this block?
130    // Note: mainly here to satisfy clippy.
131    pub fn is_empty(&self) -> bool {
132        self.num_events == 0
133    }
134
135    /// Return the bytes representing all of the events.
136    pub fn bytes(&self) -> &[u8] {
137        &self.event_bytes
138    }
139
140    /// Iterate over all of the events, using metadata to dynamically
141    /// decode them as we go, and returning the raw bytes and other associated
142    /// details. If an error occurs, all subsequent iterations return `None`.
143    // Dev note: The returned iterator is 'static + Send so that we can box it up and make
144    // use of it with our `FilterEvents` stuff.
145    pub fn iter(
146        &self,
147    ) -> impl Iterator<Item = Result<EventDetails<T>, Error>> + Send + Sync + 'static {
148        // The event bytes ignoring the compact encoded length on the front:
149        let event_bytes = self.event_bytes.clone();
150        let metadata = self.metadata.clone();
151        let num_events = self.num_events;
152
153        let mut pos = self.start_idx;
154        let mut index = 0;
155        core::iter::from_fn(move || {
156            if event_bytes.len() <= pos || num_events == index {
157                None
158            } else {
159                match EventDetails::decode_from(metadata.clone(), event_bytes.clone(), pos, index) {
160                    Ok(event_details) => {
161                        // Skip over decoded bytes in next iteration:
162                        pos += event_details.bytes().len();
163                        // Increment the index:
164                        index += 1;
165                        // Return the event details:
166                        Some(Ok(event_details))
167                    }
168                    Err(e) => {
169                        // By setting the position to the "end" of the event bytes,
170                        // the cursor len will become 0 and the iterator will return `None`
171                        // from now on:
172                        pos = event_bytes.len();
173                        Some(Err(e))
174                    }
175                }
176            }
177        })
178    }
179
180    /// Iterate through the events using metadata to dynamically decode and skip
181    /// them, and return only those which should decode to the provided `Ev` type.
182    /// If an error occurs, all subsequent iterations return `None`.
183    pub fn find<Ev: StaticEvent>(&self) -> impl Iterator<Item = Result<Ev, Error>> + '_ {
184        self.iter().filter_map(|ev| {
185            ev.and_then(|ev| ev.as_event::<Ev>().map_err(Into::into))
186                .transpose()
187        })
188    }
189
190    /// Iterate through the events using metadata to dynamically decode and skip
191    /// them, and return the first event found which decodes to the provided `Ev` type.
192    pub fn find_first<Ev: StaticEvent>(&self) -> Result<Option<Ev>, Error> {
193        self.find::<Ev>().next().transpose()
194    }
195
196    /// Iterate through the events using metadata to dynamically decode and skip
197    /// them, and return the last event found which decodes to the provided `Ev` type.
198    pub fn find_last<Ev: StaticEvent>(&self) -> Result<Option<Ev>, Error> {
199        self.find::<Ev>().last().transpose()
200    }
201
202    /// Find an event that decodes to the type provided. Returns true if it was found.
203    pub fn has<Ev: StaticEvent>(&self) -> Result<bool, Error> {
204        Ok(self.find::<Ev>().next().transpose()?.is_some())
205    }
206}

210pub enum Phase {

219/// The event details.
220#[derive(Debug, Clone)]
221pub struct EventDetails<T: Config> {
222    phase: Phase,
223    /// The index of the event in the list of events in a given block.
224    index: u32,
225    all_bytes: Arc<[u8]>,
226    // start of the bytes (phase, pallet/variant index and then fields and then topic to follow).
227    start_idx: usize,
228    // start of the event (ie pallet/variant index and then the fields and topic after).
229    event_start_idx: usize,
230    // start of the fields (ie after phase and pallet/variant index).
231    event_fields_start_idx: usize,
232    // end of the fields.
233    event_fields_end_idx: usize,
234    // end of everything (fields + topics)
235    end_idx: usize,
236    metadata: Metadata,
237    topics: Vec<T::Hash>,
238}
239
240impl<T: Config> EventDetails<T> {
241    /// Attempt to dynamically decode a single event from our events input.
242    fn decode_from(
243        metadata: Metadata,
244        all_bytes: Arc<[u8]>,
245        start_idx: usize,
246        index: u32,
247    ) -> Result<EventDetails<T>, Error> {
248        let input = &mut &all_bytes[start_idx..];
249
250        let phase = Phase::decode(input)?;
251
252        let event_start_idx = all_bytes.len() - input.len();
253
254        let pallet_index = u8::decode(input)?;
255        let variant_index = u8::decode(input)?;
256
257        let event_fields_start_idx = all_bytes.len() - input.len();
258
259        // Get metadata for the event:
260        let event_pallet = metadata.pallet_by_index_err(pallet_index)?;
261        let event_variant = event_pallet
262            .event_variant_by_index(variant_index)
263            .ok_or(MetadataError::VariantIndexNotFound(variant_index))?;
264        tracing::debug!(
265            "Decoding Event '{}::{}'",
266            event_pallet.name(),
267            &event_variant.name
268        );
269
270        // Skip over the bytes belonging to this event.
271        for field_metadata in &event_variant.fields {
272            // Skip over the bytes for this field:
273            scale_decode::visitor::decode_with_visitor(
274                input,
275                field_metadata.ty.id,
276                metadata.types(),
277                scale_decode::visitor::IgnoreVisitor::new(),
278            )
279            .map_err(scale_decode::Error::from)?;
280        }
281
282        // the end of the field bytes.
283        let event_fields_end_idx = all_bytes.len() - input.len();
284
285        // topics come after the event data in EventRecord.
286        let topics = Vec::<T::Hash>::decode(input)?;
287
288        // what bytes did we skip over in total, including topics.
289        let end_idx = all_bytes.len() - input.len();
290
291        Ok(EventDetails {
292            phase,
293            index,
294            start_idx,
295            event_start_idx,
296            event_fields_start_idx,
297            event_fields_end_idx,
298            end_idx,
299            all_bytes,
300            metadata,
301            topics,
302        })
303    }

305    /// When was the event produced?
306    pub fn phase(&self) -> Phase {
307        self.phase
308    }
309
310    /// What index is this event in the stored events for this block.
311    pub fn index(&self) -> u32 {
312        self.index
313    }
314
315    /// The index of the pallet that the event originated from.
316    pub fn pallet_index(&self) -> u8 {
317        // Note: never panics; we expect these bytes to exist
318        // in order that the EventDetails could be created.
319        self.all_bytes[self.event_fields_start_idx - 2]
320    }
321
322    /// The index of the event variant that the event originated from.
323    pub fn variant_index(&self) -> u8 {
324        // Note: never panics; we expect these bytes to exist
325        // in order that the EventDetails could be created.
326        self.all_bytes[self.event_fields_start_idx - 1]
327    }
328
329    /// The name of the pallet from whence the Event originated.
330    pub fn pallet_name(&self) -> &str {
331        self.event_metadata().pallet.name()
332    }
333
334    /// The name of the event (ie the name of the variant that it corresponds to).
335    pub fn variant_name(&self) -> &str {
336        &self.event_metadata().variant.name
337    }
338
339    /// Fetch details from the metadata for this event.
340    pub fn event_metadata(&self) -> EventMetadataDetails {
341        let pallet = self
342            .metadata
343            .pallet_by_index(self.pallet_index())
344            .expect("event pallet to be found; we did this already during decoding");
345        let variant = pallet
346            .event_variant_by_index(self.variant_index())
347            .expect("event variant to be found; we did this already during decoding");
348
349        EventMetadataDetails { pallet, variant }
350    }
351
352    /// Return _all_ of the bytes representing this event, which include, in order:
353    /// - The phase.
354    /// - Pallet and event index.
355    /// - Event fields.
356    /// - Event Topics.
357    pub fn bytes(&self) -> &[u8] {
358        &self.all_bytes[self.start_idx..self.end_idx]
359    }
360
361    /// Return the bytes representing the fields stored in this event.
362    pub fn field_bytes(&self) -> &[u8] {
363        &self.all_bytes[self.event_fields_start_idx..self.event_fields_end_idx]
364    }
365
366    /// Decode and provide the event fields back in the form of a [`scale_value::Composite`]
367    /// type which represents the named or unnamed fields that were present in the event.
368    pub fn field_values(&self) -> Result<scale_value::Composite<u32>, Error> {
369        let bytes = &mut self.field_bytes();
370        let event_metadata = self.event_metadata();
371
372        let mut fields = event_metadata
373            .variant
374            .fields
375            .iter()
376            .map(|f| scale_decode::Field::new(f.ty.id, f.name.as_deref()));
377
378        let decoded =
379            scale_value::scale::decode_as_fields(bytes, &mut fields, self.metadata.types())?;
380
381        Ok(decoded)
382    }
383
384    /// Attempt to decode these [`EventDetails`] into a type representing the event fields.
385    /// Such types are exposed in the codegen as `pallet_name::events::EventName` types.
386    pub fn as_event<E: StaticEvent>(&self) -> Result<Option<E>, Error> {
387        let ev_metadata = self.event_metadata();
388        if ev_metadata.pallet.name() == E::PALLET && ev_metadata.variant.name == E::EVENT {
389            let mut fields = ev_metadata
390                .variant
391                .fields
392                .iter()
393                .map(|f| scale_decode::Field::new(f.ty.id, f.name.as_deref()));
394            let decoded =
395                E::decode_as_fields(&mut self.field_bytes(), &mut fields, self.metadata.types())?;
396            Ok(Some(decoded))
397        } else {
398            Ok(None)
399        }
400    }
401
402    /// Attempt to decode these [`EventDetails`] into a root event type (which includes
403    /// the pallet and event enum variants as well as the event fields). A compatible
404    /// type for this is exposed via static codegen as a root level `Event` type.
405    pub fn as_root_event<E: DecodeAsType>(&self) -> Result<E, Error> {
406        let bytes = &self.all_bytes[self.event_start_idx..self.event_fields_end_idx];
407
408        let decoded = E::decode_as_type(
409            &mut &bytes[..],
410            self.metadata.outer_enums().event_enum_ty(),
411            self.metadata.types(),
412        )?;
413
414        Ok(decoded)
415    }
416
417    /// Return the topics associated with this event.
418    pub fn topics(&self) -> &[T::Hash] {
419        &self.topics
420    }
421}

488    /// Build fake metadata consisting of a single pallet that knows
489    /// about the event type provided.
490    pub fn metadata<E: TypeInfo + 'static>() -> Metadata {
491        // Extrinsic needs to contain at least the generic type parameter "Call"
492        // for the metadata to be valid.
493        // The "Call" type from the metadata is used to decode extrinsics.
494        // In reality, the extrinsic type has "Call", "Address", "Extra", "Signature" generic types.
495        #[allow(unused)]
496        #[derive(TypeInfo)]
497        struct ExtrinsicType<Call> {
498            call: Call,
499        }
500        // Because this type is used to decode extrinsics, we expect this to be a TypeDefVariant.
501        // Each pallet must contain one single variant.
502        #[allow(unused)]
503        #[derive(TypeInfo)]
504        enum RuntimeCall {
505            PalletName(Pallet),
506        }
507        // The calls of the pallet.
508        #[allow(unused)]
509        #[derive(TypeInfo)]
510        enum Pallet {
511            #[allow(unused)]
512            SomeCall,
513        }
514
515        let pallets = vec![PalletMetadata {
516            name: "Test",
517            storage: None,
518            calls: None,
519            event: Some(PalletEventMetadata {
520                ty: meta_type::<E>(),
521            }),
522            constants: vec![],
523            error: None,
524            index: 0,
525            docs: vec![],
526        }];
527
528        let extrinsic = ExtrinsicMetadata {
529            version: 0,
530            signed_extensions: vec![],
531            address_ty: meta_type::<()>(),
532            call_ty: meta_type::<RuntimeCall>(),
533            signature_ty: meta_type::<()>(),
534            extra_ty: meta_type::<()>(),
535        };
536
537        let meta = RuntimeMetadataV15::new(
538            pallets,
539            extrinsic,
540            meta_type::<()>(),
541            vec![],
542            OuterEnums {
543                call_enum_ty: meta_type::<()>(),
544                event_enum_ty: meta_type::<AllEvents<E>>(),
545                error_enum_ty: meta_type::<()>(),
546            },
547            CustomMetadata {
548                map: Default::default(),
549            },
550        );
551        let runtime_metadata: RuntimeMetadataPrefixed = meta.into();
552        let metadata: subxt_metadata::Metadata = runtime_metadata.try_into().unwrap();
553
554        Metadata::from(metadata)
555    }

557    /// Build an `Events` object for test purposes, based on the details provided,
558    /// and with a default block hash.
559    pub fn events<E: Decode + Encode>(
560        metadata: Metadata,
561        event_records: Vec<EventRecord<E>>,
562    ) -> Events<SubstrateConfig> {
563        let num_events = event_records.len() as u32;
564        let mut event_bytes = Vec::new();
565        for ev in event_records {
566            ev.encode_to(&mut event_bytes);
567        }
568        events_raw(metadata, event_bytes, num_events)
569    }

571    /// Much like [`events`], but takes pre-encoded events and event count, so that we can
572    /// mess with the bytes in tests if we need to.
573    pub fn events_raw(
574        metadata: Metadata,
575        event_bytes: Vec<u8>,
576        num_events: u32,
577    ) -> Events<SubstrateConfig> {
578        // Prepend compact encoded length to event bytes:
579        let mut all_event_bytes = Compact(num_events).encode();
580        all_event_bytes.extend(event_bytes);
581        Events::decode_from(all_event_bytes, metadata)
582    }
