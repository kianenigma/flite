5//! Construct and sign transactions.
6//!
7//! # Example
8//!
9//! ```rust
10//! use subxt_signer::sr25519::dev;
11//! use subxt_macro::subxt;
12//! use subxt_core::config::PolkadotConfig;
13//! use subxt_core::config::DefaultExtrinsicParamsBuilder as Params;
14//! use subxt_core::tx;
15//! use subxt_core::utils::H256;
16//! use subxt_core::metadata;
17//!
18//! // If we generate types without `subxt`, we need to point to `::subxt_core`:
19//! #[subxt(
20//!     crate = "::subxt_core",
21//!     runtime_metadata_path = "../artifacts/polkadot_metadata_small.scale",
22//! )]
23//! pub mod polkadot {}
24//!
25//! // Gather some other information about the chain that we'll need to construct valid extrinsics:
26//! let state = tx::ClientState::<PolkadotConfig> {
27//!     metadata: {
28//!         let metadata_bytes = include_bytes!("../../../artifacts/polkadot_metadata_small.scale");
29//!         metadata::decode_from(&metadata_bytes[..]).unwrap()
30//!     },
31//!     genesis_hash: {
32//!         let h = "91b171bb158e2d3848fa23a9f1c25182fb8e20313b2c1eb49219da7a70ce90c3";
33//!         let bytes = hex::decode(h).unwrap();
34//!         H256::from_slice(&bytes)
35//!     },
36//!     runtime_version: tx::RuntimeVersion {
37//!         spec_version: 9370,
38//!         transaction_version: 20,
39//!     }
40//! };
41//!
42//! // Now we can build a balance transfer extrinsic.
43//! let dest = dev::bob().public_key().into();
44//! let call = polkadot::tx().balances().transfer_allow_death(dest, 10_000);
45//! let params = Params::new().tip(1_000).nonce(0).build();
46//!
47//! // We can validate that this lines up with the given metadata:
48//! tx::validate(&call, &state.metadata).unwrap();
49//!
50//! // We can build a signed transaction:
51//! let signed_call = tx::create_signed(&call, &state, &dev::alice(), params).unwrap();
52//!
53//! // And log it:
54//! println!("Tx: 0x{}", hex::encode(signed_call.encoded()));
55//! ```

74/// Run the validation logic against some extrinsic you'd like to submit. Returns `Ok(())`
75/// if the call is valid (or if it's not possible to check since the call has no validation hash).
76/// Return an error if the call was not valid or something went wrong trying to validate it (ie
77/// the pallet or call in question do not exist at all).
78pub fn validate<Call: Payload>(call: &Call, metadata: &Metadata) -> Result<(), Error> {
79    if let Some(details) = call.validation_details() {
80        let expected_hash = metadata
81            .pallet_by_name_err(details.pallet_name)?
82            .call_hash(details.call_name)
83            .ok_or_else(|| MetadataError::CallNameNotFound(details.call_name.to_owned()))?;
84
85        if details.hash != expected_hash {
86            return Err(MetadataError::IncompatibleCodegen.into());
87        }
88    }
89    Ok(())
90}

92/// Return the SCALE encoded bytes representing the call data of the transaction.
93pub fn call_data<Call: Payload>(call: &Call, metadata: &Metadata) -> Result<Vec<u8>, Error> {
94    let mut bytes = Vec::new();
95    call.encode_call_data_to(metadata, &mut bytes)?;
96    Ok(bytes)
97}

99/// Creates an unsigned extrinsic without submitting it.
100pub fn create_unsigned<T: Config, Call: Payload>(
101    call: &Call,
102    metadata: &Metadata,
103) -> Result<Transaction<T>, Error> {
104    // 1. Validate this call against the current node metadata if the call comes
105    // with a hash allowing us to do so.
106    validate(call, metadata)?;
107
108    // 2. Encode extrinsic
109    let extrinsic = {
110        let mut encoded_inner = Vec::new();
111        // transaction protocol version (4) (is not signed, so no 1 bit at the front).
112        4u8.encode_to(&mut encoded_inner);
113        // encode call data after this byte.
114        call.encode_call_data_to(metadata, &mut encoded_inner)?;
115        // now, prefix byte length:
116        let len = Compact(
117            u32::try_from(encoded_inner.len()).expect("extrinsic size expected to be <4GB"),
118        );
119        let mut encoded = Vec::new();
120        len.encode_to(&mut encoded);
121        encoded.extend(encoded_inner);
122        encoded
123    };
124
125    // Wrap in Encoded to ensure that any more "encode" calls leave it in the right state.
126    Ok(Transaction::from_bytes(extrinsic))
127}

129/// Create a partial extrinsic.
130///
131/// Note: if not provided, the default account nonce will be set to 0 and the default mortality will be _immortal_.
132/// This is because this method runs offline, and so is unable to fetch the data needed for more appropriate values.
133pub fn create_partial_signed<T: Config, Call: Payload>(
134    call: &Call,
135    client_state: &ClientState<T>,
136    params: <T::ExtrinsicParams as ExtrinsicParams<T>>::Params,
137) -> Result<PartialTransaction<T>, Error> {
138    // 1. Validate this call against the current node metadata if the call comes
139    // with a hash allowing us to do so.
140    validate(call, &client_state.metadata)?;
141
142    // 2. SCALE encode call data to bytes (pallet u8, call u8, call params).
143    let call_data = call_data(call, &client_state.metadata)?;
144
145    // 3. Construct our custom additional/extra params.
146    let additional_and_extra_params =
147        <T::ExtrinsicParams as ExtrinsicParams<T>>::new(client_state, params)?;
148
149    // Return these details, ready to construct a signed extrinsic from.
150    Ok(PartialTransaction {
151        call_data,
152        additional_and_extra_params,
153    })
154}

156/// Creates a signed extrinsic without submitting it.
157///
158/// Note: if not provided, the default account nonce will be set to 0 and the default mortality will be _immortal_.
159/// This is because this method runs offline, and so is unable to fetch the data needed for more appropriate values.
160pub fn create_signed<T, Call, Signer>(
161    call: &Call,
162    client_state: &ClientState<T>,
163    signer: &Signer,
164    params: <T::ExtrinsicParams as ExtrinsicParams<T>>::Params,
165) -> Result<Transaction<T>, Error>
166where
167    T: Config,
168    Call: Payload,
169    Signer: SignerT<T>,
170{
171    // 1. Validate this call against the current node metadata if the call comes
172    // with a hash allowing us to do so.
173    validate(call, &client_state.metadata)?;
174
175    // 2. Gather the "additional" and "extra" params along with the encoded call data,
176    //    ready to be signed.
177    let partial_signed = create_partial_signed(call, client_state, params)?;
178
179    // 3. Sign and construct an extrinsic from these details.
180    Ok(partial_signed.sign(signer))
181}

183/// This represents a partially constructed transaction that needs signing before it is ready
184/// to submit. Use [`PartialTransaction::signer_payload()`] to return the payload that needs signing,
185/// [`PartialTransaction::sign()`] to sign the transaction using a [`SignerT`] impl, or
186/// [`PartialTransaction::sign_with_address_and_signature()`] to apply an existing signature and address
187/// to the transaction.
188pub struct PartialTransaction<T: Config> {
189    call_data: Vec<u8>,
190    additional_and_extra_params: T::ExtrinsicParams,
191}
192
193impl<T: Config> PartialTransaction<T> {
194    // Obtain bytes representing the signer payload and run call some function
195    // with them. This can avoid an allocation in some cases when compared to
196    // [`PartialExtrinsic::signer_payload()`].
197    fn with_signer_payload<F, R>(&self, f: F) -> R
198    where
199        F: for<'a> FnOnce(Cow<'a, [u8]>) -> R,
200    {
201        let mut bytes = self.call_data.clone();
202        self.additional_and_extra_params.encode_extra_to(&mut bytes);
203        self.additional_and_extra_params
204            .encode_additional_to(&mut bytes);
205        if bytes.len() > 256 {
206            f(Cow::Borrowed(blake2_256(&bytes).as_ref()))
207        } else {
208            f(Cow::Owned(bytes))
209        }
210    }
211
212    /// Return the signer payload for this extrinsic. These are the bytes that must
213    /// be signed in order to produce a valid signature for the extrinsic.
214    pub fn signer_payload(&self) -> Vec<u8> {
215        self.with_signer_payload(|bytes| bytes.to_vec())
216    }
217
218    /// Return the bytes representing the call data for this partially constructed
219    /// extrinsic.
220    pub fn call_data(&self) -> &[u8] {
221        &self.call_data
222    }
223
224    /// Convert this [`PartialTransaction`] into a [`Transaction`], ready to submit.
225    /// The provided `signer` is responsible for providing the "from" address for the transaction,
226    /// as well as providing a signature to attach to it.
227    pub fn sign<Signer>(&self, signer: &Signer) -> Transaction<T>
228    where
229        Signer: SignerT<T>,
230    {
231        // Given our signer, we can sign the payload representing this extrinsic.
232        let signature = self.with_signer_payload(|bytes| signer.sign(&bytes));
233        // Now, use the signature and "from" address to build the extrinsic.
234        self.sign_with_address_and_signature(&signer.address(), &signature)
235    }
236
237    /// Convert this [`PartialTransaction`] into a [`Transaction`], ready to submit.
238    /// An address, and something representing a signature that can be SCALE encoded, are both
239    /// needed in order to construct it. If you have a `Signer` to hand, you can use
240    /// [`PartialTransaction::sign()`] instead.
241    pub fn sign_with_address_and_signature(
242        &self,
243        address: &T::Address,
244        signature: &T::Signature,
245    ) -> Transaction<T> {
246        // Encode the extrinsic (into the format expected by protocol version 4)
247        let extrinsic = {
248            let mut encoded_inner = Vec::new();
249            // "is signed" + transaction protocol version (4)
250            (0b10000000 + 4u8).encode_to(&mut encoded_inner);
251            // from address for signature
252            address.encode_to(&mut encoded_inner);
253            // the signature
254            signature.encode_to(&mut encoded_inner);
255            // attach custom extra params
256            self.additional_and_extra_params
257                .encode_extra_to(&mut encoded_inner);
258            // and now, call data (remembering that it's been encoded already and just needs appending)
259            encoded_inner.extend(&self.call_data);
260            // now, prefix byte length:
261            let len = Compact(
262                u32::try_from(encoded_inner.len()).expect("extrinsic size expected to be <4GB"),
263            );
264            let mut encoded = Vec::new();
265            len.encode_to(&mut encoded);
266            encoded.extend(encoded_inner);
267            encoded
268        };
269
270        // Return an extrinsic ready to be submitted.
271        Transaction::from_bytes(extrinsic)
272    }
273}

275/// This represents a signed transaction that's ready to be submitted.
276/// Use [`Transaction::encoded()`] or [`Transaction::into_encoded()`] to
277/// get the bytes for it, or [`Transaction::hash()`] to get the hash.
278pub struct Transaction<T> {
279    encoded: Encoded,
280    marker: core::marker::PhantomData<T>,
281}
282
283impl<T: Config> Transaction<T> {
284    /// Create a [`Transaction`] from some already-signed and prepared
285    /// extrinsic bytes,
286    pub fn from_bytes(tx_bytes: Vec<u8>) -> Self {
287        Self {
288            encoded: Encoded(tx_bytes),
289            marker: core::marker::PhantomData,
290        }
291    }
292
293    /// Calculate and return the hash of the extrinsic, based on the configured hasher.
294    pub fn hash(&self) -> T::Hash {
295        T::Hasher::hash_of(&self.encoded)
296    }
297
298    /// Returns the SCALE encoded extrinsic bytes.
299    pub fn encoded(&self) -> &[u8] {
300        &self.encoded.0
301    }
302
303    /// Consumes this [`Transaction`] and returns the SCALE encoded
304    /// extrinsic bytes.
305    pub fn into_encoded(self) -> Vec<u8> {
306        self.encoded.0
307    }
