5use codec::{Decode, Encode};
6use polkadot_sdk::sp_crypto_hashing;
7
8// This code is taken from sp_core::crypto::DeriveJunction. The logic should be identical,
9// though the API is tweaked a touch.
10
11/// The length of the junction identifier. Note that this is also referred to as the
12/// `CHAIN_CODE_LENGTH` in the context of Schnorrkel.
13pub const JUNCTION_ID_LEN: usize = 32;
14
15/// A since derivation junction description. It is the single parameter used when creating
16/// a new secret key from an existing secret key and, in the case of `SoftRaw` and `SoftIndex`
17/// a new public key from an existing public key.
18#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug, Encode, Decode)]
19pub enum DeriveJunction {
20    /// Soft (vanilla) derivation. Public keys have a correspondent derivation.
21    Soft([u8; JUNCTION_ID_LEN]),
22    /// Hard ("hardened") derivation. Public keys do not have a correspondent derivation.
23    Hard([u8; JUNCTION_ID_LEN]),
24}
25
26impl DeriveJunction {
27    /// Consume self to return a soft derive junction with the same chain code.
28    pub fn soften(self) -> Self {
29        DeriveJunction::Soft(self.into_inner())
30    }
31
32    /// Consume self to return a hard derive junction with the same chain code.
33    pub fn harden(self) -> Self {
34        DeriveJunction::Hard(self.into_inner())
35    }
36
37    /// Create a new soft (vanilla) DeriveJunction from a given, encodable, value.
38    ///
39    /// If you need a hard junction, use `hard()`.
40    pub fn soft<T: Encode>(index: T) -> Self {
41        let mut cc: [u8; JUNCTION_ID_LEN] = Default::default();
42        index.using_encoded(|data| {
43            if data.len() > JUNCTION_ID_LEN {
44                cc.copy_from_slice(&sp_crypto_hashing::blake2_256(data));
45            } else {
46                cc[0..data.len()].copy_from_slice(data);
47            }
48        });
49        DeriveJunction::Soft(cc)
50    }
51
52    /// Create a new hard (hardened) DeriveJunction from a given, encodable, value.
53    ///
54    /// If you need a soft junction, use `soft()`.
55    pub fn hard<T: Encode>(index: T) -> Self {
56        Self::soft(index).harden()
57    }
58
59    /// Consume self to return the chain code.
60    pub fn into_inner(self) -> [u8; JUNCTION_ID_LEN] {
61        match self {
62            DeriveJunction::Hard(c) | DeriveJunction::Soft(c) => c,
63        }
64    }
65
66    /// Get a reference to the inner junction id.
67    pub fn inner(&self) -> &[u8; JUNCTION_ID_LEN] {
68        match self {
69            DeriveJunction::Hard(ref c) | DeriveJunction::Soft(ref c) => c,
70        }
71    }
72
73    /// Return `true` if the junction is soft.
74    pub fn is_soft(&self) -> bool {
75        matches!(*self, DeriveJunction::Soft(_))
76    }
77
78    /// Return `true` if the junction is hard.
79    pub fn is_hard(&self) -> bool {
80        matches!(*self, DeriveJunction::Hard(_))
81    }
82}
83
84impl<T: AsRef<str>> From<T> for DeriveJunction {
85    fn from(j: T) -> DeriveJunction {
86        let j = j.as_ref();
87        let (code, hard) = if let Some(stripped) = j.strip_prefix('/') {
88            (stripped, true)
89        } else {
90            (j, false)
91        };
92
93        let res = if let Ok(n) = str::parse::<u64>(code) {
94            // number
95            DeriveJunction::soft(n)
96        } else {
97            // something else
98            DeriveJunction::soft(code)
99        };
100
101        if hard {
102            res.harden()
103        } else {
104            res
105        }
106    }
107}
