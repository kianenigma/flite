41/// An ecdsa keypair implementation.
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct Keypair(pub secp256k1::Keypair);
44
45impl Keypair {
46    /// Create an ecdsa keypair from a [`SecretUri`]. See the [`SecretUri`] docs for more.
47    ///
48    /// # Example
49    ///
50    /// ```rust
51    /// use subxt_signer::{ SecretUri, ecdsa::Keypair };
52    /// use std::str::FromStr;
53    ///
54    /// let uri = SecretUri::from_str("//Alice").unwrap();
55    /// let keypair = Keypair::from_uri(&uri).unwrap();
56    ///
57    /// keypair.sign(b"Hello world!");
58    /// ```
59    pub fn from_uri(uri: &SecretUri) -> Result<Self, Error> {
60        let SecretUri {
61            junctions,
62            phrase,
63            password,
64        } = uri;
65
66        // If the phrase is hex, convert bytes directly into a seed, ignoring password.
67        // Else, parse the phrase string taking the password into account. This is
68        // the same approach taken in sp_core::crypto::Pair::from_string_with_seed.
69        let key = if let Some(hex_str) = phrase.expose_secret().strip_prefix("0x") {
70            let seed = SecretKeyBytes::from_hex(hex_str)?;
71            Self::from_secret_key(seed)?
72        } else {
73            let phrase = bip39::Mnemonic::from_str(phrase.expose_secret())?;
74            let pass_str = password.as_ref().map(|p| p.expose_secret());
75            Self::from_phrase(&phrase, pass_str)?
76        };
77
78        // Now, use any "junctions" to derive a new key from this root key.
79        key.derive(junctions.iter().copied())
80    }
81
82    /// Create an ecdsa keypair from a BIP-39 mnemonic phrase and optional password.
83    ///
84    /// # Example
85    ///
86    /// ```rust
87    /// use subxt_signer::{ bip39::Mnemonic, ecdsa::Keypair };
88    ///
89    /// let phrase = "bottom drive obey lake curtain smoke basket hold race lonely fit walk";
90    /// let mnemonic = Mnemonic::parse(phrase).unwrap();
91    /// let keypair = Keypair::from_phrase(&mnemonic, None).unwrap();
92    ///
93    /// keypair.sign(b"Hello world!");
94    /// ```
95    pub fn from_phrase(mnemonic: &bip39::Mnemonic, password: Option<&str>) -> Result<Self, Error> {
96        let (arr, len) = mnemonic.to_entropy_array();
97        let big_seed =
98            seed_from_entropy(&arr[0..len], password.unwrap_or("")).ok_or(Error::InvalidSeed)?;
99
100        let secret_key_bytes: SecretKeyBytes = big_seed[..SECRET_KEY_LENGTH]
101            .try_into()
102            .expect("should be valid Seed");
103
104        Self::from_secret_key(secret_key_bytes)
105    }
106
107    /// Turn a 32 byte seed into a keypair.
108    ///
109    /// # Warning
110    ///
111    /// This will only be secure if the seed is secure!
112    pub fn from_secret_key(secret_key: SecretKeyBytes) -> Result<Self, Error> {
113        let secret = SecretKey::from_slice(&secret_key).map_err(|_| Error::InvalidSeed)?;
114        Ok(Self(secp256k1::Keypair::from_secret_key(
115            &Secp256k1::signing_only(),
116            &secret,
117        )))
118    }
119
120    /// Derive a child key from this one given a series of junctions.
121    ///
122    /// # Example
123    ///
124    /// ```rust
125    /// use subxt_signer::{ bip39::Mnemonic, ecdsa::Keypair, DeriveJunction };
126    ///
127    /// let phrase = "bottom drive obey lake curtain smoke basket hold race lonely fit walk";
128    /// let mnemonic = Mnemonic::parse(phrase).unwrap();
129    /// let keypair = Keypair::from_phrase(&mnemonic, None).unwrap();
130    ///
131    /// // Equivalent to the URI path '//Alice//stash':
132    /// let new_keypair = keypair.derive([
133    ///     DeriveJunction::hard("Alice"),
134    ///     DeriveJunction::hard("stash")
135    /// ]);
136    /// ```
137    pub fn derive<Js: IntoIterator<Item = DeriveJunction>>(
138        &self,
139        junctions: Js,
140    ) -> Result<Self, Error> {
141        let mut acc = self.0.secret_key().clone().secret_bytes();
142        for junction in junctions {
143            match junction {
144                DeriveJunction::Soft(_) => return Err(Error::SoftJunction),
145                DeriveJunction::Hard(junction_bytes) => {
146                    acc = ("Secp256k1HDKD", acc, junction_bytes)
147                        .using_encoded(sp_crypto_hashing::blake2_256)
148                }
149            }
150        }
151        Self::from_secret_key(acc)
152    }
153
154    /// Obtain the [`PublicKey`] part of this key pair, which can be used in calls to [`verify()`].
155    /// or otherwise converted into an address. In case of ECDSA, the public key bytes are not
156    /// equivalent to a Substrate `AccountId32`. They have to be hashed to obtain `AccountId32`.
157    pub fn public_key(&self) -> PublicKey {
158        PublicKey(self.0.public_key().serialize())
159    }
160
161    /// Obtain the [`SecretKey`] part of this key pair. This should be kept secret.
162    pub fn secret_key(&self) -> SecretKeyBytes {
163        *self.0.secret_key().as_ref()
164    }
165
166    /// Sign some message. These bytes can be used directly in a Substrate `MultiSignature::Ecdsa(..)`.
167    pub fn sign(&self, message: &[u8]) -> Signature {
168        self.sign_prehashed(&sp_crypto_hashing::blake2_256(message))
169    }
170
171    /// Signs a pre-hashed message.
172    pub fn sign_prehashed(&self, message_hash: &[u8; 32]) -> Signature {
173        let wrapped = Message::from_digest_slice(message_hash).expect("Message is 32 bytes; qed");
174        Signature(internal::sign(&self.0.secret_key(), &wrapped))
175    }
176}
177
178/// Verify that some signature for a message was created by the owner of the [`PublicKey`].
179///
180/// ```rust
181/// use subxt_signer::{ bip39::Mnemonic, ecdsa };
182///
183/// let keypair = ecdsa::dev::alice();
184/// let message = b"Hello!";
185///
186/// let signature = keypair.sign(message);
187/// let public_key = keypair.public_key();
188/// assert!(ecdsa::verify(&signature, message, &public_key));
189/// ```
190pub fn verify<M: AsRef<[u8]>>(sig: &Signature, message: M, pubkey: &PublicKey) -> bool {
191    let message_hash = sp_crypto_hashing::blake2_256(message.as_ref());
192    let wrapped = Message::from_digest_slice(&message_hash).expect("Message is 32 bytes; qed");
193
194    internal::verify(&sig.0, &wrapped, pubkey)
195}

197pub(crate) mod internal {
198    use super::*;
199
200    pub fn sign(secret_key: &secp256k1::SecretKey, message: &Message) -> [u8; 65] {
201        let recsig: RecoverableSignature =
202            Secp256k1::signing_only().sign_ecdsa_recoverable(message, secret_key);
203        let (recid, sig): (_, [u8; 64]) = recsig.serialize_compact();
204        let mut signature_bytes: [u8; 65] = [0; 65];
205        signature_bytes[..64].copy_from_slice(&sig);
206        signature_bytes[64] = (i32::from(recid) & 0xFF) as u8;
207        signature_bytes
208    }
209
210    pub fn verify(sig: &[u8; 65], message: &Message, pubkey: &PublicKey) -> bool {
211        let Ok(signature) = secp256k1::ecdsa::Signature::from_compact(&sig[..64]) else {
212            return false;
213        };
214        let Ok(public) = secp256k1::PublicKey::from_slice(&pubkey.0) else {
215            return false;
216        };
217
218        Secp256k1::verification_only()
219            .verify_ecdsa(message, &signature, &public)
220            .is_ok()
221    }
222}

224/// An error handed back if creating a keypair fails.
225#[derive(Debug, PartialEq)]
226pub enum Error {
227    /// Invalid seed.
228    InvalidSeed,
229    /// Invalid seed.
230    SoftJunction,
231    /// Invalid phrase.
232    Phrase(bip39::Error),
233    /// Invalid hex.
234    Hex(hex::FromHexError),
235}
236impl Display for Error {
237    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
238        match self {
239            Error::InvalidSeed => write!(f, "Invalid seed (was it the wrong length?)"),
240            Error::SoftJunction => write!(f, "Invalid seed for ECDSA, contained soft junction"),
241            Error::Phrase(e) => write!(f, "Cannot parse phrase: {e}"),
242            Error::Hex(e) => write!(f, "Cannot parse hex string: {e}"),
243        }
244    }
245}
246
247impl_from!(bip39::Error => Error::Phrase);
248impl_from!(hex::FromHexError => Error::Hex);
249

251impl std::error::Error for Error {}

294// Make `Keypair` usable to sign transactions in Subxt. This is optional so that
295// `subxt-signer` can be used entirely independently of Subxt.
296#[cfg(feature = "subxt")]
297mod subxt_compat {
298    use super::*;
299
300    use subxt_core::config::Config;
301    use subxt_core::tx::signer::Signer as SignerT;
302    use subxt_core::utils::{AccountId32, MultiAddress, MultiSignature};
303
304    impl From<Signature> for MultiSignature {
305        fn from(value: Signature) -> Self {
306            MultiSignature::Ecdsa(value.0)
307        }
308    }
309
310    impl From<PublicKey> for AccountId32 {
311        fn from(value: PublicKey) -> Self {
312            value.to_account_id()
313        }
314    }
315
316    impl<T> From<PublicKey> for MultiAddress<AccountId32, T> {
317        fn from(value: PublicKey) -> Self {
318            value.to_address()
319        }
320    }
321
322    impl PublicKey {
323        /// A shortcut to obtain an [`AccountId32`] from a [`PublicKey`].
324        /// We often want this type, and using this method avoids any
325        /// ambiguous type resolution issues.
326        pub fn to_account_id(self) -> AccountId32 {
327            AccountId32(sp_crypto_hashing::blake2_256(&self.0))
328        }
329        /// A shortcut to obtain a [`MultiAddress`] from a [`PublicKey`].
330        /// We often want this type, and using this method avoids any
331        /// ambiguous type resolution issues.
332        pub fn to_address<T>(self) -> MultiAddress<AccountId32, T> {
333            MultiAddress::Id(self.to_account_id())
334        }
335    }
336
337    impl<T: Config> SignerT<T> for Keypair
338    where
339        T::AccountId: From<PublicKey>,
340        T::Address: From<PublicKey>,
341        T::Signature: From<Signature>,
342    {
343        fn account_id(&self) -> T::AccountId {
344            self.public_key().into()
345        }
346
347        fn address(&self) -> T::Address {
348            self.public_key().into()
349        }
350
351        fn sign(&self, signer_payload: &[u8]) -> T::Signature {
352            self.sign(signer_payload).into()
353        }
354    }
355}

389        // Some derive junctions to check that the logic there aligns:
390        let uri_paths = ["//bar", "//0001", "//1", "//0001", "//foo//bar//wibble"];
391
392        for i in 0..2 {
393            for path in &uri_paths {
394                // Build an sp_core::Pair that includes a phrase, path and password:
395                let password = format!("Testing{i}");
396                let (_sp_pair, phrase, _seed) = SpPair::generate_with_phrase(Some(&password));
397                let uri = format!("{phrase}{path}///{password}");
398                let sp_pair = SpPair::from_string(&uri, None).expect("should be valid");
399
400                // Now build a local Keypair using the equivalent API:
401                let uri = SecretUri::from_str(&uri).expect("should be valid secret URI");
402                let pair = Keypair::from_uri(&uri).expect("should be valid");
403
404                // They should match:
405                assert_eq!(sp_pair.public().0, pair.public_key().0);
406            }
407        }
