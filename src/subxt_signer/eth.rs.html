16/// Bytes representing a private key.

19/// The public key for an [`Keypair`] key pair. This is the uncompressed variant of [`ecdsa::PublicKey`].
20pub struct PublicKey(pub [u8; 65]);
21
22impl AsRef<[u8]> for PublicKey {
23    fn as_ref(&self) -> &[u8] {
24        &self.0
25    }
26}

28/// An ethereum keypair implementation.
29#[derive(Debug, Clone, PartialEq, Eq)]
30pub struct Keypair(ecdsa::Keypair);
31
32impl From<ecdsa::Keypair> for Keypair {
33    fn from(kp: ecdsa::Keypair) -> Self {
34        Self(kp)
35    }
36}
37
38impl Keypair {
39    /// Create a keypair from a BIP-39 mnemonic phrase, optional password, account index, and
40    /// derivation type.
41    ///
42    /// **Note:** if the `std` feature is not enabled, we won't attempt to normalize the provided password
43    /// to NFKD first, and so this is your responsibility. This is not a concern if only ASCII
44    /// characters are used in the password.
45    ///
46    /// # Example
47    ///
48    /// ```rust
49    /// use subxt_signer::{ bip39::Mnemonic, eth::{ Keypair, DerivationPath } };
50    ///
51    /// let phrase = "bottom drive obey lake curtain smoke basket hold race lonely fit walk";
52    /// let mnemonic = Mnemonic::parse(phrase).unwrap();
53    /// let keypair = Keypair::from_phrase(&mnemonic, None, DerivationPath::eth(0,0)).unwrap();
54    ///
55    /// keypair.sign(b"Hello world!");
56    /// ```
57    pub fn from_phrase(
58        mnemonic: &bip39::Mnemonic,
59        password: Option<&str>,
60        derivation_path: DerivationPath,
61    ) -> Result<Self, Error> {
62        // `to_seed` isn't available unless std is enabled in bip39.
63        #[cfg(feature = "std")]
64        let seed = mnemonic.to_seed(password.unwrap_or(""));
65        #[cfg(not(feature = "std"))]
66        let seed = mnemonic.to_seed_normalized(password.unwrap_or(""));
67
68        // TODO: Currently, we use bip32 to derive private keys which under the hood uses
69        // the Rust k256 crate. We _also_ use the secp256k1 crate (which is very similar).
70        // It'd be great if we could 100% use just one of the two crypto libs. bip32 has
71        // a feature flag to use secp256k1, but it's unfortunately a different version (older)
72        // than ours.
73        let private = bip32::XPrv::derive_from_path(seed, &derivation_path.inner)
74            .map_err(|_| Error::DeriveFromPath)?;
75
76        Keypair::from_secret_key(private.to_bytes())
77    }
78
79    /// Turn a 16, 32 or 64 byte seed into a keypair.
80    ///
81    /// # Warning
82    ///
83    /// This will only be secure if the seed is secure!
84    pub fn from_seed(seed: &[u8]) -> Result<Self, Error> {
85        let private = bip32::XPrv::new(seed).map_err(|_| Error::InvalidSeed)?;
86        Keypair::from_secret_key(private.to_bytes())
87    }
88
89    /// Turn a 32 byte secret key into a keypair.
90    ///
91    /// # Warning
92    ///
93    /// This will only be secure if the secret key is secure!
94    pub fn from_secret_key(secret_key: SecretKeyBytes) -> Result<Self, Error> {
95        ecdsa::Keypair::from_secret_key(secret_key)
96            .map(Self)
97            .map_err(|_| Error::InvalidSeed)
98    }
99
100    /// Obtain the [`ecdsa::SecretKeyBytes`] of this keypair.
101    pub fn secret_key(&self) -> SecretKeyBytes {
102        self.0.secret_key()
103    }
104
105    /// Obtain the [`eth::PublicKey`] of this keypair.
106    pub fn public_key(&self) -> PublicKey {
107        let uncompressed = self.0 .0.public_key().serialize_uncompressed();
108        PublicKey(uncompressed)
109    }
110
111    /// Signs an arbitrary message payload.
112    pub fn sign(&self, signer_payload: &[u8]) -> Signature {
113        self.sign_prehashed(&keccak(signer_payload).0)
114    }
115
116    /// Signs a pre-hashed message.
117    pub fn sign_prehashed(&self, message_hash: &[u8; 32]) -> Signature {
118        Signature(self.0.sign_prehashed(message_hash).0)
119    }
120}
121/// A derivation path. This can be parsed from a valid derivation path string like
122/// `"m/44'/60'/0'/0/0"`, or we can construct one using the helpers [`DerivationPath::empty()`]
123/// and [`DerivationPath::eth()`].
124#[derive(Clone, Debug)]
125pub struct DerivationPath {
126    inner: bip32::DerivationPath,
127}
128
129impl DerivationPath {
130    /// An empty derivation path (in other words, just use the master-key as is).
131    pub fn empty() -> Self {
132        let inner = bip32::DerivationPath::from_str("m").unwrap();
133        DerivationPath { inner }
134    }
135
136    /// A BIP44 Ethereum compatible derivation using the path "m/44'/60'/account'/0/address_index".
137    ///
138    /// # Panics
139    ///
140    /// Panics if the `account` or `address_index` provided are >= 2^31.
141    pub fn eth(account: u32, address_index: u32) -> Self {
142        assert!(
143            account < bip32::ChildNumber::HARDENED_FLAG,
144            "account must be less than 2^31"
145        );
146        assert!(
147            address_index < bip32::ChildNumber::HARDENED_FLAG,
148            "address_index must be less than 2^31"
149        );
150
151        let derivation_string = format!("m/44'/60'/{account}'/0/{address_index}");
152        let inner = bip32::DerivationPath::from_str(&derivation_string).unwrap();
153        DerivationPath { inner }
154    }
155}
156
157impl FromStr for DerivationPath {
158    type Err = Error;
159    fn from_str(s: &str) -> Result<Self, Self::Err> {
160        let inner = bip32::DerivationPath::from_str(s).map_err(|_| Error::DeriveFromPath)?;
161        Ok(DerivationPath { inner })
162    }
163}

165/// A signature generated by [`Keypair::sign()`].
166#[derive(Debug, Clone, Copy, PartialEq, Eq, codec::Encode)]
167pub struct Signature(pub [u8; 65]);
168
169impl AsRef<[u8; 65]> for Signature {
170    fn as_ref(&self) -> &[u8; 65] {
171        &self.0
172    }
173}
174
175/// Verify that some signature for a message was created by the owner of the [`PublicKey`].
176///
177/// ```rust
178/// use subxt_signer::{ bip39::Mnemonic, eth };
179///
180/// let keypair = eth::dev::alith();
181/// let message = b"Hello!";
182///
183/// let signature = keypair.sign(message);
184/// let public_key = keypair.public_key();
185/// assert!(eth::verify(&signature, message, &public_key));
186/// ```
187pub fn verify<M: AsRef<[u8]>>(sig: &Signature, message: M, pubkey: &PublicKey) -> bool {
188    let message_hash = keccak(message.as_ref());
189    let wrapped =
190        Message::from_digest_slice(message_hash.as_bytes()).expect("Message is 32 bytes; qed");
191    let Ok(signature) = secp256k1::ecdsa::Signature::from_compact(&sig.as_ref()[..64]) else {
192        return false;
193    };
194    let Ok(pk) = secp256k1::PublicKey::from_slice(&pubkey.0) else {
195        return false;
196    };
197
198    secp256k1::Secp256k1::verification_only()
199        .verify_ecdsa(&wrapped, &signature, &pk)
200        .is_ok()
201}
202
203/// An error handed back if creating a keypair fails.
204#[derive(Debug, PartialEq)]
205pub enum Error {
206    /// Invalid seed.
207    InvalidSeed,
208    /// Invalid derivation path.
209    DeriveFromPath,
210}
211
212impl Display for Error {
213    fn fmt(&self, f: &mut Formatter<'_>) -> core::fmt::Result {
214        match self {
215            Error::InvalidSeed => write!(f, "Invalid seed (was it the wrong length?)"),
216            Error::DeriveFromPath => write!(
217                f,
218                "Could not derive from path; some values in the path may have been >= 2^31?"
219            ),
220        }
221    }
222}
223

225impl std::error::Error for Error {}

264#[cfg(feature = "subxt")]
265mod subxt_compat {
266    use super::*;
267    use subxt_core::config::Config;
268    use subxt_core::tx::signer::Signer as SignerT;
269    use subxt_core::utils::AccountId20;
270    use subxt_core::utils::MultiAddress;
271
272    impl<T: Config> SignerT<T> for Keypair
273    where
274        T::AccountId: From<PublicKey>,
275        T::Address: From<PublicKey>,
276        T::Signature: From<Signature>,
277    {
278        fn account_id(&self) -> T::AccountId {
279            self.public_key().into()
280        }
281
282        fn address(&self) -> T::Address {
283            self.public_key().into()
284        }
285
286        fn sign(&self, signer_payload: &[u8]) -> T::Signature {
287            self.sign(signer_payload).into()
288        }
289    }
290
291    impl PublicKey {
292        /// Obtains the public address of the account by taking the last 20 bytes
293        /// of the Keccak-256 hash of the public key.
294        pub fn to_account_id(&self) -> AccountId20 {
295            let hash = keccak(&self.0[1..]).0;
296            let hash20 = hash[12..].try_into().expect("should be 20 bytes");
297            AccountId20(hash20)
298        }
299        /// A shortcut to obtain a [`MultiAddress`] from a [`PublicKey`].
300        /// We often want this type, and using this method avoids any
301        /// ambiguous type resolution issues.
302        pub fn to_address<T>(self) -> MultiAddress<AccountId20, T> {
303            MultiAddress::Address20(self.to_account_id().0)
304        }
305    }
306
307    impl From<PublicKey> for AccountId20 {
308        fn from(value: PublicKey) -> Self {
309            value.to_account_id()
310        }
311    }
312
313    impl<T> From<PublicKey> for MultiAddress<AccountId20, T> {
314        fn from(value: PublicKey) -> Self {
315            let address: AccountId20 = value.into();
316            MultiAddress::Address20(address.0)
317        }
318    }
319}
