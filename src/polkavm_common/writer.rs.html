6#[derive(Copy, Clone, Default)]
7struct InstructionBuffer {
8    bytes: [u8; program::MAX_INSTRUCTION_LENGTH],
9    length: u8,
10}
11
12impl InstructionBuffer {
13    fn len(&self) -> usize {
14        self.length as usize
15    }
16
17    fn new(position: u32, minimum_size: u8, instruction: Instruction) -> Self {
18        let mut buffer = Self {
19            bytes: [0; program::MAX_INSTRUCTION_LENGTH],
20            length: 0,
21        };
22
23        let minimum_size = minimum_size as usize;
24        let mut length = instruction.serialize_into(position, &mut buffer.bytes);
25        if length < minimum_size {
26            let Instruction::jump(target) = instruction else {
27                // We currently only need this for jumps.
28                unreachable!();
29            };
30            assert!(minimum_size >= 1 && minimum_size <= 5);
31
32            buffer.bytes[1..minimum_size].copy_from_slice(&u32::to_le_bytes(target.wrapping_sub(position))[..minimum_size - 1]);
33            length = minimum_size;
34        }
35
36        buffer.length = length as u8;
37        buffer
38    }
39}

48impl Instruction {
49    fn target_mut(&mut self) -> Option<&mut u32> {
50        match self {
51            Instruction::jump(ref mut target)
52            | Instruction::load_imm_and_jump(_, _, ref mut target)
53            | Instruction::branch_eq_imm(_, _, ref mut target)
54            | Instruction::branch_not_eq_imm(_, _, ref mut target)
55            | Instruction::branch_less_unsigned_imm(_, _, ref mut target)
56            | Instruction::branch_less_signed_imm(_, _, ref mut target)
57            | Instruction::branch_greater_or_equal_unsigned_imm(_, _, ref mut target)
58            | Instruction::branch_greater_or_equal_signed_imm(_, _, ref mut target)
59            | Instruction::branch_less_or_equal_signed_imm(_, _, ref mut target)
60            | Instruction::branch_less_or_equal_unsigned_imm(_, _, ref mut target)
61            | Instruction::branch_greater_signed_imm(_, _, ref mut target)
62            | Instruction::branch_greater_unsigned_imm(_, _, ref mut target)
63            | Instruction::branch_eq(_, _, ref mut target)
64            | Instruction::branch_not_eq(_, _, ref mut target)
65            | Instruction::branch_less_unsigned(_, _, ref mut target)
66            | Instruction::branch_less_signed(_, _, ref mut target)
67            | Instruction::branch_greater_or_equal_unsigned(_, _, ref mut target)
68            | Instruction::branch_greater_or_equal_signed(_, _, ref mut target) => Some(target),
69            _ => None,
70        }
71    }
72}

156    fn serialize_code(&self) -> SerializedCode {
157        fn mutate<T>(slot: &mut T, value: T) -> bool
158        where
159            T: PartialEq,
160        {
161            if *slot == value {
162                false
163            } else {
164                *slot = value;
165                true
166            }
167        }
168
169        // We will need to shift all of the basic block indexes by how many entries are in our injected dispatch table.
170        let basic_block_shift = self.dispatch_table.len() as u32;
171
172        let mut instructions = Vec::with_capacity(self.dispatch_table.len() + self.code.len());
173        for (nth, symbol) in self.dispatch_table.iter().enumerate() {
174            let Some(&(target_basic_block, _)) = self.exports.iter().find(|(_, export_symbol)| symbol == export_symbol.as_bytes()) else {
175                // TODO: Return an error.
176                panic!("failed to build a dispatch table: symbol not found: {}", ProgramSymbol::new(symbol));
177            };
178
179            let minimum_size = if nth + 1 == self.dispatch_table.len() {
180                // The last entry doesn't have to be padded.
181                0
182            } else {
183                5
184            };
185
186            instructions.push(SerializedInstruction {
187                instruction: Instruction::jump(target_basic_block + basic_block_shift),
188                bytes: InstructionBuffer::default(),
189                target_nth_instruction: None,
190                position: 0,
191                minimum_size,
192            });
193        }
194
195        for instruction in &self.code {
196            let mut instruction = *instruction;
197            if let Some(target_basic_block) = instruction.target_mut() {
198                *target_basic_block += basic_block_shift;
199            }
200
201            instructions.push(SerializedInstruction {
202                instruction,
203                bytes: InstructionBuffer::default(),
204                target_nth_instruction: None,
205                position: 0,
206                minimum_size: 0,
207            });
208        }
209
210        let mut basic_block_to_instruction_index = Vec::with_capacity(self.code.len());
211        basic_block_to_instruction_index.push(0);
212
213        for (nth_instruction, entry) in instructions.iter().enumerate() {
214            if entry.instruction.opcode().starts_new_basic_block() {
215                basic_block_to_instruction_index.push(nth_instruction + 1);
216            }
217        }
218
219        let mut position: u32 = 0;
220        for (nth_instruction, entry) in instructions.iter_mut().enumerate() {
221            entry.target_nth_instruction = entry.instruction.target_mut().map(|target| {
222                let target_nth_instruction = basic_block_to_instruction_index[*target as usize];
223                // Here we change the target from a basic block index into a byte offset.
224                // This is completely inaccurate, but that's fine. This is just a guess, and we'll correct it in the next loop.
225                *target = position.wrapping_add((target_nth_instruction as i32 - nth_instruction as i32) as u32);
226                target_nth_instruction
227            });
228
229            entry.position = position;
230            entry.bytes = InstructionBuffer::new(position, entry.minimum_size, entry.instruction);
231            position = position.checked_add(entry.bytes.len() as u32).expect("too many instructions");
232        }
233
234        // Adjust offsets to other instructions until we reach a steady state.
235        loop {
236            let mut any_modified = false;
237            position = 0;
238            for nth_instruction in 0..instructions.len() {
239                let mut self_modified = mutate(&mut instructions[nth_instruction].position, position);
240                if let Some(target_nth_instruction) = instructions[nth_instruction].target_nth_instruction {
241                    let new_target = instructions[target_nth_instruction].position;
242                    let old_target = instructions[nth_instruction].instruction.target_mut().unwrap();
243                    self_modified |= mutate(old_target, new_target);
244
245                    if self_modified {
246                        instructions[nth_instruction].bytes = InstructionBuffer::new(
247                            position,
248                            instructions[nth_instruction].minimum_size,
249                            instructions[nth_instruction].instruction,
250                        );
251                    }
252                }
253
254                position = position
255                    .checked_add(instructions[nth_instruction].bytes.len() as u32)
256                    .expect("too many instructions");
257
258                any_modified |= self_modified;
259            }
260
261            if !any_modified {
262                break;
263            }
264        }
265
266        let mut jump_table_entry_size = 0;
267        let mut jump_table_entries = Vec::with_capacity(self.jump_table.len());
268        for &target in &self.jump_table {
269            let target = target + basic_block_shift;
270            let target_nth_instruction = basic_block_to_instruction_index[target as usize];
271            let offset = instructions[target_nth_instruction].position.to_le_bytes();
272            jump_table_entries.push(offset);
273            jump_table_entry_size = core::cmp::max(jump_table_entry_size, offset.iter().take_while(|&&b| b != 0).count());
274        }
275
276        let mut output = SerializedCode {
277            jump_table_entry_count: jump_table_entries.len() as u32,
278            jump_table_entry_size: jump_table_entry_size as u8,
279            jump_table: Vec::with_capacity(jump_table_entry_size * jump_table_entries.len()),
280            code: Vec::with_capacity(instructions.iter().map(|entry| entry.bytes.len()).sum()),
281            bitmask: Vec::new(),
282            exports: Vec::with_capacity(self.exports.len()),
283        };
284
285        for target in jump_table_entries {
286            output.jump_table.extend_from_slice(&target[..jump_table_entry_size]);
287        }
288
289        struct BitVec {
290            bytes: Vec<u8>,
291            current: usize,
292            bits: usize,
293        }
294
295        impl BitVec {
296            fn with_capacity(capacity: usize) -> Self {
297                BitVec {
298                    bytes: Vec::with_capacity(capacity),
299                    current: 0,
300                    bits: 0,
301                }
302            }
303
304            fn push(&mut self, value: bool) {
305                self.current |= usize::from(value) << self.bits;
306                self.bits += 1;
307                if self.bits == 8 {
308                    self.bytes.push(self.current as u8);
309                    self.current = 0;
310                    self.bits = 0;
311                }
312            }
313
314            fn finish(mut self) -> Vec<u8> {
315                while self.bits > 0 {
316                    self.push(false);
317                }
318                self.bytes
319            }
320        }
321
322        let mut bitmask = BitVec::with_capacity(output.code.capacity() / 8 + 1);
323        for entry in &instructions {
324            bitmask.push(true);
325            for _ in 1..entry.bytes.len() {
326                bitmask.push(false);
327            }
328
329            output.code.extend_from_slice(&entry.bytes);
330        }
331
332        output.bitmask = bitmask.finish();
333
334        for (target_basic_block, symbol) in &self.exports {
335            let target_basic_block = *target_basic_block as usize + basic_block_shift as usize;
336            let nth_instruction = basic_block_to_instruction_index[target_basic_block];
337            let offset = instructions[nth_instruction].position;
338            output.exports.push((offset, symbol.as_bytes().to_vec()));
339        }
340
341        if cfg!(debug_assertions) {
342            // Sanity check.
343            let mut parsed = Vec::new();
344            let mut offsets = alloc::collections::BTreeSet::new();
345
346            let parsed_instructions: Vec<_> = if self.is_64 {
347                crate::program::Instructions::new_unbounded(crate::program::ISA64_V1, &output.code, &output.bitmask, 0).collect()
348            } else {
349                crate::program::Instructions::new_unbounded(crate::program::ISA32_V1, &output.code, &output.bitmask, 0).collect()
350            };
351
352            for instruction in parsed_instructions {
353                if instruction.offset.0 as usize == output.code.len() {
354                    // Implicit trap.
355                    debug_assert!(matches!(instruction.kind, Instruction::invalid));
356                    break;
357                }
358                parsed.push(instruction);
359                offsets.insert(instruction.offset);
360            }
361
362            assert_eq!(parsed.len(), instructions.len());
363            for (nth_instruction, (mut parsed, entry)) in parsed.into_iter().zip(instructions.into_iter()).enumerate() {
364                let parsed_length = parsed.next_offset.0 - parsed.offset.0;
365                if parsed.kind != entry.instruction || entry.position != parsed.offset.0 || u32::from(entry.bytes.length) != parsed_length {
366                    panic!(
367                        concat!(
368                            "Broken serialization for instruction #{}:\n",
369                            "  Serialized:\n",
370                            "    Instruction: {:?}\n",
371                            "    Offset:      {}\n",
372                            "    Length:      {}\n",
373                            "    Bytes:       {:?}\n",
374                            "  Deserialized:\n",
375                            "    Instruction: {:?}\n",
376                            "    Offset:      {}\n",
377                            "    Length:      {}\n",
378                            "    Bytes:       {:?}\n",
379                        ),
380                        nth_instruction,
381                        entry.instruction,
382                        entry.position,
383                        entry.bytes.len(),
384                        &entry.bytes.bytes[..entry.bytes.length as usize],
385                        parsed.kind,
386                        parsed.offset.0,
387                        parsed_length,
388                        &output.code[parsed.offset.0 as usize..parsed.offset.0 as usize + parsed_length as usize],
389                    );
390                }
391
392                if let Some(target) = parsed.kind.target_mut() {
393                    assert!(offsets.contains(&ProgramCounter(*target)));
394                }
395            }
396        }
397
398        output
399    }

409    pub fn to_vec(&self) -> Vec<u8> {
410        let code = self.serialize_code();
411        let mut output = Vec::new();
412        let mut writer = Writer::new(&mut output);
413
414        writer.push_raw_bytes(&program::BLOB_MAGIC);
415        if self.is_64 {
416            writer.push_byte(program::BLOB_VERSION_V1_64);
417        } else {
418            writer.push_byte(program::BLOB_VERSION_V1_32);
419        }
420        writer.push_raw_bytes(&[0; BLOB_LEN_SIZE]);
421
422        if self.ro_data_size > 0 || self.rw_data_size > 0 || self.stack_size > 0 {
423            writer.push_section_inplace(program::SECTION_MEMORY_CONFIG, |writer| {
424                writer.push_varint(self.ro_data_size);
425                writer.push_varint(self.rw_data_size);
426                writer.push_varint(self.stack_size);
427            });
428        }
429
430        writer.push_section(program::SECTION_RO_DATA, &self.ro_data);
431        writer.push_section(program::SECTION_RW_DATA, &self.rw_data);
432        if !self.imports.is_empty() {
433            writer.push_section_inplace(program::SECTION_IMPORTS, |writer| {
434                let mut offsets_blob = Vec::new();
435                let mut symbols_blob = Vec::new();
436                for symbol in &self.imports {
437                    offsets_blob.extend_from_slice(&(symbols_blob.len() as u32).to_le_bytes());
438                    symbols_blob.extend_from_slice(symbol.as_bytes())
439                }
440
441                writer.push_varint(self.imports.len().try_into().expect("too many imports"));
442                writer.push_raw_bytes(&offsets_blob);
443                writer.push_raw_bytes(&symbols_blob);
444            });
445        }
446
447        if !code.exports.is_empty() {
448            writer.push_section_inplace(program::SECTION_EXPORTS, |writer| {
449                writer.push_varint(code.exports.len().try_into().expect("too many exports"));
450                for (offset, symbol) in code.exports {
451                    writer.push_varint(offset);
452                    writer.push_bytes_with_length(&symbol);
453                }
454            });
455        }
456
457        writer.push_section_inplace(program::SECTION_CODE_AND_JUMP_TABLE, |writer| {
458            writer.push_varint(code.jump_table_entry_count);
459            writer.push_byte(code.jump_table_entry_size);
460            writer.push_varint(code.code.len() as u32);
461            writer.push_raw_bytes(&code.jump_table);
462            writer.push_raw_bytes(&code.code);
463            writer.push_raw_bytes(&code.bitmask);
464        });
465
466        for (section, contents) in &self.custom {
467            writer.push_section(*section, contents);
468        }
469
470        writer.push_raw_bytes(&[program::SECTION_END_OF_FILE]);
471
472        let blob_len = (writer.len() as u64).to_le_bytes();
473        output[BLOB_LEN_OFFSET..BLOB_LEN_OFFSET + BLOB_LEN_SIZE].copy_from_slice(&blob_len);
474
475        output
476    }

479pub struct Writer<'a> {
480    buffer: &'a mut Vec<u8>,
481}
482
483impl<'a> Writer<'a> {
484    pub fn new(buffer: &'a mut Vec<u8>) -> Self {
485        Self { buffer }
486    }
487
488    fn push_section_inplace(&mut self, section: u8, callback: impl FnOnce(&mut Self)) -> Range<usize> {
489        let section_position = self.buffer.len();
490        self.buffer.push(section);
491
492        // Reserve the space for the length varint.
493        let length_position = self.buffer.len();
494        self.push_raw_bytes(&[0xff_u8; crate::varint::MAX_VARINT_LENGTH]);
495
496        let payload_position = self.buffer.len();
497        callback(self);
498
499        let payload_length: u32 = (self.buffer.len() - payload_position).try_into().expect("section size overflow");
500        if payload_length == 0 {
501            // Nothing was written by the callback. Skip writing the section.
502            self.buffer.truncate(section_position);
503            return 0..0;
504        }
505
506        // Write the length varint.
507        let length_length = crate::varint::write_varint(payload_length, &mut self.buffer[length_position..]);
508
509        // Drain any excess length varint bytes.
510        self.buffer
511            .drain(length_position + length_length..length_position + crate::varint::MAX_VARINT_LENGTH);
512
513        length_position + length_length..self.buffer.len()
514    }
515
516    fn push_section(&mut self, section: u8, contents: &[u8]) {
517        if contents.is_empty() {
518            return;
519        }
520
521        self.push_byte(section);
522        self.push_varint(contents.len().try_into().expect("section size overflow"));
523        self.push_raw_bytes(contents);
524    }
525
526    pub fn push_raw_bytes(&mut self, slice: &[u8]) {
527        self.buffer.extend_from_slice(slice);
528    }
529
530    pub fn push_byte(&mut self, byte: u8) {
531        self.buffer.push(byte);
532    }
533
534    pub fn push_u32(&mut self, value: u32) {
535        self.push_raw_bytes(&value.to_le_bytes());
536    }
537
538    pub fn push_varint(&mut self, value: u32) {
539        let mut buffer = [0xff_u8; crate::varint::MAX_VARINT_LENGTH];
540        let length = crate::varint::write_varint(value, &mut buffer);
541        self.push_raw_bytes(&buffer[..length]);
542    }
543
544    pub fn push_bytes_with_length(&mut self, slice: &[u8]) {
545        self.push_varint(slice.len().try_into().expect("length overflow"));
546        self.push_raw_bytes(slice);
547    }
548
549    pub fn len(&self) -> usize {
550        self.buffer.len()
551    }
552
553    pub fn is_empty(&self) -> bool {
554        self.buffer.is_empty()
555    }
