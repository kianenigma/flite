16//! Managed Collective Content Pallet
17//!
18//! The pallet provides the functionality to store different types of content. This would typically
19//! be used by an on-chain collective, such as the Polkadot Alliance or Ambassador Program.
20//!
21//! The pallet stores content as an [OpaqueCid], which should correspond to some off-chain hosting
22//! service, such as IPFS, and contain any type of data. Each type of content has its own origin
23//! from which it can be managed. The origins are configurable in the runtime. Storing content does
24//! not require a deposit, as it is expected to be managed by a trusted collective.
25//!
26//! Content types:
27//!
28//! - Collective [charter](pallet::Charter): A single document (`OpaqueCid`) managed by
29//!   [CharterOrigin](pallet::Config::CharterOrigin).
30//! - Collective [announcements](pallet::Announcements): A list of announcements managed by
31//!   [AnnouncementOrigin](pallet::Config::AnnouncementOrigin).

50/// IPFS compatible CID.
51// Worst case 2 bytes base and codec, 2 bytes hash type and size, 64 bytes hash digest.
52pub type OpaqueCid = BoundedVec<u8, ConstU32<68>>;

54#[frame_support::pallet]
55pub mod pallet {

61	/// The in-code storage version.
62	const STORAGE_VERSION: StorageVersion = StorageVersion::new(0);
63
64	#[pallet::pallet]
65	#[pallet::storage_version(STORAGE_VERSION)]
66	pub struct Pallet<T, I = ()>(PhantomData<(T, I)>);
67
68	/// The module configuration trait.
69	#[pallet::config]
70	pub trait Config<I: 'static = ()>: frame_system::Config {
71		/// The overarching event type.
72		type RuntimeEvent: From<Event<Self, I>>
73			+ IsType<<Self as frame_system::Config>::RuntimeEvent>;
74
75		/// Default lifetime for an announcement before it expires.
76		type AnnouncementLifetime: Get<BlockNumberFor<Self>>;
77
78		/// The origin to control the collective announcements.
79		type AnnouncementOrigin: EnsureOrigin<Self::RuntimeOrigin>;
80
81		/// Maximum number of announcements in the storage.
82		#[pallet::constant]
83		type MaxAnnouncements: Get<u32>;
84
85		/// The origin to control the collective charter.
86		type CharterOrigin: EnsureOrigin<Self::RuntimeOrigin>;
87
88		/// Weight information needed for the pallet.
89		type WeightInfo: WeightInfo;
90	}
91
92	#[pallet::error]
93	pub enum Error<T, I = ()> {
94		/// The announcement is not found.
95		MissingAnnouncement,
96		/// Number of announcements exceeds `MaxAnnouncementsCount`.
97		TooManyAnnouncements,
98		/// Cannot expire in the past.
99		InvalidExpiration,
100	}
101
102	#[pallet::event]
103	#[pallet::generate_deposit(pub(super) fn deposit_event)]
104	pub enum Event<T: Config<I>, I: 'static = ()> {
105		/// A new charter has been set.
106		NewCharterSet { cid: OpaqueCid },
107		/// A new announcement has been made.
108		AnnouncementAnnounced { cid: OpaqueCid, expire_at: BlockNumberFor<T> },
109		/// An on-chain announcement has been removed.
110		AnnouncementRemoved { cid: OpaqueCid },
111	}
112
113	/// The collective charter.
114	#[pallet::storage]
115	pub type Charter<T: Config<I>, I: 'static = ()> = StorageValue<_, OpaqueCid, OptionQuery>;
116
117	/// The collective announcements.
118	#[pallet::storage]
119	pub type Announcements<T: Config<I>, I: 'static = ()> =
120		CountedStorageMap<_, Blake2_128Concat, OpaqueCid, BlockNumberFor<T>, OptionQuery>;
121
122	#[pallet::call]
123	impl<T: Config<I>, I: 'static> Pallet<T, I> {
124		/// Set the collective charter.
125		///
126		/// Parameters:
127		/// - `origin`: Must be the [Config::CharterOrigin].
128		/// - `cid`: [CID](super::OpaqueCid) of the IPFS document of the collective charter.
129		#[pallet::call_index(0)]
130		#[pallet::weight(T::WeightInfo::set_charter())]
131		pub fn set_charter(origin: OriginFor<T>, cid: OpaqueCid) -> DispatchResult {
132			T::CharterOrigin::ensure_origin(origin)?;
133
134			Charter::<T, I>::put(&cid);
135
136			Self::deposit_event(Event::<T, I>::NewCharterSet { cid });
137			Ok(())
138		}
139
140		/// Publish an announcement.
141		///
142		/// Parameters:
143		/// - `origin`: Must be the [Config::AnnouncementOrigin].
144		/// - `cid`: [CID](super::OpaqueCid) of the IPFS document to announce.
145		/// - `maybe_expire`: Expiration block of the announcement. If `None`
146		///   [`Config::AnnouncementLifetime`]
147		/// used as a default.
148		#[pallet::call_index(1)]
149		#[pallet::weight(T::WeightInfo::announce())]
150		pub fn announce(
151			origin: OriginFor<T>,
152			cid: OpaqueCid,
153			maybe_expire: Option<DispatchTime<BlockNumberFor<T>>>,
154		) -> DispatchResult {
155			T::AnnouncementOrigin::ensure_origin(origin)?;
156
157			let now = frame_system::Pallet::<T>::block_number();
158			let expire_at = maybe_expire
159				.map_or(now.saturating_add(T::AnnouncementLifetime::get()), |e| e.evaluate(now));
160			ensure!(expire_at > now, Error::<T, I>::InvalidExpiration);
161			ensure!(
162				T::MaxAnnouncements::get() > <Announcements<T, I>>::count(),
163				Error::<T, I>::TooManyAnnouncements
164			);
165
166			<Announcements<T, I>>::insert(cid.clone(), expire_at);
167
168			Self::deposit_event(Event::<T, I>::AnnouncementAnnounced { cid, expire_at });
169			Ok(())
170		}
171
172		/// Remove an announcement.
173		///
174		/// Transaction fee refunded for expired announcements.
175		///
176		/// Parameters:
177		/// - `origin`: Must be the [Config::AnnouncementOrigin] or signed for expired
178		///   announcements.
179		/// - `cid`: [CID](super::OpaqueCid) of the IPFS document to remove.
180		#[pallet::call_index(2)]
181		#[pallet::weight(T::WeightInfo::remove_announcement())]
182		pub fn remove_announcement(
183			origin: OriginFor<T>,
184			cid: OpaqueCid,
185		) -> DispatchResultWithPostInfo {
186			let maybe_who = match T::AnnouncementOrigin::try_origin(origin) {
187				Ok(_) => None,
188				Err(origin) => Some(ensure_signed(origin)?),
189			};
190			let expire_at = <Announcements<T, I>>::get(cid.clone())
191				.ok_or(Error::<T, I>::MissingAnnouncement)?;
192			let now = frame_system::Pallet::<T>::block_number();
193			ensure!(maybe_who.is_none() || now >= expire_at, BadOrigin);
194
195			<Announcements<T, I>>::remove(cid.clone());
196
197			Self::deposit_event(Event::<T, I>::AnnouncementRemoved { cid });
198
199			if now >= expire_at {
200				return Ok(Pays::No.into())
201			}
202			Ok(Pays::Yes.into())
203		}
204	}
