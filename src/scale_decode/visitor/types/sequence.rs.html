24/// This enables a visitor to decode items from a sequence type.
25pub struct Sequence<'scale, 'resolver, R: TypeResolver> {
26    bytes: &'scale [u8],
27    // Mostly we just delegate to our Array logic for working with sequences.
28    // The only thing we need to do otherwise is decode the compact encoded
29    // length from the beginning and keep track of the bytes including that.
30    values: Array<'scale, 'resolver, R>,
31}
32
33impl<'scale, 'resolver, R: TypeResolver> Sequence<'scale, 'resolver, R> {
34    pub(crate) fn new(
35        bytes: &'scale [u8],
36        type_id: R::TypeId,
37        types: &'resolver R,
38    ) -> Result<Sequence<'scale, 'resolver, R>, DecodeError> {
39        // Sequences are prefixed with their length in bytes. Make a note of this,
40        // as well as the number of bytes
41        let item_bytes = &mut &*bytes;
42        let len = <Compact<u64>>::decode(item_bytes)?.0 as usize;
43
44        Ok(Sequence { bytes, values: Array::new(item_bytes, type_id, len, types) })
45    }
46    /// Skip over all bytes associated with this sequence. After calling this,
47    /// [`Self::bytes_from_undecoded()`] will represent the bytes after this sequence.
48    pub fn skip_decoding(&mut self) -> Result<(), DecodeError> {
49        self.values.skip_decoding()
50    }
51    /// The bytes representing this sequence and anything following it.
52    pub fn bytes_from_start(&self) -> &'scale [u8] {
53        self.bytes
54    }
55    /// The bytes that have not yet been decoded in this sequence (this never includes the
56    /// compact length preceeding the sequence items) and anything following it.
57    pub fn bytes_from_undecoded(&self) -> &'scale [u8] {
58        self.values.bytes_from_undecoded()
59    }
60    /// The number of un-decoded items remaining in this sequence.
61    pub fn remaining(&self) -> usize {
62        self.values.remaining()
63    }
64    /// Decode an item from the sequence by providing a visitor to handle it.
65    pub fn decode_item<V: Visitor<TypeResolver = R>>(
66        &mut self,
67        visitor: V,
68    ) -> Option<Result<V::Value<'scale, 'resolver>, V::Error>> {
69        self.values.decode_item(visitor)
70    }
71}
72
73// Iterating returns a representation of each field in the tuple type.
74impl<'scale, 'resolver, R: TypeResolver> Iterator for Sequence<'scale, 'resolver, R> {
75    type Item = Result<SequenceItem<'scale, 'resolver, R>, DecodeError>;
76    fn next(&mut self) -> Option<Self::Item> {
77        Some(self.values.next()?.map(|item| SequenceItem { item }))
78    }
79}

81/// A single item in the Sequence.
82pub struct SequenceItem<'scale, 'resolver, R: TypeResolver> {
83    // Same implementation under the hood as ArrayItem:
84    item: ArrayItem<'scale, 'resolver, R>,
85}
86
87impl<'scale, 'resolver, R> Copy for SequenceItem<'scale, 'resolver, R>
88where
89    R: TypeResolver,
90    R::TypeId: Copy,
91{
92}
93impl<'scale, 'resolver, R: TypeResolver> Clone for SequenceItem<'scale, 'resolver, R> {
94    fn clone(&self) -> Self {
95        SequenceItem { item: self.item.clone() }
96    }
97}
98
99impl<'scale, 'resolver, R: TypeResolver> SequenceItem<'scale, 'resolver, R> {
100    /// The bytes associated with this item.
101    pub fn bytes(&self) -> &'scale [u8] {
102        self.item.bytes()
103    }
104    /// The type ID associated with this item.
105    pub fn type_id(&self) -> &R::TypeId {
106        self.item.type_id()
107    }
108    /// Decode this item using a visitor.
109    pub fn decode_with_visitor<V: Visitor<TypeResolver = R>>(
110        &self,
111        visitor: V,
112    ) -> Result<V::Value<'scale, 'resolver>, V::Error> {
113        self.item.decode_with_visitor(visitor)
114    }
115    /// Decode this item into a specific type via [`DecodeAsType`].
116    pub fn decode_as_type<T: DecodeAsType>(&self) -> Result<T, crate::Error> {
117        self.item.decode_as_type()
118    }
119}

125        &mut self,
